/**
 * Obtener nombre de cliente desde el cache
 * @param {string} idCliente - ID o celular del cliente
 * @returns {string} Nombre del cliente o texto por defecto
 */
function obtenerNombreCliente(idCliente) {
  if (!idCliente) return 'Sin identificar';
  const idStr = String(idCliente).trim();
  
  // Cliente genérico
  if (idStr === '2222222') return 'Consumidor Final';
  
  // Buscar en cache
  const cliente = clientesCache.find(function(c) {
    return String(c.id).trim() === idStr || String(c.celular).trim() === idStr;
  });
  
  if (cliente && cliente.nombre) return cliente.nombre;
  return 'Cliente Nuevo';
}

/**
 * Obtener datos completos del cliente desde el cache
 * @param {string} idCliente - ID o celular del cliente
 * @returns {Object} Objeto con nombre, celular y email
 */
function obtenerDatosClienteLocal(idCliente) {
  if (!idCliente) return { nombre: 'Sin identificar', celular: '-', email: '-' };
  const idStr = String(idCliente).trim();
  
  // Cliente genérico
  if (idStr === '2222222') {
    return { nombre: 'Consumidor Final', celular: '-', email: '-' };
  }
  
  // Buscar en cache
  const cliente = clientesCache.find(function(c) {
    return String(c.id).trim() === idStr || String(c.celular).trim() === idStr;
  });
  
  if (cliente) {
    return {
      nombre: cliente.nombre || 'Sin nombre',
      celular: cliente.celular || '-',
      email: cliente.email || '-'
    };
  }
  
  return { nombre: 'Cliente Nuevo', celular: idStr, email: '-' };
}

function obtenerFechaHoyColombia() {
  const ahora = new Date();
  const offsetColombiaMinutos = -5 * 60;
  const offsetLocalMinutos = ahora.getTimezoneOffset();
  const diferenciaMinutos = offsetLocalMinutos + offsetColombiaMinutos;
  
  const fechaColombia = new Date(ahora.getTime() - diferenciaMinutos * 60 * 1000);
  
  const year = fechaColombia.getFullYear();
  const month = String(fechaColombia.getMonth() + 1).padStart(2, '0');
  const day = String(fechaColombia.getDate()).padStart(2, '0');
  
  return year + '-' + month + '-' + day;
}

function aplicarPersonalizacion() {

  if (configuracion.colorSecundario) {
    document.documentElement.style.setProperty('--secondary', configuracion.colorSecundario);
    document.documentElement.style.setProperty('--secondary-dark', ajustarColor(configuracion.colorSecundario, -20));
    document.documentElement.style.setProperty('--bg-sidebar', configuracion.colorSecundario);
  }
  
  if (configuracion.colorAcento) {
    document.documentElement.style.setProperty('--warning', configuracion.colorAcento);
  }
  
  const loginHeaderP = document.querySelector('.login-header p');
  if (loginHeaderP && configuracion.nombreEmpresa) {
    loginHeaderP.textContent = configuracion.nombreEmpresa + ' - Dashboard';
  }
  
  const brandName = document.querySelector('.brand-name');
  if (brandName && configuracion.nombreEmpresa) {
    brandName.textContent = configuracion.nombreEmpresa;
  }
  
  console.log('✅ Personalización aplicada');
}

function ajustarColor(hex, porcentaje) {
  if (!hex || hex.length < 7) return hex;
  
  let r = parseInt(hex.slice(1, 3), 16);
  let g = parseInt(hex.slice(3, 5), 16);
  let b = parseInt(hex.slice(5, 7), 16);
  
  r = Math.max(0, Math.min(255, r + (r * porcentaje / 100)));
  g = Math.max(0, Math.min(255, g + (g * porcentaje / 100)));
  b = Math.max(0, Math.min(255, b + (b * porcentaje / 100)));
  
  const rr = Math.round(r).toString(16).padStart(2, '0');
  const gg = Math.round(g).toString(16).padStart(2, '0');
  const bb = Math.round(b).toString(16).padStart(2, '0');
  
  return '#' + rr + gg + bb;
}

function verificarSesionDashboard() {
  const sesionGuardada = localStorage.getItem('usuarioDashboard');
  if (sesionGuardada) {
    try {
      usuarioDashboard = JSON.parse(sesionGuardada);
      if (usuarioDashboard && usuarioDashboard.esInterno) {
        console.log('👤 Sesión recuperada para:', usuarioDashboard.nombre);
        iniciarDashboard();
      } else {
        mostrarLoginDashboard();
      }
    } catch (error) {
      console.error('❌ Error al recuperar sesión:', error);
      mostrarLoginDashboard();
    }
  } else {
    mostrarLoginDashboard();
  }
}

function mostrarLoginDashboard() {
  document.getElementById('loginDashboard').classList.add('active');
  document.getElementById('dashboardMain').classList.remove('active');
}

function loginDashboard(event) {
  event.preventDefault();
  
  const email = document.getElementById('loginEmail').value.trim();
  const password = document.getElementById('loginPassword').value;
  
  if (!email || !password) {
    mostrarErrorLogin('Por favor complete todos los campos');
    return;
  }
  
  mostrarLoading(true);
  
  google.script.run
    .withSuccessHandler(function(result) {
      mostrarLoading(false);
      
      if (result.success) {
        usuarioDashboard = result.user;
        localStorage.setItem('usuarioDashboard', JSON.stringify(usuarioDashboard));
        
        document.getElementById('loginEmail').value = '';
        document.getElementById('loginPassword').value = '';
        document.getElementById('loginError').classList.remove('show');
        
        console.log('✅ Login exitoso:', usuarioDashboard.nombre, '| Super Admin:', usuarioDashboard.esSuperAdmin);
        iniciarDashboard();
      } else {
        mostrarErrorLogin(result.message || 'Credenciales incorrectas');
      }
    })
    .withFailureHandler(function(error) {
      mostrarLoading(false);
      console.error('❌ Error en login:', error);
      mostrarErrorLogin('Error de conexión. Intente nuevamente.');
    })
    .validarLoginDashboard(email, password);
}

function mostrarErrorLogin(mensaje) {
  const errorDiv = document.getElementById('loginError');
  errorDiv.textContent = mensaje;
  errorDiv.classList.add('show');
}

function configurarVisibilidadPorRol() {
  const rol = (usuarioDashboard.rol || '').toLowerCase();
  
  const statVentas = document.querySelector('.stat-card.stat-sales');
  if (statVentas) {
    if (rol === 'cocina' || rol === 'domiciliario' || rol === 'repartidor' || rol === 'cajero') {
      statVentas.style.display = 'none';
    } else {
      statVentas.style.display = 'flex';
    }
  }
  
  const navItemAdmin = document.querySelector('[data-vista="admin"]');
  const navItemCocina = document.querySelector('[data-vista="cocina"]');
  const navItemDomicilio = document.querySelector('[data-vista="domicilio"]');
  
  if (rol === 'cocina') {
    if (navItemAdmin) navItemAdmin.style.display = 'none';
    if (navItemCocina) navItemCocina.style.display = 'flex';
    if (navItemDomicilio) navItemDomicilio.style.display = 'none';
  } else if (rol === 'domiciliario' || rol === 'repartidor') {
    if (navItemAdmin) navItemAdmin.style.display = 'none';
    if (navItemCocina) navItemCocina.style.display = 'none';
    if (navItemDomicilio) navItemDomicilio.style.display = 'flex';
  } else {
    if (navItemAdmin) navItemAdmin.style.display = 'flex';
    if (navItemCocina) navItemCocina.style.display = 'flex';
    if (navItemDomicilio) navItemDomicilio.style.display = 'flex';
  }
}

function determinarVistaInicial() {
  const rol = (usuarioDashboard.rol || '').toLowerCase();
  
  if (rol === 'cocina') {
    cambiarVista('cocina', document.querySelector('[data-vista="cocina"]'));
  } else if (rol === 'domiciliario' || rol === 'repartidor') {
    cambiarVista('domicilio', document.querySelector('[data-vista="domicilio"]'));
  } else {
    cambiarVista('admin', document.querySelector('[data-vista="admin"]'));
  }
}

function renderizarTablaPedidos() {
  const tbody = document.getElementById('tablaPedidos');
  const emptyState = document.getElementById('emptyState');
  const countSpan = document.getElementById('pedidosCount');
  
  countSpan.textContent = '(' + pedidosCache.length + ')';
  
  if (pedidosCache.length === 0) {
    tbody.innerHTML = '';
    emptyState.style.display = 'block';
    return;
  }
  
  emptyState.style.display = 'none';
  
  const moneda = configuracion.moneda || '$';
  let html = '';
  
  pedidosCache.forEach(function(pedido) {
    const fecha = new Date(pedido.fechaPedido);
    const fechaFormateada = fecha.toLocaleDateString('es-CO', { day: '2-digit', month: 'short' });
    const horaFormateada = fecha.toLocaleTimeString('es-CO', { hour: '2-digit', minute: '2-digit' });
    
    const badgeEtapa = obtenerBadgeEtapa(pedido.estadoLogistico);
    const badgeEstado = obtenerBadgeEstado(pedido.estadoCierre);

    html += '<td data-label="ID">';
    html += '<strong>' + pedido.idPedido + '</strong>';
    html += '<span style="margin-left: 8px;">' + badgeEtapa.replace('class="badge ', 'class="badge ') + '</span>';
    html += '</td>';
    html += '<td data-label="Cliente">';
    html += '<div style="font-weight: 500;">' + nombreCliente + '</div>';
    html += '<div style="font-size: 12px; color: var(--text-secondary);">' + (pedido.telefonoCliente || pedido.telefonoContacto || '-') + '</div>';
    html += '</td>';
    html += '<td data-label="Fecha">';
    html += '<div>' + fechaFormateada + '</div>';
    html += '<div style="font-size: 12px; color: var(--text-secondary);">' + horaFormateada + '</div>';
    html += '</td>';
    html += '<td data-label="Etapa">' + badgeEtapa + '</td>';
    html += '<td data-label="Total"><strong style="color: var(--primary); font-size: 16px;">' + moneda + formatearPrecio(pedido.total) + '</strong></td>';
    html += '<td data-label="Entrega">';
    html += '<div style="font-size: 13px;">' + (pedido.tipoEntrega || '-') + '</div>';
    html += '<div style="font-size: 11px; color: var(--text-secondary); max-width: 150px; white-space: nowrap; overflow: hidden; text-overflow: ellipsis;" title="' + (pedido.direccionEntrega || '') + '">' + (pedido.direccionEntrega || '-') + '</div>';

    html += 'Ver / Gestionar';

  });

// ============================================
function actualizarEstadisticas() {
  const stats = {
    pendientes: pedidosCache.filter(function(p) { return p.estadoLogistico === 'Solicitado'; }).length,
    preparacion: pedidosCache.filter(function(p) { return p.estadoLogistico === 'Recibido'; }).length,
    enviados: pedidosCache.filter(function(p) { return p.estadoLogistico === 'Enviado'; }).length,
    entregados: pedidosCache.filter(function(p) { return p.estadoLogistico === 'Entregado'; }).length,
    ventas: pedidosCache.filter(function(p) { return p.estadoCierre !== 'Cancelado'; }).reduce(function(sum, p) { return sum + (p.total || 0); }, 0)
  };
  
  const moneda = configuracion.moneda || '$';
  
  document.getElementById('statPendientes').textContent = stats.pendientes;
  document.getElementById('statPreparacion').textContent = stats.preparacion;
  document.getElementById('statEnviados').textContent = stats.enviados;
  document.getElementById('statEntregados').textContent = stats.entregados;
  document.getElementById('statVentas').textContent = moneda + formatearPrecio(stats.ventas);
}

function actualizarBadges() {
  document.getElementById('badgeAdmin').textContent = pedidosCache.length;
  document.getElementById('badgeCocina').textContent = pedidosCache.filter(function(p) { return ['Solicitado', 'Recibido'].includes(p.estadoLogistico); }).length;
  document.getElementById('badgeDomicilio').textContent = pedidosCache.filter(function(p) { return p.estadoLogistico === 'Enviado'; }).length;
}

/**
 * Renderizar historial de cambios de estado (Timeline)
 */
function renderizarHistorialEstados(historial) {
  const container = document.getElementById('historialEstados');
  
  if (!container) {
    // Si no existe el contenedor, crearlo dinámicamente en la sección de acciones
    const accionesSection = document.getElementById('accionesEstado');
    if (accionesSection) {
      const historialDiv = document.createElement('div');
      historialDiv.id = 'historialEstados';
      historialDiv.className = 'historial-estados-container';
      accionesSection.insertBefore(historialDiv, accionesSection.firstChild);
    }
  }
  
  const historialContainer = document.getElementById('historialEstados');
  if (!historialContainer) return;
  
  if (!historial || historial.length === 0) {
    historialContainer.innerHTML = '';
    historialContainer.style.display = 'none';
    return;
  }
  
  historialContainer.style.display = 'block';
  
  let html = '<h4 style="font-size: 14px; margin-bottom: 12px; color: var(--text-secondary);">📝 Historial de Estados</h4><div class="timeline-container">';
  
  historial.forEach(function(item, index) {
    const fecha = new Date(item.fechaHora);
    const fechaFormateada = fecha.toLocaleDateString('es-CO', { day: '2-digit', month: 'short' });
    const horaFormateada = fecha.toLocaleTimeString('es-CO', { hour: '2-digit', minute: '2-digit' });
    
    const iconoEstado = obtenerIconoEstado(item.estadoNuevo);
    const esUltimo = index === historial.length - 1;
    
    html += '<div class="timeline-item ' + (esUltimo ? 'timeline-active' : '') + '">';
    html += '<div class="timeline-marker"><span>' + iconoEstado + '</span></div>';
    html += '<div class="timeline-content"><div class="timeline-header"><strong>' + item.estadoNuevo + '</strong><span class="timeline-time">' + fechaFormateada + ' ' + horaFormateada + '</span></div><div class="timeline-details"><span>Por: ' + (item.nombreUsuario || 'Sistema') + '</span>' + (item.rol ? '<span class="timeline-role">(' + item.rol + ')</span>' : '') + '</div>' + (item.notas ? '<div class="timeline-notes">' + item.notas + '</div>' : '') + '</div></div>';
  });
  
  html += '</div>';

/**
 * Copiar dirección al portapapeles
 * Utiliza la API Clipboard con fallback para navegadores antiguos
 */
function copiarDireccion() {
  const direccion = document.getElementById('modalDireccion').textContent;
  
  if (!direccion || direccion === '-' || direccion.trim() === '') {
    mostrarToast('warning', 'Sin dirección', 'Este pedido no tiene dirección de entrega');
    return;
  }
  
  // Verificar disponibilidad de la API Clipboard
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(direccion)
      .then(function() {
        mostrarToast('success', '📋 Copiado', 'Dirección copiada al portapapeles');
        reproducirSonidoCopiado();
      })
      .catch(function(err) {
        console.error('Error al copiar al portapapeles:', err);
        fallbackCopiar(direccion);
      });
  } else {
    // Fallback para navegadores sin soporte de Clipboard API
    fallbackCopiar(direccion);
  }
}

/**
 * Fallback para copiar usando método antiguo
 * Necesario para navegadores como Safari antiguo o contextos no seguros
 */
function fallbackCopiar(texto) {
  var textarea = document.createElement('textarea');
  textarea.value = texto;
  textarea.style.position = 'fixed';
  textarea.style.opacity = '0';
  textarea.style.left = '-9999px';
  document.body.appendChild(textarea);
  textarea.select();
  
  try {
    var exitoso = document.execCommand('copy');
    if (exitoso) {
      mostrarToast('success', '📋 Copiado', 'Dirección copiada al portapapeles');
      reproducirSonidoCopiado();
    } else {
      mostrarToast('error', 'Error', 'No se pudo copiar. Copia manualmente.');
    }
  } catch (err) {
    console.error('Fallback copy failed:', err);
    mostrarToast('error', 'Error', 'No se pudo copiar la dirección');
  }
  
  document.body.removeChild(textarea);
}

/**
 * Reproducir sonido de confirmación (opcional)
 * Proporciona feedback auditivo adicional
 */
function reproducirSonidoCopiado() {
  try {
    var audioContext = new (window.AudioContext || window.webkitAudioContext)();
    var oscillator = audioContext.createOscillator();
    var gainNode = audioContext.createGain();
    
    oscillator.connect(gainNode);
    gainNode.connect(audioContext.destination);
    
    oscillator.type = 'sine';
    oscillator.frequency.setValueAtTime(800, audioContext.currentTime);
    oscillator.frequency.exponentialRampToValueAtTime(1200, audioContext.currentTime + 0.1);
    
    gainNode.gain.setValueAtTime(0.3, audioContext.currentTime);
    gainNode.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.2);
    
    oscillator.start(audioContext.currentTime);
    oscillator.stop(audioContext.currentTime + 0.2);
  } catch (e) {
    // Si el audio falla, no hacer nada - el toast es suficiente
  }
}

/**
 * Abrir Google Maps con la dirección de entrega
 * Utiliza el formato de URL universal de Google Maps
 */
function abrirGPS() {
  var direccion = document.getElementById('modalDireccion').textContent;
  
  if (!direccion || direccion === '-' || direccion.trim() === '') {
    mostrarToast('warning', 'Sin dirección', 'Este pedido no tiene dirección de entrega');
    return;
  }
  
  // Limpiar la dirección: eliminar caracteres problemáticos
  var direccionLimpia = direccion.replace(/\r?\n|\r/g, ' ').replace(/\s+/g, ' ').trim();
  
  // Codificar la dirección para uso en URL
  var direccionCodificada = encodeURIComponent(direccionLimpia);
  
  // Construir URL de Google Maps
  var urlMaps = 'https://www.google.com/maps/search/?api=1&query=' + direccionCodificada;
  
  // Abrir en nueva pestaña
  window.open(urlMaps, '_blank');
  
  // Feedback visual
  mostrarToast('info', '📍 Navegando', 'Abriendo Google Maps con la dirección');
}

// Variables globales para gestión de usuarios
var usuariosCache = [];
var usuarioEditando = null;
var usuarioAEliminar = null;

/**
 * Inicializar módulo de usuarios cuando el usuario es Admin
 */
function inicializarModuloUsuarios() {
  // Verificar si el usuario es Administrador
  if (usuarioDashboard && usuarioDashboard.esSuperAdmin) {
    var btnNavUsuarios = document.getElementById('btnNavUsuarios');
    if (btnNavUsuarios) {
      btnNavUsuarios.style.display = 'flex';
    }
    console.log('✅ Módulo de usuarios habilitado para Admin');
  }
}

/**
 * Cargar lista de usuarios desde el backend
 */
function cargarListaUsuarios() {
  console.log('🔄 Cargando lista de usuarios...');

  var tbodyUsuarios = document.getElementById('tbodyUsuarios');
  var listaUsuariosMobile = document.getElementById('listaUsuariosMobile');
  var emptyStateUsuarios = document.getElementById('emptyStateUsuarios');

  // Mostrar loading
  if (tbodyUsuarios) {
    tbodyUsuarios.innerHTML = '<tr class="loading-row"><td colspan="5"><div class="table-loading"><div class="spinner"></div><span>Cargando usuarios...</span></div></td></tr>';
  }
  if (listaUsuariosMobile) {
    listaUsuariosMobile.innerHTML = '<div class="table-loading"><div class="spinner"></div><span>Cargando usuarios...</span></div>';
  }
  if (emptyStateUsuarios) {
    emptyStateUsuarios.style.display = 'none';
  }

  google.script.run
    .withSuccessHandler(function(response) {
      console.log('📥 Respuesta usuarios:', response);

      if (response.success) {
        usuariosCache = response.data || [];
        renderizarTablaUsuarios();
        actualizarBadgeUsuarios();
        mostrarToast('success', 'Usuarios cargados', 'Se encontraron ' + usuariosCache.length + ' usuarios');
      } else {
        mostrarToast('error', 'Error', response.error || 'No se pudieron cargar los usuarios');
        if (tbodyUsuarios) {
          tbodyUsuarios.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--danger);">Error al cargar usuarios</td></tr>';
        }
      }
    })
    .withFailureHandler(function(error) {
      console.error('❌ Error al cargar usuarios:', error);
      mostrarToast('error', 'Error de conexión', 'No se pudo conectar con el servidor');
      if (tbodyUsuarios) {
        tbodyUsuarios.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--danger);">Error de conexión</td></tr>';
      }
    })
    .adminGestionarUsuarios('leer', {}, usuarioDashboard);
}

/**
 * Renderizar tabla de usuarios (Desktop y Mobile)
 */
function renderizarTablaUsuarios() {
  var tbodyUsuarios = document.getElementById('tbodyUsuarios');
  var listaUsuariosMobile = document.getElementById('listaUsuariosMobile');
  var emptyStateUsuarios = document.getElementById('emptyStateUsuarios');

  // Aplicar filtro de búsqueda si existe
  var busqueda = document.getElementById('busquedaUsuarios');
  var termino = busqueda ? busqueda.value.toLowerCase().trim() : '';

  var usuariosFiltrados = usuariosCache.filter(function(u) {
    if (!termino) return true;
    var nombre = (u.nombre || '').toLowerCase();
    var email = (u.email || '').toLowerCase();
    var telefono = (u.telefono || '').toLowerCase();
    return nombre.indexOf(termino) !== -1 || email.indexOf(termino) !== -1 || telefono.indexOf(termino) !== -1;
  });

  if (usuariosFiltrados.length === 0) {
    if (tbodyUsuarios) tbodyUsuarios.innerHTML = '';
    if (listaUsuariosMobile) listaUsuariosMobile.innerHTML = '';
    if (emptyStateUsuarios) emptyStateUsuarios.style.display = 'block';
    return;
  }

  if (emptyStateUsuarios) emptyStateUsuarios.style.display = 'none';

  // Renderizar tabla desktop
  var htmlDesktop = '';
  usuariosFiltrados.forEach(function(usuario) {
    var avatarClass = obtenerClaseAvatar(usuario.rol);
    var inicial = usuario.nombre ? usuario.nombre.charAt(0).toUpperCase() : 'U';
    var badgeRol = obtenerBadgeRol(usuario.rol);
    var badgeEstado = usuario.activo
      ? '<span class="badge-estado badge-activo">✅ Activo</span>'
      : '<span class="badge-estado badge-inactivo">❌ Inactivo</span>';

    var acciones = '';
    if (usuario.activo) {
      acciones = '<button class="btn-accion-usuario btn-editar-usuario" onclick="abrirModalUsuario(\'' + usuario.id + '\')">✏️ Editar</button>';
      // No permitir eliminar su propia cuenta
      if (usuario.id !== usuarioDashboard.id) {
        acciones += '<button class="btn-accion-usuario btn-eliminar-usuario" onclick="abrirModalEliminar(\'' + usuario.id + '\', \'' + escapeHtml(usuario.nombre) + '\')">🗑️</button>';
      }
    } else {
      acciones = '<button class="btn-accion-usuario btn-reactivar-usuario" onclick="reactivarUsuario(\'' + usuario.id + '\')">🔄 Reactivar</button>';
    }

    htmlDesktop += '<tr>';
    htmlDesktop += '<td><div class="usuario-info"><div class="usuario-avatar ' + avatarClass + '">' + inicial + '</div><div><div class="usuario-nombre">' + escapeHtml(usuario.nombre) + '</div><div class="usuario-id">' + usuario.id + '</div></div></div></td>';
    htmlDesktop += '<td>' + badgeRol + '</td>';
    htmlDesktop += '<td><div class="contacto-info"><span class="contacto-email">' + (usuario.email || '-') + '</span><span class="contacto-telefono">📱 ' + (usuario.telefono || '-') + '</span></div></td>';
    htmlDesktop += '<td>' + badgeEstado + '</td>';
    htmlDesktop += '<td><div class="acciones-usuario">' + acciones + '</div></td>';
    htmlDesktop += '</tr>';
  });

  if (tbodyUsuarios) tbodyUsuarios.innerHTML = htmlDesktop;

  // Renderizar cards mobile
  var htmlMobile = '';
  usuariosFiltrados.forEach(function(usuario) {
    var avatarClass = obtenerClaseAvatar(usuario.rol);
    var inicial = usuario.nombre ? usuario.nombre.charAt(0).toUpperCase() : 'U';
    var badgeRol = obtenerBadgeRol(usuario.rol);
    var badgeEstado = usuario.activo
      ? '<span class="badge-estado badge-activo">✅ Activo</span>'
      : '<span class="badge-estado badge-inactivo">❌ Inactivo</span>';

    var acciones = '';
    if (usuario.activo) {
      acciones = '<button class="btn-accion-usuario btn-editar-usuario" onclick="abrirModalUsuario(\'' + usuario.id + '\')">✏️ Editar</button>';
      if (usuario.id !== usuarioDashboard.id) {
        acciones += '<button class="btn-accion-usuario btn-eliminar-usuario" onclick="abrirModalEliminar(\'' + usuario.id + '\', \'' + escapeHtml(usuario.nombre) + '\')">🗑️ Desactivar</button>';
      }
    } else {
      acciones = '<button class="btn-accion-usuario btn-reactivar-usuario" onclick="reactivarUsuario(\'' + usuario.id + '\')">🔄 Reactivar</button>';
    }

    htmlMobile += '<div class="card-usuario-mobile">';
    htmlMobile += '<div class="card-usuario-header"><div class="card-usuario-info"><div class="usuario-avatar ' + avatarClass + '">' + inicial + '</div><div><div class="card-usuario-nombre">' + escapeHtml(usuario.nombre) + '</div><div class="card-usuario-meta">' + badgeRol + ' ' + badgeEstado + '</div></div></div></div>';
    htmlMobile += '<div class="card-usuario-contacto">';
    if (usuario.email) htmlMobile += '<div class="card-usuario-contacto-item"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M4 4h16c1.1 0 2 .9 2 2v12c0 1.1-.9 2-2 2H4c-1.1 0-2-.9-2-2V6c0-1.1.9-2 2-2z"/><polyline points="22,6 12,13 2,6"/></svg>' + usuario.email + '</div>';
    if (usuario.telefono) htmlMobile += '<div class="card-usuario-contacto-item"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M22 16.92v3a2 2 0 0 1-2.18 2 19.79 19.79 0 0 1-8.63-3.07 19.5 19.5 0 0 1-6-6 19.79 19.79 0 0 1-3.07-8.67A2 2 0 0 1 4.11 2h3a2 2 0 0 1 2 1.72 12.84 12.84 0 0 0 .7 2.81 2 2 0 0 1-.45 2.11L8.09 9.91a16 16 0 0 0 6 6l1.27-1.27a2 2 0 0 1 2.11-.45 12.84 12.84 0 0 0 2.81.7A2 2 0 0 1 22 16.92z"/></svg>' + usuario.telefono + '</div>';
    htmlMobile += '</div>';
    htmlMobile += '<div class="card-usuario-actions">' + acciones + '</div>';
    htmlMobile += '</div>';
  });

  if (listaUsuariosMobile) listaUsuariosMobile.innerHTML = htmlMobile;
}

/**
 * Obtener clase CSS del avatar según el rol
 */
function obtenerClaseAvatar(rol) {
  var clases = {
    'Administrador': 'avatar-admin',
    'Admin': 'avatar-admin',
    'Mesero': 'avatar-mesero',
    'Cocina': 'avatar-cocina',
    'Domiciliario': 'avatar-domiciliario'
  };
  return clases[rol] || 'avatar-admin';
}

/**
 * Obtener badge HTML del rol
 */
function obtenerBadgeRol(rol) {
  var badges = {
    'Administrador': '<span class="badge-rol badge-rol-admin">👑 Administrador</span>',
    'Admin': '<span class="badge-rol badge-rol-admin">👑 Admin</span>',
    'Mesero': '<span class="badge-rol badge-rol-mesero">🧑‍🍳 Mesero</span>',
    'Cocina': '<span class="badge-rol badge-rol-cocina">🍳 Cocina</span>',
    'Domiciliario': '<span class="badge-rol badge-rol-domiciliario">🛵 Domiciliario</span>'
  };
  return badges[rol] || '<span class="badge-rol">' + rol + '</span>';
}

/**
 * Abrir modal para crear o editar usuario
 */
function abrirModalUsuario(idUsuario) {
  var modal = document.getElementById('modalUsuario');
  var titulo = document.getElementById('modalUsuarioTitulo');
  var btnTexto = document.getElementById('btnGuardarUsuarioText');
  var errorDiv = document.getElementById('errorFormUsuario');

  // Limpiar formulario
  document.getElementById('formUsuario').reset();
  document.getElementById('checkActivoUsuario').checked = true;
  if (errorDiv) errorDiv.style.display = 'none';

  // Resetear campos
  actualizarCamposSegunRol();

  if (idUsuario) {
    // Modo edición
    usuarioEditando = usuariosCache.find(function(u) { return u.id === idUsuario; });

    if (!usuarioEditando) {
      mostrarToast('error', 'Error', 'Usuario no encontrado');
      return;
    }

    titulo.textContent = '✏️ Editar Usuario';
    btnTexto.textContent = 'Actualizar Usuario';

    // Llenar formulario
    document.getElementById('inputNombreUsuario').value = usuarioEditando.nombre || '';
    document.getElementById('selectRolUsuario').value = usuarioEditando.rol || '';
    document.getElementById('inputEmailUsuario').value = usuarioEditando.email || '';
    document.getElementById('inputTelefonoUsuario').value = usuarioEditando.telefono || '';
    document.getElementById('checkActivoUsuario').checked = usuarioEditando.activo;
    document.getElementById('inputCredencialUsuario').value = ''; // No mostrar credencial actual
    document.getElementById('reqCredencial').style.display = 'none'; // No requerido en edición

    // Actualizar campos según rol
    actualizarCamposSegunRol();

  } else {
    // Modo creación
    usuarioEditando = null;
    titulo.textContent = '➕ Nuevo Usuario';
    btnTexto.textContent = 'Guardar Usuario';
    document.getElementById('reqCredencial').style.display = 'inline';
  }

  modal.classList.add('show');
}

/**
 * Cerrar modal de usuario
 */
function cerrarModalUsuario() {
  var modal = document.getElementById('modalUsuario');
  modal.classList.remove('show');
  usuarioEditando = null;
}

/**
 * Actualizar campos del formulario según el rol seleccionado
 */
function actualizarCamposSegunRol() {
  var rol = document.getElementById('selectRolUsuario').value;
  var campoEmail = document.getElementById('campoEmailUsuario');
  var labelCredencial = document.getElementById('labelCredencialUsuario');
  var inputCredencial = document.getElementById('inputCredencialUsuario');
  var hintCredencial = document.getElementById('hintCredencial');
  var hintRol = document.getElementById('hintRol');

  if (rol === 'Mesero') {
    // Ocultar email para meseros
    campoEmail.style.display = 'none';

    // Cambiar a PIN
    labelCredencial.innerHTML = '🔢 PIN de Acceso <span class="required" id="reqCredencial">*</span>';
    inputCredencial.type = 'text';
    inputCredencial.maxLength = 4;
    inputCredencial.placeholder = '0000';
    inputCredencial.pattern = '[0-9]{4}';
    hintCredencial.textContent = 'PIN de 4 dígitos numéricos para acceso rápido';
    hintRol.textContent = 'Los meseros acceden con PIN desde el Panel de Meseros';

  } else {
    // Mostrar email para otros roles
    campoEmail.style.display = 'block';

    // Cambiar a Password
    labelCredencial.innerHTML = '🔐 Contraseña <span class="required" id="reqCredencial">*</span>';
    inputCredencial.type = 'password';
    inputCredencial.maxLength = 50;
    inputCredencial.placeholder = '••••••••';
    inputCredencial.pattern = '';
    hintCredencial.textContent = 'Mínimo 4 caracteres';

    if (rol === 'Administrador') {
      hintRol.textContent = 'Acceso completo al sistema, puede gestionar usuarios y pedidos';
    } else if (rol === 'Cocina') {
      hintRol.textContent = 'Acceso al Dashboard para ver y gestionar pedidos en preparación';
    } else if (rol === 'Domiciliario') {
      hintRol.textContent = 'Acceso al Dashboard para ver pedidos en camino';
    } else {
      hintRol.textContent = 'El rol determina los permisos y método de acceso';
    }
  }

  // Mostrar/ocultar requerido según modo
  var reqCredencial = document.getElementById('reqCredencial');
  if (reqCredencial && usuarioEditando) {
    reqCredencial.style.display = 'none';
  }
}

/**
 * Toggle visibilidad de password
 */
function togglePasswordVisibility() {
  var input = document.getElementById('inputCredencialUsuario');
  var iconShow = document.getElementById('iconShowPass');
  var iconHide = document.getElementById('iconHidePass');

  if (input.type === 'password') {
    input.type = 'text';
    iconShow.style.display = 'none';
    iconHide.style.display = 'block';
  } else {
    input.type = 'password';
    iconShow.style.display = 'block';
    iconHide.style.display = 'none';
  }
}

/**
 * Guardar usuario (crear o actualizar)
 */
function guardarUsuario() {
  var errorDiv = document.getElementById('errorFormUsuario');
  errorDiv.style.display = 'none';

  // Obtener valores
  var nombre = document.getElementById('inputNombreUsuario').value.trim();
  var rol = document.getElementById('selectRolUsuario').value;
  var email = document.getElementById('inputEmailUsuario').value.trim();
  var credencial = document.getElementById('inputCredencialUsuario').value;
  var telefono = document.getElementById('inputTelefonoUsuario').value.trim();
  var activo = document.getElementById('checkActivoUsuario').checked;

  // Validaciones
  var errores = [];

  if (!nombre || nombre.length < 2) {
    errores.push('El nombre es obligatorio (mínimo 2 caracteres)');
  }

  if (!rol) {
    errores.push('Debe seleccionar un rol');
  }

  if (rol === 'Mesero') {
    // Validar PIN
    if (!usuarioEditando && !credencial) {
      errores.push('El PIN es obligatorio');
    }
    if (credencial && !/^\d{4}$/.test(credencial)) {
      errores.push('El PIN debe ser de 4 dígitos numéricos');
    }
  } else {
    // Validar email
    if (!email) {
      errores.push('El email es obligatorio');
    } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      errores.push('El email no tiene un formato válido');
    }

    // Validar password
    if (!usuarioEditando && !credencial) {
      errores.push('La contraseña es obligatoria');
    }
    if (credencial && credencial.length < 4) {
      errores.push('La contraseña debe tener mínimo 4 caracteres');
    }
  }

  // Validar teléfono si se proporciona
  if (telefono && telefono.length < 7) {
    errores.push('El teléfono debe tener mínimo 7 dígitos');
  }

  if (errores.length > 0) {
    errorDiv.innerHTML = errores.join('<br>');
    errorDiv.style.display = 'block';
    return;
  }

  // Preparar datos
  var datos = {
    nombre: nombre,
    email: email,
    telefono: telefono,
    rol: rol,
    credencial: credencial,
    activo: activo,
    idActual: usuarioEditando ? usuarioEditando.id : null
  };

  console.log('💾 Guardando usuario:', datos);
  mostrarLoading(true);

  google.script.run
    .withSuccessHandler(function(response) {
      mostrarLoading(false);
      console.log('📥 Respuesta guardar:', response);

      if (response.success) {
        mostrarToast('success', 'Éxito', response.message);
        cerrarModalUsuario();
        cargarListaUsuarios();
      } else {
        errorDiv.innerHTML = response.error || 'Error al guardar el usuario';
        errorDiv.style.display = 'block';
      }
    })
    .withFailureHandler(function(error) {
      mostrarLoading(false);
      console.error('❌ Error al guardar:', error);
      errorDiv.innerHTML = 'Error de conexión: ' + error;
      errorDiv.style.display = 'block';
    })
    .adminGestionarUsuarios('guardar', datos, usuarioDashboard);
}

}

/**
 * Confirmar eliminación (soft delete)
 */
function confirmarEliminarUsuario() {
  if (!usuarioAEliminar) return;

  console.log('🗑️ Desactivando usuario:', usuarioAEliminar);
  mostrarLoading(true);

  google.script.run
    .withSuccessHandler(function(response) {
      mostrarLoading(false);
      cerrarModalConfirmarEliminar();

      if (response.success) {
        mostrarToast('success', 'Usuario desactivado', response.message);
        cargarListaUsuarios();
      } else {
        mostrarToast('error', 'Error', response.error || 'No se pudo desactivar el usuario');
      }
    })
    .withFailureHandler(function(error) {
      mostrarLoading(false);
      cerrarModalConfirmarEliminar();
      console.error('❌ Error al eliminar:', error);
      mostrarToast('error', 'Error de conexión', error.toString());
    })
    .adminGestionarUsuarios('eliminar', { id: usuarioAEliminar }, usuarioDashboard);
}

/**
 * Reactivar usuario desactivado
 */
function reactivarUsuario(idUsuario) {
  if (!confirm('¿Desea reactivar este usuario?')) return;

  console.log('🔄 Reactivando usuario:', idUsuario);
  mostrarLoading(true);

  google.script.run
    .withSuccessHandler(function(response) {
      mostrarLoading(false);

      if (response.success) {
        mostrarToast('success', 'Usuario reactivado', response.message);
        cargarListaUsuarios();
      } else {
        mostrarToast('error', 'Error', response.error || 'No se pudo reactivar el usuario');
      }
    })
    .withFailureHandler(function(error) {
      mostrarLoading(false);
      console.error('❌ Error al reactivar:', error);
      mostrarToast('error', 'Error de conexión', error.toString());
    })
    .adminReactivarEntidad('usuario', idUsuario, usuarioDashboard);
}

/**
 * Escapar HTML para prevenir XSS
 */
function escapeHtml(text) {
  if (!text) return '';
  var div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;
}

/**
 * Modificar función cambiarVista para soportar vista de usuarios
 */
var cambiarVistaOriginal = cambiarVista;
cambiarVista = function(vista, elemento) {
  vistaActual = vista;

  document.querySelectorAll('.nav-item').forEach(function(item) { item.classList.remove('active'); });
  if (elemento) elemento.classList.add('active');

  // Ocultar todas las secciones
  var statsGrid = document.querySelector('.stats-grid');
  var filtersBar = document.querySelector('.filters-bar');
  var tableContainer = document.querySelector('.table-container');
  var seccionUsuarios = document.getElementById('seccionUsuarios');

  if (vista === 'usuarios') {
    // Mostrar sección de usuarios
    if (statsGrid) statsGrid.style.display = 'none';
    if (filtersBar) filtersBar.style.display = 'none';
    if (tableContainer) tableContainer.style.display = 'none';
    if (seccionUsuarios) seccionUsuarios.style.display = 'block';

    document.getElementById('vistaTitle').textContent = '👥 Gestión de Usuarios';
    document.getElementById('vistaSubtitle').textContent = 'Administrar usuarios del sistema';

    // Cargar usuarios si no están cargados
    if (usuariosCache.length === 0) {
      cargarListaUsuarios();
    }
  } else {
    // Mostrar vista de pedidos
    if (statsGrid) statsGrid.style.display = 'grid';
    if (filtersBar) filtersBar.style.display = 'flex';
    if (tableContainer) tableContainer.style.display = 'block';
    if (seccionUsuarios) seccionUsuarios.style.display = 'none';

    var titulos = {
      'admin': { titulo: '📊 Dashboard Pedidos', subtitulo: 'Vista completa de todos los pedidos' },
      'cocina': { titulo: '👨‍🍳 Pedidos Comanda', subtitulo: 'Pedidos pendientes y en preparación' },
      'domicilio': { titulo: '🛵 Pedidos Domicilio', subtitulo: 'Pedidos listos para entregar' }
    };

    if (titulos[vista]) {
      document.getElementById('vistaTitle').textContent = titulos[vista].titulo;
      document.getElementById('vistaSubtitle').textContent = titulos[vista].subtitulo;
    }

    filtrosActuales.vistaRol = vista === 'admin' ? null : vista;
    cargarDatosDashboard();
  }
};

};

/**
 * Cargar lista de clientes desde el backend
 */
function cargarListaClientes() {
  try {
    console.log('[CLIENTES] 🔄 Solicitando lista de clientes al backend');

    var tbodyClientes = document.getElementById('tbodyClientes');
    var listaClientesMobile = document.getElementById('listaClientesMobile');
    var emptyStateClientes = document.getElementById('emptyStateClientes');

    // Mostrar loading

      tbodyClientes.innerHTML = '<tr class="loading-row"><td colspan="5"><div class="table-loading"><div class="spinner"></div><span>⏳ Cargando clientes...</span></div></td></tr>';

          renderizarTablaClientes(clientesGlobal);

          if (tbodyClientes) {

          }
        }
      })
      .withFailureHandler(function(error) {
        console.error('[CLIENTES] ❌ Error de conexión:', error);
        if (tbodyClientes) {
          tbodyClientes.innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--danger);">❌ Error de conexión</td></tr>';

        mostrarToast('error', 'Error de conexión', 'No se pudo conectar con el servidor');
      })
      .adminGestionarClientes('leer', null);

  } catch(error) {
    console.error('[CLIENTES] Error al cargar lista:', error);
  }

/**
 * Renderizar tabla de clientes (Desktop y Mobile)
 */
function renderizarTablaClientes(clientes) {
  try {
    var tbodyClientes = document.getElementById('tbodyClientes');
    var listaClientesMobile = document.getElementById('listaClientesMobile');
    var emptyStateClientes = document.getElementById('emptyStateClientes');

    if (!clientes || clientes.length === 0) {
      if (tbodyClientes) tbodyClientes.innerHTML = '';
      if (listaClientesMobile) listaClientesMobile.innerHTML = '';
      if (emptyStateClientes) emptyStateClientes.style.display = 'block';
      return;
    }

    if (emptyStateClientes) emptyStateClientes.style.display = 'none';

    // Renderizar tabla desktop
    var htmlDesktop = '';
    clientes.forEach(function(cliente) {
      var clienteJson = JSON.stringify(cliente).replace(/'/g, "\\'").replace(/"/g, '&quot;');

      htmlDesktop += '<tr>';
      htmlDesktop += '<td><div class="cliente-nombre-celda"><strong>' + escapeHtml(cliente.nombre || '-') + '</strong></div></td>';
      htmlDesktop += '<td><span class="cliente-telefono">📱 ' + escapeHtml(cliente.celular || '-') + '</span></td>';
      htmlDesktop += '<td><div class="cliente-direccion" title="' + escapeHtml(cliente.domicilio || '') + '">' + escapeHtml(truncarTexto(cliente.domicilio, 30) || '-') + '</div></td>';
      htmlDesktop += '<td>' + escapeHtml(cliente.email || '-') + '</td>';
      htmlDesktop += '<td><div class="acciones-cliente"><button class="btn-accion-cliente btn-editar-cliente" onclick="abrirModalEditarCliente(' + clienteJson + ')">✏️ Editar</button></div></td>';
      htmlDesktop += '</tr>';
    });

    if (tbodyClientes) tbodyClientes.innerHTML = htmlDesktop;

    // Renderizar cards mobile
    var htmlMobile = '';
    clientes.forEach(function(cliente) {
      var clienteJson = JSON.stringify(cliente).replace(/'/g, "\\'").replace(/"/g, '&quot;');

      htmlMobile += '<div class="card-cliente-mobile">';
      htmlMobile += '<div class="card-cliente-header"><div class="card-cliente-info"><div class="cliente-avatar-mobile">👤</div><div><div class="card-cliente-nombre">' + escapeHtml(cliente.nombre || 'Sin nombre') + '</div><div class="card-cliente-telefono">📱 ' + escapeHtml(cliente.celular || '-') + '</div></div></div></div>';
      htmlMobile += '<div class="card-cliente-contacto">';
      if (cliente.domicilio) htmlMobile += '<div class="card-cliente-contacto-item">🏠 ' + escapeHtml(cliente.domicilio) + '</div>';
      if (cliente.email) htmlMobile += '<div class="card-cliente-contacto-item">✉️ ' + escapeHtml(cliente.email) + '</div>';
      htmlMobile += '</div>';
      htmlMobile += '<div class="card-cliente-actions"><button class="btn-accion-cliente btn-editar-cliente" onclick="abrirModalEditarCliente(' + clienteJson + ')">✏️ Editar</button></div>';
      htmlMobile += '</div>';
    });

    if (listaClientesMobile) listaClientesMobile.innerHTML = htmlMobile;

  } catch(error) {
    console.error('[CLIENTES] Error al renderizar tabla:', error);
  }
}

/**
 * Filtrar clientes en tiempo real
 */
function filtrarClientes() {
  try {
    var busqueda = document.getElementById('buscadorClientes').value.toLowerCase().trim();

    if (!busqueda) {
      renderizarTablaClientes(clientesGlobal);
      return;
    }

    var clientesFiltrados = clientesGlobal.filter(function(cliente) {
      var nombre = (cliente.nombre || '').toLowerCase();
      var telefono = (cliente.celular || '').toLowerCase();
      var direccion = (cliente.domicilio || '').toLowerCase();

      return nombre.indexOf(busqueda) !== -1 ||
             telefono.indexOf(busqueda) !== -1 ||
             direccion.indexOf(busqueda) !== -1;
    });

    renderizarTablaClientes(clientesFiltrados);

  } catch(error) {
    console.error('[CLIENTES] Error al filtrar:', error);
  }
}

/**
 * Guardar cambios del cliente
 */
function guardarCliente() {
  try {
    var errorDiv = document.getElementById('errorFormCliente');
    if (errorDiv) errorDiv.style.display = 'none';

    // Obtener valores
    var nombre = document.getElementById('inputClienteNombre').value.trim();
    var telefono = document.getElementById('inputClienteTelefono').value.trim();
    var direccion = document.getElementById('inputClienteDireccion').value.trim();
    var email = document.getElementById('inputClienteEmail').value.trim();
    var notas = document.getElementById('inputClienteNotas').value.trim();

    // Validaciones
    var errores = [];

    if (!nombre) {
      errores.push('El nombre es obligatorio');
    }

    if (!telefono || telefono.length !== 10 || !/^\d+$/.test(telefono)) {
      errores.push('El teléfono debe tener exactamente 10 dígitos');
    }

    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      errores.push('El email no tiene un formato válido');
    }

    if (errores.length > 0) {
      if (errorDiv) {
        errorDiv.innerHTML = '⚠️ ' + errores.join('<br>⚠️ ');
        errorDiv.style.display = 'block';
      }
      mostrarToast('warning', 'Validación', errores[0]);
      return;
    }

    // Preparar datos
    var datosCliente = {
      idActual: clienteEditando.id,
      nombre: nombre,
      celular: telefono,
      domicilio: direccion,
      email: email,
      notas: notas
    };

    console.log('[CLIENTES] 💾 Guardando cliente:', datosCliente);
    mostrarLoading(true);

    google.script.run
      .withSuccessHandler(function(response) {
        mostrarLoading(false);

        if (response.success) {
          mostrarToast('success', '✅ Cliente actualizado', response.message || 'Los datos se guardaron correctamente');
          cerrarModalCliente();
          cargarListaClientes(); // Recargar tabla
        } else {
          if (errorDiv) {
            errorDiv.innerHTML = '❌ ' + (response.error || 'Error al guardar');
            errorDiv.style.display = 'block';
          }
          mostrarToast('error', 'Error', response.error || 'No se pudo guardar el cliente');
        }
      })
      .withFailureHandler(function(error) {
        mostrarLoading(false);
        console.error('[CLIENTES] ❌ Error de conexión:', error);
        mostrarToast('error', 'Error de conexión', 'No se pudo guardar el cliente');
      })
      .adminGestionarClientes('guardar', datosCliente);

  } catch(error) {
    console.error('[CLIENTES] Error al guardar:', error);
    mostrarToast('error', 'Error', 'Error al guardar cliente');
  }
}

/**
 * Confirmar eliminación de cliente
 */
function confirmarEliminarCliente() {
  try {
    if (!clienteEditando) {
      mostrarToast('warning', 'Advertencia', 'No hay cliente seleccionado');
      return;
    }

    var confirmar = confirm(
      '⚠️ ¿Estás seguro de eliminar a ' + clienteEditando.nombre + '?\n\n' +
      'Esta acción marcará al cliente como inactivo.'
    );

    if (!confirmar) return;

    console.log('[CLIENTES] 🗑️ Eliminando cliente:', clienteEditando.id);
    mostrarLoading(true);

    google.script.run
      .withSuccessHandler(function(response) {
        mostrarLoading(false);

        if (response.success) {
          mostrarToast('success', '✅ Cliente eliminado', response.message || 'El cliente fue eliminado correctamente');
          cerrarModalCliente();
          cargarListaClientes();
        } else {
          mostrarToast('error', 'Error', response.error || 'No se pudo eliminar el cliente');
        }
      })
      .withFailureHandler(function(error) {
        mostrarLoading(false);
        console.error('[CLIENTES] ❌ Error al eliminar:', error);
        mostrarToast('error', 'Error de conexión', 'No se pudo eliminar el cliente');
      })
      .adminGestionarClientes('eliminar', { idActual: clienteEditando.id });

  } catch(error) {
    console.error('[CLIENTES] Error al eliminar:', error);
    mostrarToast('error', 'Error', 'Error al eliminar cliente');
  }
}

/**
 * Función auxiliar para truncar texto
 */
function truncarTexto(texto, maxLength) {
  if (!texto) return '';
  if (texto.length <= maxLength) return texto;
  return texto.substring(0, maxLength) + '...';
}
