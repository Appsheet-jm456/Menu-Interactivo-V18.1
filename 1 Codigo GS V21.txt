/**
 * Validador de Teléfono
 * Acepta: Solo dígitos, 7-10 caracteres
 */
function validarTelefono(telefono) {
  if (!telefono) {
    return { valido: false, error: 'Telefono requerido' };
  }

  // Limpiar espacios y caracteres especiales
  var telLimpio = String(telefono).replace(/\s/g, '').replace(/\D/g, '');

  // Validar: solo numeros, 7-10 digitos
  if (telLimpio.length < 7 || telLimpio.length > 10) {
    return {
      valido: false,
      error: 'Telefono invalido. Debe tener 7-10 digitos numericos'
    };
  }

  return { valido: true, valor: telLimpio };
}

/**
 * Validador de Email
 * Acepta: Formato valido o campo vacio
 */
function validarEmail(email) {
  // Email es opcional, puede estar vacio
  if (!email || String(email).trim() === '') {
    return { valido: true, valor: '' };
  }

  var emailLimpio = String(email).trim().toLowerCase();

  // Regex estandar para email
  var regexEmail = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  if (!regexEmail.test(emailLimpio)) {
    return {
      valido: false,
      error: 'Email invalido. Formato esperado: usuario@dominio.com'
    };
  }

  if (emailLimpio.length > 100) {
    return { valido: false, error: 'Email muy largo (max 100 caracteres)' };
  }

  return { valido: true, valor: emailLimpio };
}

/**
 * Validador de Nombre
 * Acepta: 2-100 caracteres
 */
function validarNombre(nombre) {
  if (!nombre || String(nombre).trim().length === 0) {
    return { valido: false, error: 'Nombre requerido' };
  }

  var nombreLimpio = String(nombre).trim();

  if (nombreLimpio.length < 2) {
    return { valido: false, error: 'Nombre muy corto (min 2 caracteres)' };
  }

  if (nombreLimpio.length > 100) {
    return { valido: false, error: 'Nombre muy largo (max 100 caracteres)' };
  }

  // No permitir solo numeros
  if (/^\d+$/.test(nombreLimpio)) {
    return { valido: false, error: 'Nombre no puede ser solo numeros' };
  }

  return { valido: true, valor: nombreLimpio };
}

/**
 * Validador de Precio
 * Acepta: Numeros >= 0, max 10,000,000
 */
function validarPrecio(precio) {
  var precioNum = parseFloat(precio);

  if (isNaN(precioNum)) {
    return { valido: false, error: 'Precio debe ser un numero' };
  }

  if (precioNum < 0) {
    return { valido: false, error: 'Precio no puede ser negativo' };
  }

  if (precioNum > 10000000) {
    return { valido: false, error: 'Precio supera el limite maximo' };
  }

  return { valido: true, valor: precioNum };
}

/**
 * Validador de Cantidad
 * Acepta: Enteros entre 1 y 999
 */
function validarCantidad(cantidad) {

  if (isNaN(cantidadNum)) {

/**
 * Validador de Metodo de Pago
 * Acepta: Efectivo, Tarjeta, Transferencia, Mixto
 */
function validarMetodoPago(metodo) {
  var metodosValidos = ['Efectivo', 'Tarjeta', 'Transferencia', 'Mixto', 'Nequi', 'Daviplata'];
  var metodoLimpio = String(metodo || '').trim();

  if (!metodoLimpio) {
    return { valido: false, error: 'Metodo de pago requerido' };
  }

  // Buscar coincidencia case-insensitive
  var encontrado = metodosValidos.find(function(m) {
    return m.toLowerCase() === metodoLimpio.toLowerCase();
  });

  if (!encontrado) {
    return {
      valido: false,
      error: 'Metodo de pago invalido. Opciones: ' + metodosValidos.join(', ')
    };
  }

  return { valido: true, valor: encontrado };
}

/**
 * Validador de Descuento
 * Acepta: Numeros >= 0
 */
function validarDescuento(descuento, subtotal) {
  var descuentoNum = parseFloat(descuento) || 0;
  var subtotalNum = parseFloat(subtotal) || 0;

  if (descuentoNum < 0) {
    return { valido: false, error: 'Descuento no puede ser negativo' };
  }

  if (subtotalNum > 0 && descuentoNum > subtotalNum) {
    return { valido: false, error: 'Descuento no puede ser mayor al subtotal' };
  }

  return { valido: true, valor: descuentoNum };
}

/**
 * Validador Maestro de Cliente
 * Valida todos los campos de un cliente
 */
function validarDatosCliente(datos) {
  var errores = [];
  var datosLimpios = {};

  // Validar nombre (requerido)
  if (datos.nombre) {
    var validNombre = validarNombre(datos.nombre);
    if (!validNombre.valido) {
      errores.push(validNombre.error);
    } else {
      datosLimpios.nombre = validNombre.valor;
    }
  } else {
    datosLimpios.nombre = '';
  }

  // Validar telefono (requerido)
  var validTel = validarTelefono(datos.celular || datos.telefono);
  if (!validTel.valido) {
    errores.push(validTel.error);
  } else {
    datosLimpios.celular = validTel.valor;
  }

  // Validar email (opcional)
  if (datos.email) {
    var validEmail = validarEmail(datos.email);
    if (!validEmail.valido) {
      errores.push(validEmail.error);
    } else {
      datosLimpios.email = validEmail.valor;
    }
  } else {
    datosLimpios.email = '';
  }

  // Direccion (opcional, sin validacion estricta)
  datosLimpios.domicilio = String(datos.domicilio || datos.direccion || '').trim();

  // Notas (opcional)
  datosLimpios.notas = String(datos.notas || '').trim();

  return {
    valido: errores.length === 0,
    errores: errores,
    datosLimpios: datosLimpios
  };
}

/**
 * Validador Maestro de Pedido
 * Valida datos basicos de un pedido
 */
function validarDatosPedido(datos) {
  var errores = [];

  // Validar tipo de entrega
  var tiposValidos = ['Mesa', 'Domicilio', 'Llevar', 'Recoger'];
  if (!datos.tipoEntrega || tiposValidos.indexOf(datos.tipoEntrega) === -1) {
    errores.push('Tipo de entrega invalido');
  }

  // Si es mesa, validar numero
  if (datos.tipoEntrega === 'Mesa' && datos.numeroMesa) {
    var validMesa = validarNumeroMesa(datos.numeroMesa);
    if (!validMesa.valido) {
      errores.push(validMesa.error);
    }
  }

  // Si es domicilio, validar direccion
  if (datos.tipoEntrega === 'Domicilio' && datos.direccion) {
    var validDir = validarDireccion(datos.direccion);
    if (!validDir.valido) {
      errores.push(validDir.error);
    }
  }

  return {
    valido: errores.length === 0,
    errores: errores
  };
}

/**
 * Validador de Items de Pedido
 * Valida array de productos
 */
function validarItemsPedido(items) {
  if (!items || !Array.isArray(items) || items.length === 0) {
    return { valido: false, error: 'Debe agregar al menos un producto' };
  }

  var errores = [];

  for (var i = 0; i < items.length; i++) {
    var item = items[i];

    // Validar ID producto
    if (!item.id && !item.idProducto) {
      errores.push('Item ' + (i + 1) + ': ID de producto requerido');
      continue;
    }

    // Validar cantidad
    var validCant = validarCantidad(item.cantidad);
    if (!validCant.valido) {
      errores.push('Item ' + (i + 1) + ': ' + validCant.error);
    }

    // Validar precio
    var validPrecio = validarPrecio(item.precio || item.precioUnitario);
    if (!validPrecio.valido) {
      errores.push('Item ' + (i + 1) + ': ' + validPrecio.error);
    }
  }

  return {
    valido: errores.length === 0,
    errores: errores
  };
}

function obtenerFechaColombia(fecha) {
  var fechaObj = fecha ? new Date(fecha) : new Date();
  return Utilities.formatDate(fechaObj, "America/Bogota", "dd/MM/yyyy");
}

function normalizarFechaAString(fecha) {
  if (!fecha) return null;
  
  if (fecha instanceof Date) {
    return Utilities.formatDate(fecha, "America/Bogota", "yyyy-MM-dd");
  }
  
  const fechaStr = String(fecha).trim();
  
  if (/^\d{4}-\d{2}-\d{2}$/.test(fechaStr)) {
    return fechaStr;
  }
  
  if (/^\d{2}\/\d{2}\/\d{4}$/.test(fechaStr)) {
    const parts = fechaStr.split('/');
    return parts[2] + '-' + parts[1] + '-' + parts[0];
  }
  
  if (fechaStr.includes('T')) {
    return fechaStr.split('T')[0];
  }
  
  try {
    const dateObj = new Date(fechaStr);
    if (!isNaN(dateObj.getTime())) {
      return Utilities.formatDate(dateObj, "America/Bogota", "yyyy-MM-dd");
    }
  } catch (e) {
    Logger.log('Error al normalizar fecha: ' + e.toString());
  }
  
  return null;
}

function registrarCambioEstado(idPedido, idCliente, idUsuario, nombreUsuario, rol, estadoAnterior, estadoNuevo, notas) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    let sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    if (!sheet) {
      inicializarTablaProcesoDelivery();
      sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    }
    const lastRow = sheet.getLastRow();
    const idDelivery = 'DEL' + String(lastRow).padStart(6, '0');
    const timestamp = obtenerTimestampColombia();
    const newRow = [idDelivery, idPedido, idCliente, idUsuario, nombreUsuario || 'Sistema', rol || 'Sistema', estadoAnterior || 'N/A', estadoNuevo, timestamp, notas || ''];
    sheet.appendRow(newRow);
    Logger.log('✅ Cambio de estado registrado: ' + idDelivery);
    return { success: true, idDelivery: idDelivery };
  } catch (error) {
    Logger.log('❌ Error al registrar cambio: ' + error.toString());
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function obtenerHistorialPedido(idPedido) {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    if (!sheet) return { success: true, historial: [] };
    
    const data = sheet.getDataRange().getValues();
    const historial = [];
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (String(row[1]).trim().toUpperCase() === idPedidoStr) {
        historial.push({
          idDelivery: String(row[0]),
          idPedido: String(row[1]),
          idCliente: String(row[2]),
          idUsuario: String(row[3]),
          nombreUsuario: String(row[4]),
          rol: String(row[5]),
          estadoAnterior: String(row[6]),
          estadoNuevo: String(row[7]),
          fechaHora: row[8] instanceof Date ? row[8].toISOString() : String(row[8]),
          notas: String(row[9] || '')
        });
      }
    }
    historial.sort((a, b) => new Date(a.fechaHora) - new Date(b.fechaHora));
    return { success: true, historial: historial };
  } catch (error) {
    Logger.log('Error en obtenerHistorialPedido: ' + error.toString());
    return { success: false, message: error.toString(), historial: [] };
  }
}

function validarLogin(email, password) {
  try {
    Logger.log('Intentando login para: ' + email);
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.USUARIOS);
    const data = sheet.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const userEmail = String(row[1] || '').trim().toLowerCase();
      const userPassword = String(row[2] || '');
      const activo = row[8];
      
      if (userEmail === email.toLowerCase().trim() && userPassword === password && activo === true) {
        Logger.log('Login exitoso para usuario ID: ' + row[0]);
        const rol = row[7];
        return {
          success: true,
          user: {
            id: String(row[0]).trim(),
            email: row[1],
            nombre: row[3],
            telefono: row[4],
            direccion: row[5],
            rol: rol,
            permisos: row[9],
            esInterno: esRolInterno(rol),
            esSuperAdmin: esRolSuperAdmin(rol)
          }
        };
      }
    }
    Logger.log('Login fallido - credenciales incorrectas');
    return { success: false, message: 'Usuario o contraseña incorrectos' };
  } catch (error) {
    Logger.log('Error en validarLogin: ' + error.toString());
    return { success: false, message: 'Error al validar credenciales: ' + error.toString() };
  }
}

function validarLoginDashboard(email, password) {
  try {
    const result = validarLogin(email, password);
    if (!result.success) return result;
    if (!result.user.esInterno) {

    return result;
  } catch (error) {
    Logger.log('Error en validarLoginDashboard: ' + error.toString());
    return { success: false, message: 'Error al validar credenciales: ' + error.toString() };
  }
}

function registrarUsuario(datos) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.USUARIOS);
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][1]).toLowerCase().trim() === datos.email.toLowerCase().trim()) {
        return { success: false, message: 'El correo electrónico ya está registrado' };
      }
    }
    const lastRow = sheet.getLastRow();
    const newId = lastRow > 1 ? 'CLI' + String(lastRow).padStart(5, '0') : 'CLI00001';
    const newRow = [newId, datos.email, datos.password, datos.nombre, datos.telefono, datos.direccion, new Date(), 'Clientes', true, ''];
    sheet.appendRow(newRow);
    Logger.log('Usuario registrado con ID: ' + newId);
    return {
      success: true,
      message: 'Usuario registrado exitosamente',
      user: { id: newId, email: datos.email, nombre: datos.nombre, telefono: datos.telefono, direccion: datos.direccion, rol: 'Clientes', esInterno: false, esSuperAdmin: false }
    };
  } catch (error) {
    Logger.log('Error en registrarUsuario: ' + error.toString());
    return { success: false, message: 'Error al registrar usuario: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

// ============================================
function obtenerConfiguracion() {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.CONFIGURACION);
    const data = sheet.getDataRange().getValues();
    const config = {};
    for (let i = 1; i < data.length; i++) {
      const parametro = String(data[i][0] || '').trim();
      const valor = data[i][1];
      if (parametro) {
        const parametroNormalizado = parametro.replace(/\s+/g, '').replace(/^(.)/, (m, p1) => p1.toLowerCase());
        config[parametroNormalizado] = valor;
        config[parametro] = valor;
      }
    }
    const defaults = { nombreEmpresa: 'CloudPyme Restaurant', colorPrimario: '#3B82F6', colorSecundario: '#1E40AF', colorAcento: '#F59E0B', moneda: '$', logoUrl: '', nit: '', direccion: '', telefono: '', email: '', slogan: 'Sistema de Gestión de Pedidos' };
    for (const [key, defaultValue] of Object.entries(defaults)) {
      if (!config[key] && config[key] !== 0 && config[key] !== false) config[key] = defaultValue;
    }
    return { success: true, config: config };
  } catch (error) {
    Logger.log('Error en obtenerConfiguracion: ' + error.toString());
    return { success: false, message: 'Error al cargar configuración', config: { nombreEmpresa: 'CloudPyme Restaurant', colorPrimario: '#3B82F6', colorSecundario: '#1E40AF', colorAcento: '#F59E0B', moneda: '$', logoUrl: '', nit: '', direccion: '', telefono: '' } };
  }
}

function obtenerCategorias() {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.CATEGORIAS);
    const data = sheet.getDataRange().getValues();
    const categorias = [];
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const activo = row[5];
      const esActivo = activo === true || String(activo).toUpperCase() === 'SI' || String(activo).toUpperCase() === 'SÍ' || String(activo).toUpperCase() === 'TRUE' || activo === 1;
      if (esActivo) {
        categorias.push({ id: String(row[0]).trim(), nombre: String(row[1] || '').trim(), descripcion: String(row[2] || '').trim(), icono: String(row[3] || '🍽️').trim(), orden: Number(row[4]) || 0, activo: true, imagen: String(row[6] || '').trim() });
      }
    }
    categorias.sort((a, b) => a.orden - b.orden);
    return { success: true, categorias: categorias };
  } catch (error) {
    Logger.log('Error en obtenerCategorias: ' + error.toString());
    return { success: false, message: 'Error al cargar categorías: ' + error.toString(), categorias: [] };
  }
}

// PRODUCTOS
// ============================================
function obtenerProductos(idCategoria) {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.PRODUCTOS);
    const data = sheet.getDataRange().getValues();
    const productos = [];
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row[6] === true) {
        const categoriaProducto = String(row[1]);
        if (!idCategoria || idCategoria === 'TODOS' || compararIds(categoriaProducto, idCategoria)) {
          productos.push({ id: String(row[0]), categoria: categoriaProducto, nombre: row[2], descripcion: row[3], precio: row[4], imagen: row[5], disponible: row[6], tiempoPreparacion: row[7], calorias: row[8] });
        }
      }
    }
    return { success: true, productos: productos };
  } catch (error) {
    Logger.log('Error en obtenerProductos: ' + error.toString());
    return { success: false, message: 'Error al cargar productos', productos: [] };
  }
}

// ADICIONES
// ============================================
function obtenerAdiciones(idCategoria) {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.ADICIONES);
    const data = sheet.getDataRange().getValues();
    const adiciones = [];
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row[4] === true) {
        const categoriasAplicables = row[3] ? String(row[3]).split(',').map(c => c.trim().toUpperCase()) : [];
        const idCatUpper = String(idCategoria || '').toUpperCase();
        if (categoriasAplicables.length === 0 || categoriasAplicables.includes(idCatUpper) || categoriasAplicables.includes('TODOS')) {
          adiciones.push({ id: String(row[0]), nombre: row[1], precio: row[2] });
        }
      }
    }
    return { success: true, adiciones: adiciones };
  } catch (error) {
    Logger.log('Error en obtenerAdiciones: ' + error.toString());
    return { success: false, message: 'Error al cargar adiciones', adiciones: [] };
  }
}

// PEDIDOS - FUNCIONES CLIENTE
// ============================================
function obtenerMisPedidos(idCliente) {
  try {
    Logger.log('INICIO obtenerMisPedidos - ID Cliente: ' + idCliente);
    if (!idCliente) return { success: false, message: 'ID de cliente no proporcionado', pedidos: [] };
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    if (!sheetPedidos) return { success: false, message: 'Hoja de Pedidos no encontrada', pedidos: [] };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const pedidos = [];
    const idClienteStr = String(idCliente).trim().toUpperCase();
    
    for (let i = 1; i < dataPedidos.length; i++) {
      const row = dataPedidos[i];
      const pedidoIdCliente = String(row[1]).trim().toUpperCase();
      if (pedidoIdCliente === idClienteStr) {
        const tieneNuevaEstructura = row.length >= 16;
        let fechaPedido, estadoLogistico;
        if (tieneNuevaEstructura) {
          fechaPedido = row[2] instanceof Date ? row[2].toISOString() : String(row[2]);
          estadoLogistico = String(row[4] || 'Solicitado');
        } else {
          fechaPedido = row[2] instanceof Date ? row[2].toISOString() : String(row[2]);
          estadoLogistico = String(row[3] || 'Solicitado');
        }
        pedidos.push({
          idPedido: String(row[0]),
          idCliente: String(row[1]),
          fechaPedido: fechaPedido,
          estado: estadoLogistico,
          tipoEntrega: String(row[tieneNuevaEstructura ? 5 : 4] || ''),
          direccionEntrega: String(row[tieneNuevaEstructura ? 6 : 5] || ''),
          telefonoContacto: String(row[tieneNuevaEstructura ? 7 : 6] || ''),
          subtotal: Number(row[tieneNuevaEstructura ? 8 : 7]) || 0,
          descuento: Number(row[tieneNuevaEstructura ? 9 : 8]) || 0,
          total: Number(row[tieneNuevaEstructura ? 10 : 9]) || 0,
          metodoPago: String(row[tieneNuevaEstructura ? 11 : 10] || 'Efectivo'),
          notas: String(row[tieneNuevaEstructura ? 12 : 11] || '')
        });
      }
    }
    pedidos.sort((a, b) => new Date(b.fechaPedido) - new Date(a.fechaPedido));
    return { success: true, pedidos: pedidos };
  } catch (error) {
    Logger.log('ERROR en obtenerMisPedidos: ' + error.toString());
    return { success: false, message: 'Error al cargar pedidos: ' + error.toString(), pedidos: [] };
  }
}

function obtenerDetallePedidoConHistorial(idPedido, idCliente) {

    return { success: false, message: 'Error al cargar detalle: ' + error.toString(), items: [], historial: [] };
  }
}

function obtenerDetallePedido(idPedido, idCliente) {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    const idClienteStr = String(idCliente).trim().toUpperCase();
    
    let pedidoValido = false;
    for (let i = 1; i < dataPedidos.length; i++) {
      const pedidoEnFila = String(dataPedidos[i][0]).trim().toUpperCase();
      const clienteEnFila = String(dataPedidos[i][1]).trim().toUpperCase();
      if (pedidoEnFila === idPedidoStr && clienteEnFila === idClienteStr) {
        pedidoValido = true;
        break;
      }
    }
    if (!pedidoValido) return { success: false, message: 'No tiene permisos para ver este pedido', items: [] };
    
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const dataDetalle = sheetDetalle.getDataRange().getValues();
    const sheetProductos = ss.getSheetByName(CONFIG.SHEETS.PRODUCTOS);
    const dataProductos = sheetProductos.getDataRange().getValues();
    const items = [];
    
    for (let i = 1; i < dataDetalle.length; i++) {
      const row = dataDetalle[i];
      const detalleIdPedido = String(row[1]).trim().toUpperCase();
      if (detalleIdPedido === idPedidoStr) {
        let nombreProducto = 'Producto no encontrado';
        const idProductoBuscado = String(row[3]).trim().toUpperCase();
        for (let j = 1; j < dataProductos.length; j++) {
          const idProductoEnFila = String(dataProductos[j][0]).trim().toUpperCase();
          if (idProductoEnFila === idProductoBuscado) {
            nombreProducto = dataProductos[j][2];
            break;
          }
        }
        items.push({ idDetalle: String(row[0]), idPedido: String(row[1]), idProducto: String(row[3]), nombreProducto: nombreProducto, cantidad: Number(row[4]) || 1, precioUnitario: Number(row[5]) || 0, adiciones: String(row[6] || ''), precioAdiciones: Number(row[7]) || 0, subtotalItem: Number(row[8]) || 0 });
      }
    }
    return { success: true, items: items };
  } catch (error) {
    Logger.log('Error en obtenerDetallePedido: ' + error.toString());
    return { success: false, message: 'Error al cargar detalle del pedido: ' + error.toString(), items: [] };
  }
}

function crearPedido(pedidoData) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    Logger.log('Creando pedido para cliente: ' + pedidoData.idCliente);
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const lastRow = sheetPedidos.getLastRow();
    const idPedido = 'PED' + String(lastRow).padStart(6, '0');
    const timestamp = obtenerTimestampColombia();
    const fechaSolo = obtenerFechaColombia(timestamp);
    
    const rowPedido = [idPedido, pedidoData.idCliente, timestamp, fechaSolo, 'Solicitado', pedidoData.tipoEntrega || 'Recoger en tienda', pedidoData.direccionEntrega || '', pedidoData.telefonoContacto || '', pedidoData.subtotal, pedidoData.descuento || 0, pedidoData.total, pedidoData.metodoPago || 'Efectivo', pedidoData.notas || '', 'Abierto', '', ''];
    sheetPedidos.appendRow(rowPedido);
    
    registrarCambioEstado(idPedido, pedidoData.idCliente, 'SISTEMA', 'Sistema', 'Sistema', 'N/A', 'Solicitado', 'Pedido creado por cliente');
    
    pedidoData.items.forEach((item, index) => {
      const idDetalle = idPedido + '-' + String(index + 1).padStart(3, '0');
      const rowDetalle = [idDetalle, idPedido, pedidoData.idCliente, item.idProducto, item.cantidad, item.precioUnitario, item.adiciones || '', item.precioAdiciones || 0, item.subtotalItem];
      sheetDetalle.appendRow(rowDetalle);
    });
    
    Logger.log('Pedido creado exitosamente: ' + idPedido);
    return { success: true, message: 'Pedido creado exitosamente', idPedido: idPedido };
  } catch (error) {
    Logger.log('Error en crearPedido: ' + error.toString());
    return { success: false, message: 'Error al crear pedido: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

// DASHBOARD - FUNCIONES PRINCIPALES
// ============================================
function obtenerPedidosDashboard(filtros) {
  try {
    Logger.log('========================================');
    Logger.log('DASHBOARD - Obteniendo pedidos');
    Logger.log('Filtros recibidos: ' + JSON.stringify(filtros));
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetUsuarios = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);
    const sheetClientes = ss.getSheetByName(CONFIG.SHEETS.CLIENTES); // FASE 9.5: Cargar clientes

    if (!sheetPedidos) return { success: false, message: 'Hoja de Pedidos no encontrada', pedidos: [], clientes: [] };

    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const dataUsuarios = sheetUsuarios ? sheetUsuarios.getDataRange().getValues() : [];
    const dataClientes = sheetClientes ? sheetClientes.getDataRange().getValues() : []; // FASE 9.5

    // FASE 9.5: Crear lista de clientes para enviar al frontend
    const clientesLista = [];
    if (dataClientes.length > 1) {
      for (var c = 1; c < dataClientes.length; c++) {
        var filaCliente = dataClientes[c];
        var activo = filaCliente[9];
        // Solo incluir clientes activos
        if (activo !== false && activo !== 'FALSE' && activo !== 'false') {
          clientesLista.push({
            id: String(filaCliente[0] || '').trim(),       // ID_Key_Clientes
            nombre: String(filaCliente[1] || '').trim(),   // Nombre_Completo
            celular: String(filaCliente[2] || '').replace(/\D/g, '').trim(), // Celular normalizado
            domicilio: String(filaCliente[3] || '').trim(),// Domicilio
            email: String(filaCliente[6] || '').trim(),    // Email
            rol: String(filaCliente[8] || 'Cliente').trim()// Rol
          });
        }
      }
      Logger.log('[DASHBOARD] Clientes activos cargados: ' + clientesLista.length);
    }
    
    const headers = dataPedidos[0];
    const tieneNuevaEstructura = headers.length >= 16 && (String(headers[2]).toLowerCase().includes('marca') || String(headers[3]).toLowerCase().includes('fecha'));
    Logger.log('Estructura detectada: ' + (tieneNuevaEstructura ? 'NUEVA' : 'ANTIGUA'));
    
    const COL = tieneNuevaEstructura ? {
      ID_PEDIDO: 0, ID_CLIENTE: 1, MARCA_TIEMPO: 2, FECHA: 3, ESTADO_LOGISTICO: 4,
      TIPO_ENTREGA: 5, DIRECCION_ENTREGA: 6, TELEFONO_CONTACTO: 7, SUBTOTAL: 8,
      DESCUENTO: 9, TOTAL: 10, METODO_PAGO: 11, NOTAS: 12, ESTADO_CIERRE: 13,
      MOTIVO_CANCELACION: 14, ATENDIDO_POR: 15
    } : {
      ID_PEDIDO: 0, ID_CLIENTE: 1, MARCA_TIEMPO: 2, FECHA: 2, ESTADO_LOGISTICO: 3,
      TIPO_ENTREGA: 4, DIRECCION_ENTREGA: 5, TELEFONO_CONTACTO: 6, SUBTOTAL: 7,
      DESCUENTO: 8, TOTAL: 9, METODO_PAGO: 10, NOTAS: 11, ESTADO_CIERRE: 12,
      MOTIVO_CANCELACION: 13, ATENDIDO_POR: 14
    };
    
    // Crear mapa de clientes
    const clientesMap = {};
    for (let i = 1; i < dataUsuarios.length; i++) {
      const row = dataUsuarios[i];
      const idCliente = String(row[0]).trim().toUpperCase();
      clientesMap[idCliente] = { nombre: row[3] || 'Sin nombre', email: row[1] || '', telefono: row[4] || '', direccion: row[5] || '' };
    }
    
    const pedidos = [];
    
    // Normalizar fecha del filtro a string YYYY-MM-DD
    let fechaFiltroStr = null;
    if (filtros && filtros.fecha) {
      fechaFiltroStr = String(filtros.fecha).trim();
      if (!/^\d{4}-\d{2}-\d{2}$/.test(fechaFiltroStr)) {
        fechaFiltroStr = normalizarFechaAString(filtros.fecha);
      }
      Logger.log('Fecha filtro normalizada: ' + fechaFiltroStr);
    }
    
    for (let i = 1; i < dataPedidos.length; i++) {
      const row = dataPedidos[i];
      const marcaTiempo = row[COL.MARCA_TIEMPO];
      const fechaPedido = marcaTiempo instanceof Date ? marcaTiempo : new Date(marcaTiempo);
      const estadoLogistico = String(row[COL.ESTADO_LOGISTICO] || 'Solicitado').trim();
      const estadoCierre = String(row[COL.ESTADO_CIERRE] || 'Abierto').trim();
      const tipoEntrega = String(row[COL.TIPO_ENTREGA] || '').trim();
      
      // Filtro de fecha usando comparación de strings
      if (fechaFiltroStr) {
        let fechaPedidoStr = null;
        if (tieneNuevaEstructura && row[COL.FECHA]) {
          fechaPedidoStr = normalizarFechaAString(row[COL.FECHA]);
        } else {
          fechaPedidoStr = normalizarFechaAString(marcaTiempo);
        }
        // Comparación directa de strings - SIN problemas de timezone
        if (fechaPedidoStr !== fechaFiltroStr) {
          continue;
        }
      }
      
      // FILTRO: Estado Logístico (Etapa)
      if (filtros && filtros.etapa && filtros.etapa !== 'Todos') {
        if (estadoLogistico.toLowerCase() !== filtros.etapa.toLowerCase()) continue;
      }
      
      // FILTRO: Estado de Cierre Contable
      if (filtros && filtros.estadoCierre && filtros.estadoCierre !== 'Todos') {
        if (estadoCierre.toLowerCase() !== filtros.estadoCierre.toLowerCase()) continue;
      }
      
      // FILTRO: Tipo de Entrega (incluye opción especial "Mesas")
      const idMesa = tieneNuevaEstructura && row[17] ? String(row[17]).trim() : '';
      if (filtros && filtros.tipoEntrega && filtros.tipoEntrega !== 'Todos') {
        if (filtros.tipoEntrega.toLowerCase() === 'mesas') {
          // Filtro especial: mostrar solo pedidos de mesa (tienen idMesa o tipoEntrega = Mesa)
          if (!idMesa && tipoEntrega.toLowerCase() !== 'mesa') continue;
        } else {
          if (tipoEntrega.toLowerCase() !== filtros.tipoEntrega.toLowerCase()) continue;
        }
      }
      
      // FILTRO: Vista según rol
      if (filtros && filtros.vistaRol) {
        if (filtros.vistaRol === 'cocina') {
          if (!['solicitado', 'recibido'].includes(estadoLogistico.toLowerCase())) continue;
        } else if (filtros.vistaRol === 'domicilio') {
          if (estadoLogistico.toLowerCase() !== 'enviado') continue;
        }
      }
      
      const idCliente = String(row[COL.ID_CLIENTE]).trim().toUpperCase();
      const cliente = clientesMap[idCliente] || { nombre: 'Cliente no encontrado', email: '', telefono: '', direccion: '' };

      // MEJORA #1: Resolver nombre del usuario que atendió el pedido
      const atendidoPorRaw = String(row[COL.ATENDIDO_POR] || '').trim();
      let atendidoPorNombre = atendidoPorRaw;
      // Si el valor parece ser un ID de usuario (ej: US001, US002), buscar el nombre
      if (atendidoPorRaw && /^US\d+$/i.test(atendidoPorRaw)) {
        const usuarioAtendio = clientesMap[atendidoPorRaw.toUpperCase()];
        if (usuarioAtendio && usuarioAtendio.nombre) {
          atendidoPorNombre = usuarioAtendio.nombre;
        } else {
          atendidoPorNombre = 'Usuario no disponible';
        }
      } else if (!atendidoPorRaw) {
        atendidoPorNombre = 'Sistema';
      }

      pedidos.push({
        idPedido: String(row[COL.ID_PEDIDO]),
        idCliente: String(row[COL.ID_CLIENTE]),
        nombreCliente: cliente.nombre,
        emailCliente: cliente.email,
        telefonoCliente: cliente.telefono || row[COL.TELEFONO_CONTACTO],
        fechaPedido: fechaPedido.toISOString(),
        estadoLogistico: estadoLogistico,
        tipoEntrega: tipoEntrega,
        direccionEntrega: String(row[COL.DIRECCION_ENTREGA] || cliente.direccion || ''),
        telefonoContacto: String(row[COL.TELEFONO_CONTACTO] || ''),
        subtotal: Number(row[COL.SUBTOTAL]) || 0,
        descuento: Number(row[COL.DESCUENTO]) || 0,
        total: Number(row[COL.TOTAL]) || 0,
        metodoPago: String(row[COL.METODO_PAGO] || 'Efectivo'),
        notas: String(row[COL.NOTAS] || ''),
        estadoCierre: estadoCierre,
        motivoCancelacion: String(row[COL.MOTIVO_CANCELACION] || ''),
        atendidoPor: atendidoPorRaw,
        atendidoPorNombre: atendidoPorNombre,
        idMesa: idMesa
      });
    }
    
    pedidos.sort((a, b) => new Date(b.fechaPedido) - new Date(a.fechaPedido));
    Logger.log('Total pedidos encontrados: ' + pedidos.length);
    // FASE 9.5: Incluir lista de clientes en la respuesta
    return { success: true, pedidos: pedidos, total: pedidos.length, clientes: clientesLista };
  } catch (error) {
    Logger.log('ERROR en obtenerPedidosDashboard: ' + error.toString());
    Logger.log('Stack: ' + error.stack);
    return { success: false, message: 'Error al cargar pedidos: ' + error.toString(), pedidos: [] };
  }
}

          notas: String(row[COL.NOTAS] || ''),
          estadoCierre: String(row[COL.ESTADO_CIERRE] || 'Abierto'),
          motivoCancelacion: String(row[COL.MOTIVO_CANCELACION] || ''),
          atendidoPor: String(row[COL.ATENDIDO_POR] || ''),
          // V21: Campos adicionales para Mesa y Comensales
          idMesa: tieneNuevaEstructura && row[17] ? String(row[17]) : '',
          comensales: tieneNuevaEstructura && row[19] ? Number(row[19]) : 0,
          // Datos adicionales de la mesa (nombre completo y ubicación)
          nombreMesa: '',
          ubicacionMesa: ''
        };

        // Buscar datos de la mesa si existe idMesa
        if (pedido.idMesa) {
          const mesaData = mesasMap[pedido.idMesa.toUpperCase()];
          if (mesaData) {
            pedido.nombreMesa = mesaData.nombre || pedido.idMesa;
            pedido.ubicacionMesa = mesaData.ubicacion || '';
          } else {
            pedido.nombreMesa = pedido.idMesa;
          }
        }
        break;
      }
    }
    
    if (!pedido) return { success: false, message: 'Pedido no encontrado' };

    // Crear mapa de usuarios para búsqueda de nombres
    const usuariosMap = {};
    for (let i = 1; i < dataUsuarios.length; i++) {
      const idUsuario = String(dataUsuarios[i][0]).trim().toUpperCase();
      usuariosMap[idUsuario] = {
        nombre: dataUsuarios[i][3] || 'Sin nombre',
        email: dataUsuarios[i][1] || '',
        telefono: dataUsuarios[i][4] || '',
        direccion: dataUsuarios[i][5] || ''
      };
    }

    // Buscar datos del cliente
    const idClienteStr = pedido.idCliente.trim().toUpperCase();
    const clienteData = usuariosMap[idClienteStr];
    if (clienteData) {
      pedido.nombreCliente = clienteData.nombre;
      pedido.emailCliente = clienteData.email;
      pedido.telefonoCliente = clienteData.telefono;
      pedido.direccionCliente = clienteData.direccion;
    }

    // MEJORA #1: Resolver nombre del usuario que atendió el pedido
    const atendidoPorRaw = pedido.atendidoPor || '';
    if (atendidoPorRaw && /^US\d+$/i.test(atendidoPorRaw)) {
      const usuarioAtendio = usuariosMap[atendidoPorRaw.toUpperCase()];
      if (usuarioAtendio && usuarioAtendio.nombre) {
        pedido.atendidoPorNombre = usuarioAtendio.nombre;
      } else {
        pedido.atendidoPorNombre = 'Usuario no disponible';
      }
    } else if (!atendidoPorRaw) {
      pedido.atendidoPorNombre = 'Sistema';
    } else {
      // Si no es un ID, mantener el valor original (ya es un nombre)
      pedido.atendidoPorNombre = atendidoPorRaw;
    }
    
    const items = [];
    for (let i = 1; i < dataDetalle.length; i++) {
      const row = dataDetalle[i];
      if (String(row[1]).trim().toUpperCase() === idPedidoStr) {
        let nombreProducto = 'Producto no encontrado';
        const idProductoBuscado = String(row[3]).trim().toUpperCase();
        for (let j = 1; j < dataProductos.length; j++) {
          if (String(dataProductos[j][0]).trim().toUpperCase() === idProductoBuscado) {
            nombreProducto = dataProductos[j][2];
            break;
          }
        }
        items.push({ idDetalle: String(row[0]), idProducto: String(row[3]), nombreProducto: nombreProducto, cantidad: Number(row[4]) || 1, precioUnitario: Number(row[5]) || 0, adiciones: String(row[6] || ''), precioAdiciones: Number(row[7]) || 0, subtotalItem: Number(row[8]) || 0 });
      }
    }
    pedido.items = items;
    
    const historialResult = obtenerHistorialPedido(idPedido);
    pedido.historial = historialResult.success ? historialResult.historial : [];
    
    return { success: true, pedido: pedido };
  } catch (error) {
    Logger.log('Error en obtenerDetallePedidoDashboard: ' + error.toString());
    return { success: false, message: 'Error al cargar detalle: ' + error.toString() };
  }
}

function cambiarEstadoPedido(idPedido, nuevoEstado, idUsuario, motivoCancelacion, esSuperAdmin, nombreUsuario, rolUsuario) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    Logger.log('CAMBIO DE ESTADO - Pedido: ' + idPedido + ', Nuevo Estado: ' + nuevoEstado);
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    
    const headers = dataPedidos[0];
    const tieneNuevaEstructura = headers.length >= 16;
    
    const COL = tieneNuevaEstructura ? { ESTADO_LOGISTICO: 4, ESTADO_CIERRE: 13, MOTIVO_CANCELACION: 14, ATENDIDO_POR: 15 } : { ESTADO_LOGISTICO: 3, ESTADO_CIERRE: 12, MOTIVO_CANCELACION: 13, ATENDIDO_POR: 14 };
    
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let filaEncontrada = -1;
    let estadoActual = '';
    let idCliente = '';
    
    for (let i = 1; i < dataPedidos.length; i++) {
      if (String(dataPedidos[i][0]).trim().toUpperCase() === idPedidoStr) {
        filaEncontrada = i + 1;
        estadoActual = String(dataPedidos[i][COL.ESTADO_LOGISTICO] || 'Solicitado');
        idCliente = String(dataPedidos[i][1]);
        break;
      }
    }
    
    if (filaEncontrada === -1) return { success: false, message: 'Pedido no encontrado' };
    
    const estadoNormalizado = nuevoEstado.trim();
    
    if (estadoNormalizado === 'Cancelado') {
      if (estadoActual === 'Entregado') return { success: false, message: 'No se puede cancelar un pedido ya entregado' };
      if (estadoActual === 'Cancelado') return { success: false, message: 'El pedido ya está cancelado' };
      if (!motivoCancelacion || motivoCancelacion.trim() === '') return { success: false, message: 'Debe proporcionar un motivo de cancelación' };
      
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_LOGISTICO + 1).setValue('Cancelado');
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_CIERRE + 1).setValue('Cancelado');
      sheetPedidos.getRange(filaEncontrada, COL.MOTIVO_CANCELACION + 1).setValue(motivoCancelacion);
      sheetPedidos.getRange(filaEncontrada, COL.ATENDIDO_POR + 1).setValue(idUsuario);
      
      registrarCambioEstado(idPedido, idCliente, idUsuario, nombreUsuario || 'Usuario', rolUsuario || 'N/A', estadoActual, 'Cancelado', 'Cancelado: ' + motivoCancelacion);
      return { success: true, message: 'Pedido cancelado exitosamente', nuevoEstado: 'Cancelado' };
    }
    
    const indiceActual = CONFIG.FLUJO_ESTADOS.indexOf(estadoActual);
    const indiceNuevo = CONFIG.FLUJO_ESTADOS.indexOf(estadoNormalizado);
    
    if (indiceNuevo === -1) return { success: false, message: 'Estado no válido: ' + estadoNormalizado };
    
    if (!esSuperAdmin) {
      if (indiceNuevo !== indiceActual + 1) {
        return { success: false, message: 'Transición no permitida. De "' + estadoActual + '" solo puede pasar a "' + CONFIG.FLUJO_ESTADOS[indiceActual + 1] + '"' };
      }
    }
    
    sheetPedidos.getRange(filaEncontrada, COL.ESTADO_LOGISTICO + 1).setValue(estadoNormalizado);
    
    if (estadoNormalizado === 'Entregado') {
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_CIERRE + 1).setValue('Cerrado');
    } else if (estadoActual === 'Entregado' && estadoNormalizado !== 'Entregado') {
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_CIERRE + 1).setValue('Abierto');
    }
    
    sheetPedidos.getRange(filaEncontrada, COL.ATENDIDO_POR + 1).setValue(idUsuario);
    
    const notaAuditoria = esSuperAdmin && indiceNuevo !== indiceActual + 1 ? 'Cambio directo por Super Admin' : 'Cambio secuencial';
    registrarCambioEstado(idPedido, idCliente, idUsuario, nombreUsuario || 'Usuario', rolUsuario || 'N/A', estadoActual, estadoNormalizado, notaAuditoria);
    
    return { success: true, message: 'Estado actualizado a "' + estadoNormalizado + '"', nuevoEstado: estadoNormalizado };
  } catch (error) {
    Logger.log('Error en cambiarEstadoPedido: ' + error.toString());
    return { success: false, message: 'Error al cambiar estado: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function obtenerEstadisticasDashboard(fecha) {
  try {
    const filtros = { fecha: fecha || new Date().toISOString().split('T')[0] };
    const resultado = obtenerPedidosDashboard(filtros);
    if (!resultado.success) return resultado;
    
    const pedidos = resultado.pedidos;
    const stats = {
      totalPedidos: pedidos.length,
      pendientes: pedidos.filter(p => p.estadoLogistico === 'Solicitado').length,
      enPreparacion: pedidos.filter(p => p.estadoLogistico === 'Recibido').length,
      enCamino: pedidos.filter(p => p.estadoLogistico === 'Enviado').length,
      entregados: pedidos.filter(p => p.estadoLogistico === 'Entregado').length,
      cancelados: pedidos.filter(p => p.estadoCierre === 'Cancelado').length,
      ventasTotal: pedidos.filter(p => p.estadoCierre !== 'Cancelado').reduce((sum, p) => sum + p.total, 0),
      ventasCerradas: pedidos.filter(p => p.estadoCierre === 'Cerrado').reduce((sum, p) => sum + p.total, 0)
    };

  } catch (error) {
    Logger.log('Error en obtenerEstadisticasDashboard: ' + error.toString());
    return { success: false, message: 'Error al obtener estadísticas: ' + error.toString() };
  }
}

function exportarPedidosCSV(filtros) {
  try {
    const resultado = obtenerPedidosDashboard(filtros);
    if (!resultado.success) return { success: false, message: resultado.message };
    
    const pedidos = resultado.pedidos;
    const headers = ['ID Pedido', 'Fecha', 'Cliente', 'Email', 'Teléfono', 'Estado Logístico', 'Tipo Entrega', 'Dirección', 'Subtotal', 'Descuento', 'Total', 'Método Pago', 'Estado Cierre', 'Notas'];
    let csv = headers.join(',') + '\n';
    
    pedidos.forEach(p => {
      const fecha = new Date(p.fechaPedido);
      const fechaFormateada = fecha.toLocaleDateString('es-CO') + ' ' + fecha.toLocaleTimeString('es-CO');
      const row = [p.idPedido, fechaFormateada, '"' + (p.nombreCliente || '').replace(/"/g, '""') + '"', p.emailCliente || '', p.telefonoCliente || p.telefonoContacto || '', p.estadoLogistico, p.tipoEntrega, '"' + (p.direccionEntrega || '').replace(/"/g, '""') + '"', p.subtotal, p.descuento, p.total, p.metodoPago, p.estadoCierre, '"' + (p.notas || '').replace(/"/g, '""') + '"'];
      csv += row.join(',') + '\n';
    });
    
    return { success: true, csv: csv, filename: 'pedidos_' + new Date().toISOString().split('T')[0] + '.csv' };
  } catch (error) {
    Logger.log('Error en exportarPedidosCSV: ' + error.toString());
    return { success: false, message: 'Error al exportar: ' + error.toString() };
  }
}

function migrarEstructuraPedidos() {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(60000);
    Logger.log('INICIANDO MIGRACIÓN DE ESTRUCTURA');
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheet = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    if (!sheet) return { success: false, message: 'Hoja de Pedidos no encontrada' };
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    
    if (headers.length >= 16 && String(headers[3]).toLowerCase().includes('fecha')) {
      Logger.log('La estructura ya está actualizada');
      return { success: true, message: 'La estructura ya está actualizada' };
    }
    
    sheet.insertColumnAfter(3);
    sheet.getRange(1, 3).setValue('Marca_Tiempo');
    sheet.getRange(1, 4).setValue('Fecha');
    
    for (let i = 2; i <= data.length; i++) {
      const timestamp = sheet.getRange(i, 3).getValue();
      if (timestamp instanceof Date) {
        const fechaSolo = Utilities.formatDate(timestamp, 'America/Bogota', 'dd/MM/yyyy');
        sheet.getRange(i, 4).setValue(fechaSolo);
      } else if (timestamp) {
        try {
          const fecha = new Date(timestamp);
          const fechaSolo = Utilities.formatDate(fecha, 'America/Bogota', 'dd/MM/yyyy');
          sheet.getRange(i, 4).setValue(fechaSolo);
        } catch (e) {
          sheet.getRange(i, 4).setValue('');
        }
      }
    }
    
    sheet.getRange(2, 4, data.length - 1, 1).setNumberFormat('dd/MM/yyyy');
    Logger.log('✅ Migración completada exitosamente');
    return { success: true, message: 'Migración completada. Se agregó la columna Fecha y se actualizaron ' + (data.length - 1) + ' registros.' };
  } catch (error) {
    Logger.log('❌ Error en migración: ' + error.toString());
    return { success: false, message: 'Error en migración: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function validarPinMesero(pin) {
  try {
    Logger.log('🔐 Validando PIN: ' + pin);
    
    if (!pin || String(pin).length !== 4) {
      return { success: false, message: 'PIN debe tener 4 dígitos' };
    }
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheet = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);
    
    if (!sheet) {
      Logger.log('❌ Hoja USUARIOS no encontrada');
      return { success: false, message: 'Error de configuración' };
    }
    
    const data = sheet.getDataRange().getValues();
    const pinStr = String(pin).trim();
    const pinNum = parseInt(pin, 10);
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const pinUsuario = row[10];
      const activo = row[8];
      const rol = String(row[7] || '').trim();
      
      let pinCoincide = false;
      if (pinUsuario !== null && pinUsuario !== undefined && pinUsuario !== '') {
        const pinUsuarioStr = String(pinUsuario).trim();
        const pinUsuarioNum = parseInt(pinUsuario, 10);
        pinCoincide = (pinUsuarioStr === pinStr) || (pinUsuarioNum === pinNum);
      }
      
      if (pinCoincide && activo === true) {
        const rolesPermitidos = ['Administrador', 'Admin', 'Mesero', 'Cocina', 'Cajero', 'Domiciliario'];
        const esRolPermitido = rolesPermitidos.some(r => r.toLowerCase() === rol.toLowerCase());
        
        if (!esRolPermitido) {
          continue;
        }
        
        Logger.log('✅ PIN válido - Usuario: ' + row[3] + ' | Rol: ' + rol);
        
        return {
          success: true,
          usuario: {
            id: String(row[0]).trim(),
            nombre: row[3] || 'Usuario',
            email: row[1] || '',
            telefono: row[4] || '',
            rol: rol,
            esAdmin: ['Administrador', 'Admin'].some(r => r.toLowerCase() === rol.toLowerCase())
          }
        };
      }
    }
    
    Logger.log('❌ PIN no encontrado');
    return { success: false, message: 'PIN inválido' };
    
  } catch (error) {
    Logger.log('❌ Error: ' + error.toString());
    return { success: false, message: 'Error: ' + error.toString() };
  }
}

function obtenerMesasPanel() {

    if (!sheet) {
      Logger.log('⚠️ Hoja Mesas no existe, creando...');
      return crearTablaMesasConDatos();
    }
    
    const data = sheet.getDataRange().getValues();
    
    if (data.length <= 1) {
      Logger.log('⚠️ Hoja Mesas vacía, agregando datos...');
      return agregarDatosMesasExistente(sheet);
    }
    
    const mesas = [];
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const activo = row[8];
      
      const esActivo = activo === true || activo === '' || activo === null || 
                       activo === undefined || String(activo).toUpperCase() === 'TRUE' ||
                       String(activo).toUpperCase() === 'SI' || String(activo).toUpperCase() === 'SÍ';
      
      if (esActivo && row[0]) {
        let totalConsumo = 0;
        const idPedidoActivo = String(row[5] || '').trim();
        
        if (idPedidoActivo && idPedidoActivo !== '') {
          totalConsumo = obtenerTotalPedidoMesa(idPedidoActivo);
        }
        
        mesas.push({
          id: String(row[0]).trim(),
          nombre: String(row[1] || 'Mesa').trim(),
          capacidad: Number(row[2]) || 4,
          ubicacion: String(row[3] || 'Principal').trim(),
          estado: String(row[4] || 'Libre').trim(),
          idPedidoActivo: idPedidoActivo,
          meseroAsignado: String(row[6] || '').trim(),
          horaOcupacion: row[7] ? new Date(row[7]).toISOString() : null,
          totalConsumo: totalConsumo
        });
      }
    }
    
    Logger.log('✅ Mesas cargadas: ' + mesas.length);
    return { success: true, mesas: mesas };
    
  } catch (error) {
    Logger.log('❌ Error en obtenerMesasPanel: ' + error.toString());
    return { success: false, mesas: [], message: error.toString() };
  }
}

function abrirMesa(idMesa, datos) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    Logger.log('Abriendo mesa: ' + idMesa);

    // ═══════════════════════════════════════════════════════════
    // VALIDACIÓN ANTIFRÁGIL: Validar datos de entrada
    // ═══════════════════════════════════════════════════════════
    if (!idMesa) {
      return { success: false, message: 'ID de mesa requerido' };
    }

    // Validar comensales si se proporcionan
    if (datos && datos.comensales) {
      var validComensales = validarComensales(datos.comensales);
      if (!validComensales.valido) {
        Logger.log('[MESA] Validacion comensales fallida: ' + validComensales.error);
        return { success: false, message: validComensales.error };
      }
      datos.comensales = validComensales.valor;
    }

    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetMesas = ss.getSheetByName('Mesas');
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    
    if (!sheetMesas || !sheetPedidos) {
      return { success: false, message: 'Hojas no encontradas' };
    }
    
    const dataMesas = sheetMesas.getDataRange().getValues();
    const idMesaStr = String(idMesa).trim().toUpperCase();
    let filaEncontrada = -1;
    
    for (let i = 1; i < dataMesas.length; i++) {
      if (String(dataMesas[i][0]).trim().toUpperCase() === idMesaStr) {
        filaEncontrada = i + 1;
        const estadoActual = String(dataMesas[i][4] || '').trim();
        if (estadoActual !== 'Libre') {
          return { success: false, message: 'Mesa no disponible: ' + estadoActual };
        }
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { success: false, message: 'Mesa no encontrada' };
    }
    
    const idClienteNormalizado = obtenerOCrearClienteGenerico();
    
    const lastRow = sheetPedidos.getLastRow();
    const idPedido = 'PED' + String(lastRow).padStart(6, '0');
    const timestamp = obtenerTimestampColombia();
    const fechaSolo = obtenerFechaColombia(timestamp);
    
    const rowPedido = [
      idPedido, idClienteNormalizado, timestamp, fechaSolo, 'Solicitado',
      'Mesa', '', '', 0, 0, 0, '', datos.notas || '', 'Abierto', '',
      // V21 FIX: Guardar NOMBRE del mesero en ATENDIDO_POR para visualización correcta en Dashboard
      datos.meseroNombre || datos.meseroId || '', 'Panel Meseros', idMesa, datos.meseroId || '',
      datos.comensales || 1
    ];
    
    sheetPedidos.appendRow(rowPedido);
    
    sheetMesas.getRange(filaEncontrada, 5).setValue('Ocupada');
    sheetMesas.getRange(filaEncontrada, 6).setValue(idPedido);
    // V21 FIX: Guardar NOMBRE del mesero en lugar de ID para visualización correcta
    sheetMesas.getRange(filaEncontrada, 7).setValue(datos.meseroNombre || datos.meseroId || '');
    sheetMesas.getRange(filaEncontrada, 8).setValue(timestamp);
    
    registrarCambioEstado(idPedido, idClienteNormalizado, datos.meseroId, datos.meseroNombre || 'Mesero', 'Mesero', 'N/A', 'Solicitado', 'Mesa ' + idMesa + ' abierta');
    
    return { success: true, idPedido: idPedido };
    
  } catch (error) {
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function cerrarMesa(idMesa, datosCierre) {
  // ============================================
  // V21 ANTIFRÁGIL: LockService para cerrarMesa
  // Protege escrituras concurrentes en Mesas + Pedidos
  // ============================================
  const lock = LockService.getScriptLock();
  var idPedidoActivo = '';
  var filaEncontrada = -1;
  var estadoAnteriorPedido = '';
  var idClientePedido = '';

  try {
    // Timeout 15 segundos - operación crítica multi-hoja
    if (!lock.tryLock(15000)) {
      Logger.log('LOCK TIMEOUT: cerrarMesa - Sistema ocupado');
      return {
        success: false,
        message: 'Sistema ocupado procesando otra operación. Intente en 2 segundos.'
      };
    }

    Logger.log('========================================');
    Logger.log('CERRAR MESA V21 ANTIFRÁGIL - Inicio (LOCK ADQUIRIDO)');
    Logger.log('Mesa: ' + idMesa);
    Logger.log('Timestamp: ' + new Date().toISOString());
    Logger.log('========================================');

    if (!idMesa) {
      Logger.log('ERROR: ID de mesa no proporcionado');
      return { success: false, message: 'ID de mesa requerido' };
    }

    // ═══════════════════════════════════════════════════════════
    // VALIDACIÓN ANTIFRÁGIL: Validar datos de cierre
    // ═══════════════════════════════════════════════════════════
    if (datosCierre && datosCierre.metodoPago) {
      var validMetodo = validarMetodoPago(datosCierre.metodoPago);
      if (!validMetodo.valido) {
        Logger.log('[CIERRE] Validacion metodo pago fallida: ' + validMetodo.error);
        return { success: false, message: validMetodo.error };
      }
      datosCierre.metodoPago = validMetodo.valor;
    }

    if (datosCierre && datosCierre.descuento !== undefined) {
      var validDescuento = validarDescuento(datosCierre.descuento, datosCierre.subtotal);
      if (!validDescuento.valido) {
        Logger.log('[CIERRE] Validacion descuento fallida: ' + validDescuento.error);
        return { success: false, message: validDescuento.error };
      }
      datosCierre.descuento = validDescuento.valor;
    }

    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheetMesas = ss.getSheetByName('Mesas');
    var sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    
    if (!sheetMesas) {
      Logger.log('ERROR: Hoja Mesas no encontrada');
      return { success: false, message: 'Hoja Mesas no encontrada' };
    }
    
    if (!sheetPedidos) {
      Logger.log('ERROR: Hoja Pedidos no encontrada');
      return { success: false, message: 'Hoja Pedidos no encontrada' };
    }
    
    var dataMesas = sheetMesas.getDataRange().getValues();
    var idMesaStr = String(idMesa).trim().toUpperCase();
    
    for (var i = 1; i < dataMesas.length; i++) {
      if (String(dataMesas[i][0]).trim().toUpperCase() === idMesaStr) {
        filaEncontrada = i + 1;
        idPedidoActivo = String(dataMesas[i][5] || '').trim();
        Logger.log('Mesa encontrada - Fila: ' + filaEncontrada + ', Pedido: ' + (idPedidoActivo || 'NINGUNO'));
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      Logger.log('ERROR: Mesa no encontrada: ' + idMesaStr);
      return { success: false, message: 'Mesa no encontrada' };
    }
    
    if (!idPedidoActivo || idPedidoActivo === '') {
      Logger.log('Mesa sin pedido activo - Liberando directamente');
      
      sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
      sheetMesas.getRange(filaEncontrada, 6).setValue('');
      sheetMesas.getRange(filaEncontrada, 7).setValue('');
      sheetMesas.getRange(filaEncontrada, 8).setValue('');
      SpreadsheetApp.flush();
      
      Logger.log('Mesa liberada (sin pedido)');
      return { 
        success: true, 
        idPedido: null, 
        message: 'Mesa liberada correctamente'
      };
    }
    
    var dataPedidos = sheetPedidos.getDataRange().getValues();
    var idPedidoStr = idPedidoActivo.toUpperCase();
    var filaPedido = -1;
    var estadoLogisticoActual = '';
    var estadoCierreActual = '';
    
    for (var j = 1; j < dataPedidos.length; j++) {
      if (String(dataPedidos[j][0]).trim().toUpperCase() === idPedidoStr) {
        filaPedido = j + 1;
        idClientePedido = String(dataPedidos[j][1] || '');
        estadoLogisticoActual = String(dataPedidos[j][4] || '').trim();
        estadoCierreActual = String(dataPedidos[j][13] || '').trim();
        estadoAnteriorPedido = estadoLogisticoActual;
        Logger.log('Pedido encontrado - Fila: ' + filaPedido);
        Logger.log('Estado actual: ' + estadoLogisticoActual + ' / ' + estadoCierreActual);
        break;
      }
    }
    
    if (filaPedido === -1) {
      Logger.log('Pedido no encontrado - Liberando mesa');
      
      sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
      sheetMesas.getRange(filaEncontrada, 6).setValue('');
      sheetMesas.getRange(filaEncontrada, 7).setValue('');
      sheetMesas.getRange(filaEncontrada, 8).setValue('');
      SpreadsheetApp.flush();
      
      return { 
        success: true, 
        idPedido: idPedidoActivo, 
        message: 'Mesa liberada (pedido no encontrado)'
      };
    }
    
    var yaEstaCerrado = (estadoCierreActual.toLowerCase() === 'cerrado' || 
                         estadoCierreActual.toLowerCase() === 'cancelado');
    var yaEstaEntregado = (estadoLogisticoActual.toLowerCase() === 'entregado');
    
    if (yaEstaCerrado || yaEstaEntregado) {
      Logger.log('Pedido ya procesado - Solo liberando mesa');
      
      sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
      sheetMesas.getRange(filaEncontrada, 6).setValue('');
      sheetMesas.getRange(filaEncontrada, 7).setValue('');
      sheetMesas.getRange(filaEncontrada, 8).setValue('');
      SpreadsheetApp.flush();
      
      Logger.log('Mesa liberada (pedido ya estaba cerrado)');
      return { 
        success: true, 
        idPedido: idPedidoActivo, 
        message: 'Mesa cerrada correctamente'
      };
    }
    
    var metodoPago = (datosCierre && datosCierre.metodoPago) ? datosCierre.metodoPago : 'Efectivo';
    
    Logger.log('Actualizando pedido...');
    
    sheetPedidos.getRange(filaPedido, 5).setValue('Entregado');
    sheetPedidos.getRange(filaPedido, 12).setValue(metodoPago);
    sheetPedidos.getRange(filaPedido, 14).setValue('Cerrado');
    
    Logger.log('Pedido actualizado: Entregado, ' + metodoPago + ', Cerrado');
    
    Logger.log('Liberando mesa...');
    
    sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
    sheetMesas.getRange(filaEncontrada, 6).setValue('');
    sheetMesas.getRange(filaEncontrada, 7).setValue('');
    sheetMesas.getRange(filaEncontrada, 8).setValue('');
    
    SpreadsheetApp.flush();
    
    Logger.log('Mesa liberada correctamente');
    
    try {
      registrarCambioEstadoSinLock(
        idPedidoActivo, 
        idClientePedido || 'MESA-' + idMesa, 
        (datosCierre && datosCierre.meseroId) ? datosCierre.meseroId : '',
        (datosCierre && datosCierre.meseroNombre) ? datosCierre.meseroNombre : 'Mesero',
        'Mesero',
        estadoAnteriorPedido || 'Ocupada',
        'Entregado',
        'Mesa ' + idMesa + ' cerrada - Pago: ' + metodoPago
      );
    } catch (logError) {
      Logger.log('Advertencia historial: ' + logError.toString());
    }
    
    Logger.log('========================================');
    Logger.log('CERRAR MESA V18 - EXITO');
    Logger.log('Mesa: ' + idMesa + ' | Pedido: ' + idPedidoActivo);
    Logger.log('========================================');
    
    return { 
      success: true, 
      idPedido: idPedidoActivo, 
      message: 'Mesa cerrada correctamente'
    };
    
  } catch (error) {
    Logger.log('========================================');
    Logger.log('ERROR CRITICO en cerrarMesa');
    Logger.log('Error: ' + error.toString());
    Logger.log('========================================');
    
    try {
      if (filaEncontrada > 0) {
        var ssRecovery = SpreadsheetApp.openById(CONFIG.SHEET_ID);
        var sheetRecovery = ssRecovery.getSheetByName('Mesas');
        if (sheetRecovery) {
          sheetRecovery.getRange(filaEncontrada, 5).setValue('Libre');
          sheetRecovery.getRange(filaEncontrada, 6).setValue('');
          sheetRecovery.getRange(filaEncontrada, 7).setValue('');
          sheetRecovery.getRange(filaEncontrada, 8).setValue('');
          SpreadsheetApp.flush();
          Logger.log('Mesa liberada en recuperacion');
        }
      }
    } catch (e) {
      Logger.log('Error en recuperacion: ' + e.toString());
    }
    
    return {
      success: false,
      message: 'Error: ' + error.toString()
    };
  } finally {
    // V21 ANTIFRÁGIL: SIEMPRE liberar el lock
    try {
      lock.releaseLock();
      Logger.log('CERRAR MESA V21: Lock liberado correctamente');
    } catch (lockError) {
      Logger.log('Error liberando lock cerrarMesa: ' + lockError.toString());
    }
  }
}

function registrarCambioEstadoSinLock(idPedido, idCliente, idUsuario, nombreUsuario, rol, estadoAnterior, estadoNuevo, notas) {
  try {
    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    
    if (!sheet) {
      sheet = ss.insertSheet(CONFIG.SHEETS.PROCESO_DELIVERY);
      var headers = ['ID_Delivery', 'ID_Pedido', 'Cliente_ID', 'Usuario_Cambio', 'Nombre_Usuario', 'Area_Rol', 'Estado_Anterior', 'Estado_Nuevo', 'Fecha_Hora', 'Notas'];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.setFrozenRows(1);
    }
    
    var lastRow = sheet.getLastRow();
    var idDelivery = 'DEL' + String(lastRow).padStart(6, '0');
    var timestamp = obtenerTimestampColombia();
    
    var newRow = [
      idDelivery, 
      idPedido, 
      idCliente, 
      idUsuario, 
      nombreUsuario || 'Sistema', 
      rol || 'Sistema', 
      estadoAnterior || 'N/A', 
      estadoNuevo, 
      timestamp, 
      notas || ''
    ];
    
    sheet.appendRow(newRow);
    SpreadsheetApp.flush();
    
    Logger.log('Historial registrado: ' + idDelivery);
    return { success: true, idDelivery: idDelivery };
    
  } catch (error) {
    Logger.log('Error en historial (no critico): ' + error.toString());
    return { success: false, message: error.toString() };
  }
}

function cancelarPedidoBackend(idPedido, motivo, idMesa, datosUsuario) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    Logger.log('🚫 Cancelando pedido: ' + idPedido);
    
    if (!idPedido) return { success: false, message: 'ID de pedido no proporcionado' };
    if (!motivo || motivo.trim() === '') return { success: false, message: 'Debe proporcionar un motivo' };
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetMesas = ss.getSheetByName('Mesas');
    
    if (!sheetPedidos) return { success: false, message: 'Hoja de Pedidos no encontrada' };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let filaPedido = -1;
    let idCliente = '';
    let estadoActual = '';
    
    for (let i = 1; i < dataPedidos.length; i++) {
      if (String(dataPedidos[i][0]).trim().toUpperCase() === idPedidoStr) {
        filaPedido = i + 1;
        idCliente = String(dataPedidos[i][1]);
        estadoActual = String(dataPedidos[i][4] || 'Solicitado');
        break;
      }
    }
    
    if (filaPedido === -1) return { success: false, message: 'Pedido no encontrado' };
    if (estadoActual === 'Entregado') return { success: false, message: 'No se puede cancelar pedido entregado' };
    
    sheetPedidos.getRange(filaPedido, 5).setValue('Entregado');
    sheetPedidos.getRange(filaPedido, 14).setValue('Cancelado');
    sheetPedidos.getRange(filaPedido, 15).setValue(motivo);
    
    if (idMesa && sheetMesas) {
      const dataMesas = sheetMesas.getDataRange().getValues();
      const idMesaStr = String(idMesa).trim().toUpperCase();
      
      for (let i = 1; i < dataMesas.length; i++) {
        if (String(dataMesas[i][0]).trim().toUpperCase() === idMesaStr) {
          const filaMesa = i + 1;
          sheetMesas.getRange(filaMesa, 5).setValue('Libre');
          sheetMesas.getRange(filaMesa, 6).setValue('');
          sheetMesas.getRange(filaMesa, 7).setValue('');
          sheetMesas.getRange(filaMesa, 8).setValue('');
          break;
        }
      }
    }
    
    const usuarioId = datosUsuario && datosUsuario.id ? datosUsuario.id : 'SISTEMA';
    const usuarioNombre = datosUsuario && datosUsuario.nombre ? datosUsuario.nombre : 'Sistema';
    
    registrarCambioEstado(idPedido, idCliente, usuarioId, usuarioNombre, 'Mesero', estadoActual, 'Cancelado', 'CANCELADO: ' + motivo);
    
    return { success: true, message: 'Pedido cancelado', idPedido: idPedido };
    
  } catch (error) {
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function agregarItemsPedido(idPedido, items, datosUsuario) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    Logger.log('Agregando items: ' + idPedido);

    // ═══════════════════════════════════════════════════════════
    // VALIDACIÓN ANTIFRÁGIL: Validar items antes de procesar
    // ═══════════════════════════════════════════════════════════
    if (!idPedido) {
      return { success: false, message: 'ID de pedido requerido' };
    }

    var validItems = validarItemsPedido(items);
    if (!validItems.valido) {
      Logger.log('[ITEMS] Validacion fallida: ' + validItems.errores.join(', '));
      return { success: false, message: validItems.errores.join(', ') };
    }

    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    
    if (!sheetPedidos || !sheetDetalle) return { success: false, message: 'Hojas no encontradas' };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let filaPedido = -1;
    let idCliente = '';
    let subtotalActual = 0;
    
    for (let i = 1; i < dataPedidos.length; i++) {
      if (String(dataPedidos[i][0]).trim().toUpperCase() === idPedidoStr) {
        filaPedido = i + 1;
        idCliente = String(dataPedidos[i][1]);
        subtotalActual = Number(dataPedidos[i][8]) || 0;
        break;
      }
    }
    
    if (filaPedido === -1) return { success: false, message: 'Pedido no encontrado' };
    
    const dataDetalle = sheetDetalle.getDataRange().getValues();
    let ultimoNumero = 0;
    for (let i = 1; i < dataDetalle.length; i++) {
      if (String(dataDetalle[i][1]).trim().toUpperCase() === idPedidoStr) {
        const partes = String(dataDetalle[i][0]).split('-');
        if (partes.length > 1) {
          const num = parseInt(partes[1], 10);
          if (num > ultimoNumero) ultimoNumero = num;
        }
      }
    }
    
    let nuevoSubtotal = 0;
    items.forEach(function(item, index) {
      const idDetalle = idPedido + '-' + String(ultimoNumero + index + 1).padStart(3, '0');
      const subtotalItem = ((item.precioUnitario || 0) + (item.precioAdiciones || 0)) * (item.cantidad || 1);
      
      sheetDetalle.appendRow([
        idDetalle, idPedido, idCliente, item.idProducto, item.cantidad || 1,
        item.precioUnitario || 0, item.adiciones || '', item.precioAdiciones || 0, subtotalItem
      ]);
      nuevoSubtotal += subtotalItem;
    });
    
    const subtotalFinal = subtotalActual + nuevoSubtotal;
    sheetPedidos.getRange(filaPedido, 9).setValue(subtotalFinal);
    sheetPedidos.getRange(filaPedido, 11).setValue(subtotalFinal);
    
    Logger.log('✅ Items agregados. Total: ' + subtotalFinal);
    return { success: true, subtotal: subtotalFinal, total: subtotalFinal };
    
  } catch (error) {
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function obtenerItemsPedidoMesero(idPedido) {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const sheetProductos = ss.getSheetByName(CONFIG.SHEETS.PRODUCTOS);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);

    if (!sheetDetalle || !sheetProductos) return { success: false, items: [] };

    const dataDetalle = sheetDetalle.getDataRange().getValues();
    const dataProductos = sheetProductos.getDataRange().getValues();

    var productosMap = {};
    for (var i = 1; i < dataProductos.length; i++) {
      productosMap[String(dataProductos[i][0]).trim().toUpperCase()] = {
        nombre: dataProductos[i][2],
        precio: dataProductos[i][4]
      };
    }

    const idPedidoStr = String(idPedido).trim().toUpperCase();
    const items = [];

    for (var j = 1; j < dataDetalle.length; j++) {
      if (String(dataDetalle[j][1]).trim().toUpperCase() === idPedidoStr) {
        const idProducto = String(dataDetalle[j][3]).trim().toUpperCase();
        const producto = productosMap[idProducto] || { nombre: 'Producto', precio: 0 };

        items.push({
          idDetalle: String(dataDetalle[j][0]),
          idProducto: String(dataDetalle[j][3]),
          nombreProducto: producto.nombre,
          cantidad: Number(dataDetalle[j][4]) || 1,
          precioUnitario: Number(dataDetalle[j][5]) || 0,
          adiciones: String(dataDetalle[j][6] || ''),
          precioAdiciones: Number(dataDetalle[j][7]) || 0,
          subtotalItem: Number(dataDetalle[j][8]) || 0
        });
      }
    }

    // V18.3 FIX: Obtener estadoLogistico del pedido para el badge de cocina
    var estadoPedido = 'Solicitado';
    if (sheetPedidos) {
      var dataPedidos = sheetPedidos.getDataRange().getValues();
      var headers = dataPedidos[0] || [];
      var tieneNuevaEstructura = headers.length >= 16;
      var colEstado = tieneNuevaEstructura ? 4 : 3; // Columna del estadoLogistico

      for (var k = 1; k < dataPedidos.length; k++) {
        if (String(dataPedidos[k][0]).trim().toUpperCase() === idPedidoStr) {
          estadoPedido = String(dataPedidos[k][colEstado] || 'Solicitado').trim();
          break;
        }
      }
    }

    return { success: true, items: items, estado: estadoPedido };
  } catch (error) {
    return { success: false, items: [], estado: 'Solicitado' };

  // Protege contra duplicación de clientes por celular
  // ============================================
  const lock = LockService.getScriptLock();

  try {

      Logger.log('[DOMICILIO] LOCK TIMEOUT - Sistema ocupado');
      return {
        success: false,
        error: 'Sistema ocupado. Intente guardar el cliente nuevamente.'
      };

        return { success: false, error: validNombre.error };
      }

      if (!validEmail.valido) {
        Logger.log('[DOMICILIO] Validacion email fallida: ' + validEmail.error);
        return { success: false, error: validEmail.error };
      }

    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);

        if (datos.notas) sheet.getRange(fila, 5).setValue(datos.notas);
        SpreadsheetApp.flush();
        Logger.log('[DOMICILIO] Cliente actualizado: ' + celular);
        return { success: true, celular: celular, id: data[i][0], accion: 'actualizado' };
      }

    // CREAR nuevo cliente con 10 columnas
    var idCliente = 'CLI-DOM-' + String(sheet.getLastRow()).padStart(4, '0');
    var fechaRegistro = obtenerTimestampColombia();

    // FIX Fase 9.5: Array con 10 elementos - incluye Activo = true

      'Cliente',                    // 8: Rol

    ];

    Logger.log('[DOMICILIO] Valor Activo: ' + nuevaFila[9] + ' (tipo: ' + typeof nuevaFila[9] + ')');

    sheet.appendRow(nuevaFila);
    // CRÍTICO: Forzar escritura ANTES de liberar lock
    SpreadsheetApp.flush();

    Logger.log('[DOMICILIO] Cliente creado: ' + idCliente + ' | Celular: ' + celular);

    return { success: true, celular: celular, id: idCliente, nombre: datos.nombre, accion: 'creado' };

  } finally {
    // V21 ANTIFRÁGIL: SIEMPRE liberar el lock
    try {
      lock.releaseLock();
      Logger.log('[DOMICILIO V21] Lock liberado correctamente');
    } catch (lockError) {
      Logger.log('[DOMICILIO] Error liberando lock: ' + lockError.toString());
    }
  }

function finalizarPedido(idPedido, datosCierre) {
  // ============================================
  // V21 ANTIFRÁGIL: LockService para finalizarPedido
  // Protege escrituras concurrentes en Pedidos
  // ============================================
  const lock = LockService.getScriptLock();

  try {
    // Timeout 10 segundos - operación de una sola hoja
    if (!lock.tryLock(10000)) {
      Logger.log('LOCK TIMEOUT: finalizarPedido - Sistema ocupado');
      return {
        success: false,
        message: 'Sistema ocupado. Intente finalizar nuevamente en unos segundos.'
      };
    }

    Logger.log('========================================');
    Logger.log('FINALIZAR PEDIDO V21 ANTIFRÁGIL - Inicio (LOCK ADQUIRIDO)');
    Logger.log('Pedido: ' + idPedido);
    Logger.log('========================================');

    if (!idPedido) {
      return { success: false, message: 'ID de pedido requerido' };
    }

    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);

    if (!sheetPedidos) {
      return { success: false, message: 'Hoja Pedidos no encontrada' };
    }

    var data = sheetPedidos.getDataRange().getValues();
    var idPedidoStr = String(idPedido).trim().toUpperCase();
    var encontrado = false;

    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim().toUpperCase() === idPedidoStr) {
        var fila = i + 1;
        encontrado = true;

        // VALIDACIÓN DENTRO DEL LOCK: Verificar estado actual
        var estadoCierre = String(data[i][13] || '').trim().toLowerCase();
        if (estadoCierre === 'cerrado' || estadoCierre === 'cancelado') {
          Logger.log('Pedido ya estaba cerrado - evitando doble escritura');
          return { success: true, message: 'Pedido procesado', yaEstabaCerrado: true };
        }

        if (datosCierre && datosCierre.metodoPago) {
          sheetPedidos.getRange(fila, 12).setValue(datosCierre.metodoPago);
        }

        if (datosCierre && datosCierre.cerrar) {
          sheetPedidos.getRange(fila, 5).setValue('Entregado');
          sheetPedidos.getRange(fila, 14).setValue('Cerrado');
        }

        // CRÍTICO: Forzar escritura ANTES de liberar lock
        SpreadsheetApp.flush();
        Logger.log('Pedido finalizado correctamente (FLUSH completado)');
        break;
      }
    }

    if (!encontrado) {
      Logger.log('Pedido no encontrado: ' + idPedidoStr);
      return { success: false, message: 'Pedido no encontrado' };
    }

    return { success: true, message: 'Pedido finalizado' };

  } catch (error) {
    Logger.log('Error en finalizarPedido: ' + error.toString());
    return { success: false, message: error.toString() };
  } finally {
    // V21 ANTIFRÁGIL: SIEMPRE liberar el lock
    try {
      lock.releaseLock();
      Logger.log('FINALIZAR PEDIDO V21: Lock liberado correctamente');
    } catch (lockError) {
      Logger.log('Error liberando lock finalizarPedido: ' + lockError.toString());
    }
  }
}

    sheet.appendRow(newRow);
    // CRÍTICO: Forzar escritura ANTES de liberar lock
    SpreadsheetApp.flush();

    Logger.log('V21 Cliente nuevo creado: ' + idCliente);
    return idCliente;

  } catch (error) {
    Logger.log('Error en buscarOcrearClienteUnificado: ' + error.toString());
    return obtenerOCrearClienteGenerico();
  } finally {
    // V21 ANTIFRÁGIL: SIEMPRE liberar el lock
    try {
      lock.releaseLock();
      Logger.log('[V21] buscarOcrearClienteUnificado - Lock liberado');
    } catch (lockError) {
      Logger.log('[V21] Error liberando lock: ' + lockError.toString());
    }
  }
}

// Roles permitidos para usuarios internos
const ROLES_ADMIN_PERMITIDOS = ['Administrador', 'Mesero', 'Cocina', 'Domiciliario'];

/**
 * FUNCION PRINCIPAL: Gestion de Usuarios Internos
 * @param {string} accion - 'leer', 'guardar', 'eliminar'
 * @param {Object} datos - Datos del usuario o configuracion
 * @param {Object} usuarioSolicitante - Usuario que realiza la accion (para validar permisos)
 * @returns {Object} {success: boolean, data/error: any}
 */
function adminGestionarUsuarios(accion, datos, usuarioSolicitante) {
  var lock = LockService.getScriptLock();

  try {
    // Validar que el usuario solicitante sea Administrador
    if (!usuarioSolicitante || !esRolSuperAdmin(usuarioSolicitante.rol)) {
      Logger.log('⛔ Acceso denegado: Usuario sin permisos de administrador');
      return { success: false, error: 'Acceso denegado. Se requieren permisos de Administrador.' };
    }

    lock.waitLock(5000);
    Logger.log('🔧 ADMIN - Gestion Usuarios | Accion: ' + accion);

    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheet = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);

    if (!sheet) {
      return { success: false, error: 'Hoja USUARIOS no encontrada' };
    }

    switch (accion.toLowerCase()) {
      case 'leer':
        resultado = _adminLeerUsuarios(sheet);

        break;
      case 'eliminar':
        resultado = _adminEliminarUsuario(sheet, datos, usuarioSolicitante);
        break;
      default:
        return { success: false, error: 'Accion no reconocida: ' + accion };
    }

    SpreadsheetApp.flush(); // Asegurar escritura atomica
    return resultado;

  } catch (error) {
    Logger.log('❌ Error en adminGestionarUsuarios: ' + error.toString());
    return { success: false, error: 'Error interno: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

/**
 * LEER: Obtiene todos los usuarios internos (roles admin)
 * SEGURIDAD: Oculta contraseñas y PINs
 */
function _adminLeerUsuarios(sheet) {
  try {
    var data = sheet.getDataRange().getValues();
    var usuarios = [];

    // Empezar desde fila 1 (saltando encabezados)
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var rol = String(row[7] || '').trim();

      // Solo incluir usuarios con roles internos (no clientes de App)
      if (esRolInterno(rol) || rol === 'Mesero') {
        usuarios.push({
          id: String(row[0] || ''),
          email: String(row[1] || ''),
          password: row[2] ? '******' : '', // Ocultar password
          nombre: String(row[3] || ''),
          telefono: String(row[4] || ''),
          direccion: String(row[5] || ''),
          fechaRegistro: row[6] ? (row[6] instanceof Date ? row[6].toISOString() : String(row[6])) : '',
          rol: rol,
          activo: row[8] === true || row[8] === 'true' || row[8] === 'TRUE',
          permisos: String(row[9] || ''),
          pinAcceso: row[10] ? '****' : '', // Ocultar PIN
          fechaBaja: row[11] ? (row[11] instanceof Date ? row[11].toISOString() : String(row[11])) : '',
          usuarioQueElimino: String(row[12] || ''),
          filaSheet: i + 1 // Para referencia interna
        });
      }
    }

    // Ordenar: Activos primero, luego por nombre
    usuarios.sort(function(a, b) {
      if (a.activo !== b.activo) {
        return a.activo ? -1 : 1; // Activos primero
      }
      return (a.nombre || '').localeCompare(b.nombre || '');
    });

    Logger.log('✅ Usuarios internos leidos: ' + usuarios.length);
    return { success: true, data: usuarios, total: usuarios.length };

  } catch (error) {
    Logger.log('❌ Error en _adminLeerUsuarios: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

/**
 * GUARDAR: Crear o actualizar usuario interno
 * Valida email/telefono unicos, maneja credenciales segun rol
 */
function _adminGuardarUsuario(sheet, datos, usuarioSolicitante) {
  try {
    // Validar datos requeridos
    if (!datos) {
      return { success: false, error: 'Datos de usuario requeridos' };
    }

    var nombre = String(datos.nombre || '').trim();
    var email = String(datos.email || '').trim().toLowerCase();
    var telefono = String(datos.telefono || '').replace(/\D/g, '').trim();
    var rol = String(datos.rol || '').trim();
    var credencial = String(datos.credencial || '').trim();
    var activo = datos.activo !== false && datos.activo !== 'false';
    var idActual = datos.idActual ? String(datos.idActual).trim() : null;
    var direccion = String(datos.direccion || '').trim();

    // ═══════════════════════════════════════════
    // VALIDACIONES OBLIGATORIAS
    // ═══════════════════════════════════════════

    // Validar nombre
    if (!nombre || nombre.length < 2) {
      return { success: false, error: 'El nombre es obligatorio (minimo 2 caracteres)' };
    }

    // Validar rol permitido
    if (!ROLES_ADMIN_PERMITIDOS.includes(rol)) {
      return { success: false, error: 'Rol no valido. Permitidos: ' + ROLES_ADMIN_PERMITIDOS.join(', ') };
    }

    // Validar email (obligatorio para todos excepto Mesero)
    if (rol !== 'Mesero') {
      if (!email || !_validarFormatoEmail(email)) {
        return { success: false, error: 'Email valido es obligatorio para rol ' + rol };
      }
    }

    // Validar telefono (numerico, minimo 7 digitos)
    if (telefono && telefono.length < 7) {
      return { success: false, error: 'Telefono debe tener minimo 7 digitos' };
    }

    // Validar credencial segun rol
    var validacionCredencial = _validarCredencialSegunRol(rol, credencial, idActual);
    if (!validacionCredencial.valido) {
      return { success: false, error: validacionCredencial.error };
    }

    // Obtener datos actuales para validar unicidad
    var data = sheet.getDataRange().getValues();

    // Validar email unico (si no es Mesero)
    if (rol !== 'Mesero' && email) {
      for (var i = 1; i < data.length; i++) {
        var emailExistente = String(data[i][1] || '').trim().toLowerCase();
        var idExistente = String(data[i][0] || '').trim();

        if (emailExistente === email && idExistente !== idActual) {
          return { success: false, error: 'El email "' + email + '" ya esta registrado' };
        }
      }
    }

    // Validar telefono unico (si se proporciona)
    if (telefono) {
      for (var j = 1; j < data.length; j++) {
        var telExistente = String(data[j][4] || '').replace(/\D/g, '').trim();
        var idExistenteTel = String(data[j][0] || '').trim();

        if (telExistente === telefono && idExistenteTel !== idActual) {
          return { success: false, error: 'El telefono "' + telefono + '" ya esta registrado' };
        }
      }
    }

    // Validar PIN unico (si es Mesero o Admin)
    if ((rol === 'Mesero' || rol === 'Administrador') && credencial) {
      var pinValidacion = _validarPinUnico(sheet, data, credencial, idActual);
      if (!pinValidacion.valido) {
        return { success: false, error: pinValidacion.error };
      }
    }

    // ═══════════════════════════════════════════
    // PREPARAR DATOS SEGUN ROL
    // ═══════════════════════════════════════════

    var password = '';
    var pinAcceso = '';

    // Logica de credenciales segun rol
    if (rol === 'Mesero') {
      // Mesero: Solo PIN, sin password
      pinAcceso = credencial;
      password = '';
    } else if (rol === 'Cocina' || rol === 'Domiciliario') {
      // Cocina/Domiciliario: Solo password, sin PIN
      password = credencial;
      pinAcceso = '';
    } else if (rol === 'Administrador') {
      // Administrador: Ambos (PIN y Password)
      // Credencial principal va a password, y tambien a PIN
      password = credencial;
      pinAcceso = credencial.length === 4 ? credencial : '';

      // Si se proporciona PIN separado en datos.pin
      if (datos.pin && String(datos.pin).length === 4) {
        pinAcceso = String(datos.pin);
      }
    }

    // ═══════════════════════════════════════════
    // CREAR O ACTUALIZAR
    // ═══════════════════════════════════════════

    if (idActual) {
      // ACTUALIZAR usuario existente
      var filaActualizar = -1;

      for (var k = 1; k < data.length; k++) {
        if (String(data[k][0]).trim() === idActual) {
          filaActualizar = k + 1; // +1 porque getRange es 1-indexed
          break;
        }
      }

      if (filaActualizar === -1) {
        return { success: false, error: 'Usuario con ID "' + idActual + '" no encontrado' };
      }

      // Actualizar campos (mantener password/pin si no se proporciona nueva credencial)
      sheet.getRange(filaActualizar, 2).setValue(email); // Email

      // Solo actualizar password si se proporciona nueva credencial
      if (credencial) {
        sheet.getRange(filaActualizar, 3).setValue(password);
      }

      sheet.getRange(filaActualizar, 4).setValue(nombre);
      sheet.getRange(filaActualizar, 5).setValue(telefono);
      sheet.getRange(filaActualizar, 6).setValue(direccion);
      sheet.getRange(filaActualizar, 8).setValue(rol);
      sheet.getRange(filaActualizar, 9).setValue(activo);

      // Solo actualizar PIN si se proporciona nueva credencial
      if (credencial) {
        sheet.getRange(filaActualizar, 11).setValue(pinAcceso);
      }

      Logger.log('✅ Usuario actualizado: ' + idActual + ' | Por: ' + usuarioSolicitante.nombre);
      return {
        success: true,
        data: { id: idActual, nombre: nombre, rol: rol, accion: 'actualizado' },
        message: 'Usuario "' + nombre + '" actualizado correctamente'
      };

    } else {
      // CREAR nuevo usuario
      var timestamp = new Date().getTime();
      var nuevoId = 'USR_' + timestamp;
      var fechaRegistro = obtenerTimestampColombia();

      // Asegurar que hay suficientes columnas (13 para incluir campos de baja)
      var headers = data[0];
      if (headers.length < 13) {
        // Agregar columnas faltantes si no existen
        if (headers.length <= 11) {
          sheet.getRange(1, 12).setValue('Fecha_Baja');
        }
        if (headers.length <= 12) {
          sheet.getRange(1, 13).setValue('Usuario_Que_Elimino');
        }
      }

      var nuevaFila = [
        nuevoId,           // 0: ID
        email,             // 1: Email
        password,          // 2: Password
        nombre,            // 3: Nombre
        telefono,          // 4: Telefono
        direccion,         // 5: Direccion
        fechaRegistro,     // 6: Fecha_Registro
        rol,               // 7: Rol
        activo,            // 8: Activo
        '',                // 9: Permisos
        pinAcceso,         // 10: Pin_Acceso
        '',                // 11: Fecha_Baja
        ''                 // 12: Usuario_Que_Elimino
      ];

      sheet.appendRow(nuevaFila);

      Logger.log('✅ Usuario creado: ' + nuevoId + ' | Rol: ' + rol + ' | Por: ' + usuarioSolicitante.nombre);
      return {
        success: true,
        data: { id: nuevoId, nombre: nombre, rol: rol, accion: 'creado' },
        message: 'Usuario "' + nombre + '" creado correctamente'
      };
    }

  } catch (error) {
    Logger.log('❌ Error en _adminGuardarUsuario: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

/**
 * ELIMINAR: Soft delete - Marca como inactivo
 */
function _adminEliminarUsuario(sheet, datos, usuarioSolicitante) {
  try {
    var idEliminar = datos.id ? String(datos.id).trim() : null;

    if (!idEliminar) {
      return { success: false, error: 'ID de usuario requerido para eliminar' };
    }

    // No permitir auto-eliminacion
    if (idEliminar === usuarioSolicitante.id) {
      return { success: false, error: 'No puede desactivar su propia cuenta' };
    }

    var data = sheet.getDataRange().getValues();
    var filaEliminar = -1;
    var nombreUsuario = '';

    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === idEliminar) {
        filaEliminar = i + 1;
        nombreUsuario = String(data[i][3] || 'Usuario');
        break;
      }
    }

    if (filaEliminar === -1) {
      return { success: false, error: 'Usuario con ID "' + idEliminar + '" no encontrado' };
    }

    // Soft delete: Marcar como inactivo
    var fechaBaja = obtenerTimestampColombia();

    sheet.getRange(filaEliminar, 9).setValue(false); // Activo = FALSE
    sheet.getRange(filaEliminar, 12).setValue(fechaBaja); // Fecha_Baja
    sheet.getRange(filaEliminar, 13).setValue(usuarioSolicitante.id + ' - ' + usuarioSolicitante.nombre); // Usuario_Que_Elimino

    Logger.log('✅ Usuario desactivado: ' + idEliminar + ' | Por: ' + usuarioSolicitante.nombre);
    return {
      success: true,
      data: { id: idEliminar, nombre: nombreUsuario, accion: 'desactivado' },
      message: 'Usuario "' + nombreUsuario + '" desactivado correctamente'
    };

  } catch (error) {
    Logger.log('❌ Error en _adminEliminarUsuario: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

/**
 * Validar formato de email
 */
function _validarFormatoEmail(email) {
  var regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return regex.test(email);
}

/**
 * Validar credencial segun el rol
 */
function _validarCredencialSegunRol(rol, credencial, idActual) {
  // Si es actualizacion y no se proporciona credencial, es valido (mantiene la anterior)
  if (idActual && !credencial) {
    return { valido: true };
  }

  // Para nuevos usuarios, credencial es obligatoria
  if (!idActual && !credencial) {
    return { valido: false, error: 'Credencial obligatoria para nuevo usuario' };
  }

  if (!credencial) {
    return { valido: true };
  }

  if (rol === 'Mesero') {
    // PIN de 4 digitos
    if (!/^\d{4}$/.test(credencial)) {
      return { valido: false, error: 'El PIN para Mesero debe ser de 4 digitos numericos' };
    }
  } else if (rol === 'Cocina' || rol === 'Domiciliario') {
    // Password minimo 4 caracteres
    if (credencial.length < 4) {
      return { valido: false, error: 'La contraseña debe tener minimo 4 caracteres' };
    }
  } else if (rol === 'Administrador') {
    // Password minimo 4 caracteres
    if (credencial.length < 4) {
      return { valido: false, error: 'La contraseña de Administrador debe tener minimo 4 caracteres' };
    }
  }

  return { valido: true };
}

/**
 * Validar que el PIN no este duplicado
 */
function _validarPinUnico(sheet, data, pin, idActual) {
  for (var i = 1; i < data.length; i++) {
    var pinExistente = String(data[i][10] || '').trim();
    var idExistente = String(data[i][0] || '').trim();
    var activo = data[i][8];

    // Solo validar contra usuarios activos
    if (activo === true || activo === 'true') {
      if (pinExistente === pin && idExistente !== idActual) {
        return { valido: false, error: 'El PIN "' + pin + '" ya esta asignado a otro usuario' };
      }
    }
  }
  return { valido: true };
}

/**
 * LEER: Obtiene lista de clientes con paginacion opcional
 */
function _adminLeerClientes(sheet, filtros) {
  try {
    var data = sheet.getDataRange().getValues();
    var clientes = [];

    // Parametros de paginacion
    var pagina = (filtros && filtros.pagina) ? parseInt(filtros.pagina) : 1;
    var porPagina = (filtros && filtros.porPagina) ? parseInt(filtros.porPagina) : 50;
    var busqueda = (filtros && filtros.busqueda) ? String(filtros.busqueda).toLowerCase().trim() : '';

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var nombre = String(row[1] || '').toLowerCase();
      var celular = String(row[2] || '');
      var email = String(row[6] || '').toLowerCase();

      // Filtro de busqueda
      if (busqueda) {
        if (nombre.indexOf(busqueda) === -1 &&
            celular.indexOf(busqueda) === -1 &&
            email.indexOf(busqueda) === -1) {
          continue;
        }
      }

      clientes.push({
        id: String(row[0] || ''),
        nombre: String(row[1] || ''),
        celular: String(row[2] || ''),
        domicilio: String(row[3] || ''),
        notas: String(row[4] || ''),
        fechaRegistro: row[5] ? (row[5] instanceof Date ? row[5].toISOString() : String(row[5])) : '',
        email: String(row[6] || ''),
        rol: String(row[8] || 'Cliente'),
        activo: row[9] !== false && row[9] !== 'false' && row[9] !== 'FALSE',
        filaSheet: i + 1
      });
    }

    // Ordenar por nombre
    clientes.sort(function(a, b) {
      return (a.nombre || '').localeCompare(b.nombre || '');
    });

    // Calcular paginacion
    var totalClientes = clientes.length;
    var totalPaginas = Math.ceil(totalClientes / porPagina);
    var inicio = (pagina - 1) * porPagina;
    var fin = inicio + porPagina;
    var clientesPaginados = clientes.slice(inicio, fin);

    Logger.log('✅ Clientes leidos: ' + clientesPaginados.length + ' de ' + totalClientes);
    return {
      success: true,
      data: clientesPaginados,

    };

  } catch (error) {

  }
}

/**
 * GUARDAR: Crear o actualizar cliente
 */
function _adminGuardarCliente(sheet, datos, usuarioSolicitante) {
  try {
    if (!datos) {
      return { success: false, error: 'Datos de cliente requeridos' };
    }

    var nombre = String(datos.nombre || '').trim();
    var celular = String(datos.celular || '').replace(/\D/g, '').trim();
    var domicilio = String(datos.domicilio || '').trim();
    var email = String(datos.email || '').trim().toLowerCase();
    var notas = String(datos.notas || '').trim();
    // Aceptar tanto 'id' como 'idActual' para compatibilidad con frontend
    var idActual = datos.id ? String(datos.id).trim() : (datos.idActual ? String(datos.idActual).trim() : null);

    Logger.log('[ADMIN] Guardando cliente - ID actual: ' + idActual + ', Celular: ' + celular);

    // Validaciones
    if (!nombre || nombre.length < 2) {
      return { success: false, error: 'Nombre es obligatorio (minimo 2 caracteres)' };
    }

    if (!celular || celular.length < 7) {
      return { success: false, error: 'Celular es obligatorio (minimo 7 digitos)' };
    }

    // Validar celular unico
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      var celularExistente = String(data[i][2] || '').replace(/\D/g, '').trim();
      var idExistente = String(data[i][0] || '').trim();

      if (celularExistente === celular && idExistente !== idActual) {
        Logger.log('[ADMIN] Celular duplicado detectado: ' + celular + ' pertenece a ID: ' + idExistente);
        return { success: false, error: 'El celular "' + celular + '" ya esta registrado en otro cliente' };
      }
    }

    Logger.log('[ADMIN] Validación de celular OK: ' + celular);

    if (idActual) {
      // ACTUALIZAR cliente existente
      var filaActualizar = -1;

      for (var j = 1; j < data.length; j++) {
        if (String(data[j][0]).trim() === idActual) {
          filaActualizar = j + 1;
          break;
        }
      }

      if (filaActualizar === -1) {
        return { success: false, error: 'Cliente con ID "' + idActual + '" no encontrado' };
      }

      // Actualizar campos editables (NO modificar ID ni Fecha_Registro)
      sheet.getRange(filaActualizar, 2).setValue(nombre);
      sheet.getRange(filaActualizar, 3).setValue(celular);
      sheet.getRange(filaActualizar, 4).setValue(domicilio);
      sheet.getRange(filaActualizar, 5).setValue(notas);
      sheet.getRange(filaActualizar, 7).setValue(email);

      Logger.log('✅ Cliente actualizado: ' + idActual + ' | Por: ' + usuarioSolicitante.nombre);
      return {
        success: true,
        data: { id: idActual, nombre: nombre, accion: 'actualizado' },
        message: 'Cliente "' + nombre + '" actualizado correctamente'
      };

    } else {
      // CREAR nuevo cliente
      var nuevoId = celular; // Usar celular como ID (patron existente)
      var fechaRegistro = obtenerTimestampColombia();

      var nuevaFila = [
        nuevoId,           // 0: ID_Key_Clientes
        nombre,            // 1: Nombre_Completo
        celular,           // 2: Celular
        domicilio,         // 3: Domicilio
        notas,             // 4: Notas_Adicionales
        fechaRegistro,     // 5: Fecha_Registro
        email,             // 6: Email
        '',                // 7: Password
        'Cliente',         // 8: Rol
        true               // 9: Activo
      ];

      sheet.appendRow(nuevaFila);

      Logger.log('✅ Cliente creado: ' + nuevoId + ' | Por: ' + usuarioSolicitante.nombre);
      return {
        success: true,
        data: { id: nuevoId, nombre: nombre, accion: 'creado' },
        message: 'Cliente "' + nombre + '" creado correctamente'
      };
    }

  } catch (error) {
    Logger.log('❌ Error en _adminGuardarCliente: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

/**
 * ELIMINAR: Soft delete de cliente
 */
function _adminEliminarCliente(sheet, datos, usuarioSolicitante) {
  try {
    Logger.log('[ADMIN] Intentando eliminar cliente - Datos recibidos: ' + JSON.stringify(datos));

    // Aceptar tanto 'id' como 'idActual' para compatibilidad con frontend
    var idEliminar = datos.id ? String(datos.id).trim() : (datos.idActual ? String(datos.idActual).trim() : null);

    Logger.log('[ADMIN] ID a eliminar: ' + idEliminar);

    if (!idEliminar) {
      Logger.log('[ADMIN] ERROR: No se recibió ID de cliente');
      return { success: false, error: 'ID de cliente requerido para eliminar' };
    }

    var data = sheet.getDataRange().getValues();
    var filaEliminar = -1;
    var nombreCliente = '';

    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === idEliminar) {
        filaEliminar = i + 1;
        nombreCliente = String(data[i][1] || 'Cliente');
        Logger.log('[ADMIN] Cliente encontrado en fila: ' + filaEliminar + ' - Nombre: ' + nombreCliente);
        break;
      }
    }

    if (filaEliminar === -1) {
      Logger.log('[ADMIN] ERROR: Cliente no encontrado con ID: ' + idEliminar);
      return { success: false, error: 'Cliente con ID "' + idEliminar + '" no encontrado' };
    }

    // Soft delete: Marcar como inactivo
    sheet.getRange(filaEliminar, 10).setValue(false); // Activo = FALSE

    Logger.log('✅ Cliente desactivado: ' + idEliminar + ' | Por: ' + usuarioSolicitante.nombre);
    return {
      success: true,
      data: { id: idEliminar, nombre: nombreCliente, accion: 'desactivado' },
      message: 'Cliente "' + nombreCliente + '" desactivado correctamente'
    };

  } catch (error) {
    Logger.log('❌ Error en _adminEliminarCliente: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

/**
 * FUNCION AUXILIAR: Reactivar usuario/cliente desactivado
 */
function adminReactivarEntidad(tipo, id, usuarioSolicitante) {
  var lock = LockService.getScriptLock();

  try {
    if (!usuarioSolicitante || !esRolSuperAdmin(usuarioSolicitante.rol)) {
      return { success: false, error: 'Acceso denegado' };
    }

    lock.waitLock(5000);

    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheet;
    var colActivo;

    if (tipo === 'usuario') {
      sheet = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);
      colActivo = 9; // Columna Activo en USUARIOS
    } else if (tipo === 'cliente') {
      sheet = ss.getSheetByName(CONFIG.SHEETS.CLIENTES);
      colActivo = 10; // Columna Activo en CLIENTES
    } else {
      return { success: false, error: 'Tipo no valido. Use "usuario" o "cliente"' };
    }

    if (!sheet) {
      return { success: false, error: 'Hoja no encontrada' };
    }

    var data = sheet.getDataRange().getValues();

    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(id).trim()) {
        sheet.getRange(i + 1, colActivo).setValue(true);

        // Limpiar fecha de baja si es usuario
        if (tipo === 'usuario') {
          sheet.getRange(i + 1, 12).setValue(''); // Fecha_Baja
          sheet.getRange(i + 1, 13).setValue(''); // Usuario_Que_Elimino
        }

        SpreadsheetApp.flush();
        Logger.log('✅ ' + tipo + ' reactivado: ' + id);
        return { success: true, message: tipo + ' reactivado correctamente' };
      }
    }

    return { success: false, error: tipo + ' no encontrado' };

  } catch (error) {
    Logger.log('❌ Error en adminReactivarEntidad: ' + error.toString());
    return { success: false, error: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

  };

    var lock1 = LockService.getScriptLock();

      Logger.log('✅ TEST 1: Lock adquirido correctamente');

      lock1.releaseLock();
      Logger.log('✅ TEST 1: Lock liberado correctamente');
      resultados.testRelease = true;

      Logger.log('❌ TEST 1: No se pudo adquirir lock');
    }

    Logger.log('❌ TEST 1 ERROR: ' + e.toString());
  }

  // TEST 2: Verificar que segundo lock espera (simula concurrencia)

    var lock2 = LockService.getScriptLock();
    var lock3 = LockService.getScriptLock();

    if (lock2.tryLock(5000)) {
      Logger.log('✅ TEST 2: Primer lock adquirido');

      // Intentar segundo lock con timeout corto
      var tiempoInicio = new Date().getTime();
      var segundoLockAdquirido = lock3.tryLock(1000);
      var tiempoEspera = new Date().getTime() - tiempoInicio;

      if (!segundoLockAdquirido && tiempoEspera >= 900) {
        Logger.log('✅ TEST 2: Timeout funcionando correctamente (' + tiempoEspera + 'ms)');
        resultados.testTimeout = true;
      } else if (segundoLockAdquirido) {
        Logger.log('⚠️ TEST 2: Segundo lock adquirido (mismo script, comportamiento esperado)');
        resultados.testTimeout = true;
        lock3.releaseLock();
      }

      lock2.releaseLock();
    }
  } catch (e) {
    Logger.log('❌ TEST 2 ERROR: ' + e.toString());
  }

  Logger.log('========================================');
  Logger.log('RESULTADOS TEST CONCURRENCIA V21');
  Logger.log('Lock básico: ' + (resultados.testLock ? '✅ PASS' : '❌ FAIL'));
  Logger.log('Timeout: ' + (resultados.testTimeout ? '✅ PASS' : '❌ FAIL'));
  Logger.log('Release: ' + (resultados.testRelease ? '✅ PASS' : '❌ FAIL'));
  Logger.log('========================================');

  return resultados;
}
