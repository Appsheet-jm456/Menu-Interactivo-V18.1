function compararIds(id1, id2) {
  const str1 = String(id1 || '').trim().toUpperCase();
  const str2 = String(id2 || '').trim().toUpperCase();
  return str1 === str2;
}

function obtenerTimestampColombia() {
  const ahora = new Date();
  const offsetColombia = -5 * 60 * 60 * 1000;
  const offsetLocal = ahora.getTimezoneOffset() * 60 * 1000;
  return new Date(ahora.getTime() + offsetLocal + offsetColombia);
}

function obtenerFechaColombia(fecha) {
  var fechaObj = fecha ? new Date(fecha) : new Date();
  return Utilities.formatDate(fechaObj, "America/Bogota", "dd/MM/yyyy");
}

function normalizarFechaAString(fecha) {
  if (!fecha) return null;
  
  if (fecha instanceof Date) {
    return Utilities.formatDate(fecha, "America/Bogota", "yyyy-MM-dd");
  }
  
  const fechaStr = String(fecha).trim();
  
  if (/^\d{4}-\d{2}-\d{2}$/.test(fechaStr)) {
    return fechaStr;
  }
  
  if (/^\d{2}\/\d{2}\/\d{4}$/.test(fechaStr)) {
    const parts = fechaStr.split('/');
    return parts[2] + '-' + parts[1] + '-' + parts[0];
  }
  
  if (fechaStr.includes('T')) {
    return fechaStr.split('T')[0];
  }
  
  try {
    const dateObj = new Date(fechaStr);
    if (!isNaN(dateObj.getTime())) {
      return Utilities.formatDate(dateObj, "America/Bogota", "yyyy-MM-dd");
    }
  } catch (e) {
    Logger.log('Error al normalizar fecha: ' + e.toString());
  }
  
  return null;
}

function registrarCambioEstado(idPedido, idCliente, idUsuario, nombreUsuario, rol, estadoAnterior, estadoNuevo, notas) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    let sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    if (!sheet) {
      inicializarTablaProcesoDelivery();
      sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    }
    const lastRow = sheet.getLastRow();
    const idDelivery = 'DEL' + String(lastRow).padStart(6, '0');
    const timestamp = obtenerTimestampColombia();
    const newRow = [idDelivery, idPedido, idCliente, idUsuario, nombreUsuario || 'Sistema', rol || 'Sistema', estadoAnterior || 'N/A', estadoNuevo, timestamp, notas || ''];
    sheet.appendRow(newRow);
    Logger.log('✅ Cambio de estado registrado: ' + idDelivery);
    return { success: true, idDelivery: idDelivery };
  } catch (error) {
    Logger.log('❌ Error al registrar cambio: ' + error.toString());
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function obtenerHistorialPedido(idPedido) {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheet = ss.getSheetByName(CONFIG.SHEETS.PROCESO_DELIVERY);
    if (!sheet) return { success: true, historial: [] };
    
    const data = sheet.getDataRange().getValues();
    const historial = [];
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (String(row[1]).trim().toUpperCase() === idPedidoStr) {
        historial.push({
          idDelivery: String(row[0]),
          idPedido: String(row[1]),
          idCliente: String(row[2]),
          idUsuario: String(row[3]),
          nombreUsuario: String(row[4]),
          rol: String(row[5]),
          estadoAnterior: String(row[6]),
          estadoNuevo: String(row[7]),
          fechaHora: row[8] instanceof Date ? row[8].toISOString() : String(row[8]),
          notas: String(row[9] || '')
        });
      }
    }
    historial.sort((a, b) => new Date(a.fechaHora) - new Date(b.fechaHora));
    return { success: true, historial: historial };
  } catch (error) {
    Logger.log('Error en obtenerHistorialPedido: ' + error.toString());
    return { success: false, message: error.toString(), historial: [] };
  }
}

function validarLogin(email, password) {
  try {
    Logger.log('Intentando login para: ' + email);
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.USUARIOS);
    const data = sheet.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const userEmail = String(row[1] || '').trim().toLowerCase();
      const userPassword = String(row[2] || '');
      const activo = row[8];
      
      if (userEmail === email.toLowerCase().trim() && userPassword === password && activo === true) {
        Logger.log('Login exitoso para usuario ID: ' + row[0]);
        const rol = row[7];
        return {
          success: true,
          user: {
            id: String(row[0]).trim(),
            email: row[1],
            nombre: row[3],
            telefono: row[4],
            direccion: row[5],
            rol: rol,
            permisos: row[9],
            esInterno: esRolInterno(rol),
            esSuperAdmin: esRolSuperAdmin(rol)
          }
        };
      }
    }
    Logger.log('Login fallido - credenciales incorrectas');
    return { success: false, message: 'Usuario o contraseña incorrectos' };
  } catch (error) {
    Logger.log('Error en validarLogin: ' + error.toString());
    return { success: false, message: 'Error al validar credenciales: ' + error.toString() };
  }
}

function validarLoginDashboard(email, password) {
  try {
    const result = validarLogin(email, password);
    if (!result.success) return result;
    if (!result.user.esInterno) {
      return { success: false, message: 'No tiene permisos para acceder al Dashboard. Solo personal interno.' };
    }
    return result;
  } catch (error) {
    Logger.log('Error en validarLoginDashboard: ' + error.toString());
    return { success: false, message: 'Error al validar credenciales: ' + error.toString() };
  }
}

function registrarUsuario(datos) {

  try {
    lock.waitLock(30000);
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.USUARIOS);

      user: { id: newId, email: datos.email, nombre: datos.nombre, telefono: datos.telefono, direccion: datos.direccion, rol: 'Clientes', esInterno: false, esSuperAdmin: false }

    Logger.log('Error en registrarUsuario: ' + error.toString());

    lock.releaseLock();
  }

function obtenerConfiguracion() {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.SHEET_ID).getSheetByName(CONFIG.SHEETS.CONFIGURACION);
    const data = sheet.getDataRange().getValues();
    const config = {};
    for (let i = 1; i < data.length; i++) {
      const parametro = String(data[i][0] || '').trim();
      const valor = data[i][1];
      if (parametro) {
        const parametroNormalizado = parametro.replace(/\s+/g, '').replace(/^(.)/, (m, p1) => p1.toLowerCase());
        config[parametroNormalizado] = valor;
        config[parametro] = valor;
      }
    }
    const defaults = { nombreEmpresa: 'CloudPyme Restaurant', colorPrimario: '#3B82F6', colorSecundario: '#1E40AF', colorAcento: '#F59E0B', moneda: '$', logoUrl: '', nit: '', direccion: '', telefono: '', email: '', slogan: 'Sistema de Gestión de Pedidos' };
    for (const [key, defaultValue] of Object.entries(defaults)) {
      if (!config[key] && config[key] !== 0 && config[key] !== false) config[key] = defaultValue;
    }
    return { success: true, config: config };
  } catch (error) {
    Logger.log('Error en obtenerConfiguracion: ' + error.toString());
    return { success: false, message: 'Error al cargar configuración', config: { nombreEmpresa: 'CloudPyme Restaurant', colorPrimario: '#3B82F6', colorSecundario: '#1E40AF', colorAcento: '#F59E0B', moneda: '$', logoUrl: '', nit: '', direccion: '', telefono: '' } };
  }
}

    Logger.log('Error en obtenerCategorias: ' + error.toString());
    return { success: false, message: 'Error al cargar categorías: ' + error.toString(), categorias: [] };

function obtenerMisPedidos(idCliente) {
  try {
    Logger.log('INICIO obtenerMisPedidos - ID Cliente: ' + idCliente);
    if (!idCliente) return { success: false, message: 'ID de cliente no proporcionado', pedidos: [] };
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    if (!sheetPedidos) return { success: false, message: 'Hoja de Pedidos no encontrada', pedidos: [] };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const pedidos = [];
    const idClienteStr = String(idCliente).trim().toUpperCase();
    
    for (let i = 1; i < dataPedidos.length; i++) {
      const row = dataPedidos[i];
      const pedidoIdCliente = String(row[1]).trim().toUpperCase();
      if (pedidoIdCliente === idClienteStr) {
        const tieneNuevaEstructura = row.length >= 16;
        let fechaPedido, estadoLogistico;
        if (tieneNuevaEstructura) {
          fechaPedido = row[2] instanceof Date ? row[2].toISOString() : String(row[2]);
          estadoLogistico = String(row[4] || 'Solicitado');
        } else {
          fechaPedido = row[2] instanceof Date ? row[2].toISOString() : String(row[2]);
          estadoLogistico = String(row[3] || 'Solicitado');
        }
        pedidos.push({
          idPedido: String(row[0]),
          idCliente: String(row[1]),
          fechaPedido: fechaPedido,
          estado: estadoLogistico,
          tipoEntrega: String(row[tieneNuevaEstructura ? 5 : 4] || ''),
          direccionEntrega: String(row[tieneNuevaEstructura ? 6 : 5] || ''),
          telefonoContacto: String(row[tieneNuevaEstructura ? 7 : 6] || ''),
          subtotal: Number(row[tieneNuevaEstructura ? 8 : 7]) || 0,
          descuento: Number(row[tieneNuevaEstructura ? 9 : 8]) || 0,
          total: Number(row[tieneNuevaEstructura ? 10 : 9]) || 0,
          metodoPago: String(row[tieneNuevaEstructura ? 11 : 10] || 'Efectivo'),
          notas: String(row[tieneNuevaEstructura ? 12 : 11] || '')
        });
      }
    }
    pedidos.sort((a, b) => new Date(b.fechaPedido) - new Date(a.fechaPedido));
    return { success: true, pedidos: pedidos };
  } catch (error) {
    Logger.log('ERROR en obtenerMisPedidos: ' + error.toString());
    return { success: false, message: 'Error al cargar pedidos: ' + error.toString(), pedidos: [] };
  }
}

function obtenerDetallePedido(idPedido, idCliente) {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    const idClienteStr = String(idCliente).trim().toUpperCase();
    
    let pedidoValido = false;
    for (let i = 1; i < dataPedidos.length; i++) {
      const pedidoEnFila = String(dataPedidos[i][0]).trim().toUpperCase();
      const clienteEnFila = String(dataPedidos[i][1]).trim().toUpperCase();
      if (pedidoEnFila === idPedidoStr && clienteEnFila === idClienteStr) {
        pedidoValido = true;
        break;
      }
    }
    if (!pedidoValido) return { success: false, message: 'No tiene permisos para ver este pedido', items: [] };
    
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const dataDetalle = sheetDetalle.getDataRange().getValues();
    const sheetProductos = ss.getSheetByName(CONFIG.SHEETS.PRODUCTOS);
    const dataProductos = sheetProductos.getDataRange().getValues();
    const items = [];
    
    for (let i = 1; i < dataDetalle.length; i++) {
      const row = dataDetalle[i];
      const detalleIdPedido = String(row[1]).trim().toUpperCase();
      if (detalleIdPedido === idPedidoStr) {
        let nombreProducto = 'Producto no encontrado';
        const idProductoBuscado = String(row[3]).trim().toUpperCase();
        for (let j = 1; j < dataProductos.length; j++) {
          const idProductoEnFila = String(dataProductos[j][0]).trim().toUpperCase();
          if (idProductoEnFila === idProductoBuscado) {
            nombreProducto = dataProductos[j][2];
            break;
          }
        }
        items.push({ idDetalle: String(row[0]), idPedido: String(row[1]), idProducto: String(row[3]), nombreProducto: nombreProducto, cantidad: Number(row[4]) || 1, precioUnitario: Number(row[5]) || 0, adiciones: String(row[6] || ''), precioAdiciones: Number(row[7]) || 0, subtotalItem: Number(row[8]) || 0 });
      }
    }
    return { success: true, items: items };
  } catch (error) {
    Logger.log('Error en obtenerDetallePedido: ' + error.toString());
    return { success: false, message: 'Error al cargar detalle del pedido: ' + error.toString(), items: [] };
  }
}

function crearPedido(pedidoData) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    Logger.log('Creando pedido para cliente: ' + pedidoData.idCliente);
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const lastRow = sheetPedidos.getLastRow();
    const idPedido = 'PED' + String(lastRow).padStart(6, '0');
    const timestamp = obtenerTimestampColombia();
    const fechaSolo = obtenerFechaColombia(timestamp);
    
    const rowPedido = [idPedido, pedidoData.idCliente, timestamp, fechaSolo, 'Solicitado', pedidoData.tipoEntrega || 'Recoger en tienda', pedidoData.direccionEntrega || '', pedidoData.telefonoContacto || '', pedidoData.subtotal, pedidoData.descuento || 0, pedidoData.total, pedidoData.metodoPago || 'Efectivo', pedidoData.notas || '', 'Abierto', '', ''];
    sheetPedidos.appendRow(rowPedido);
    
    registrarCambioEstado(idPedido, pedidoData.idCliente, 'SISTEMA', 'Sistema', 'Sistema', 'N/A', 'Solicitado', 'Pedido creado por cliente');
    
    pedidoData.items.forEach((item, index) => {
      const idDetalle = idPedido + '-' + String(index + 1).padStart(3, '0');
      const rowDetalle = [idDetalle, idPedido, pedidoData.idCliente, item.idProducto, item.cantidad, item.precioUnitario, item.adiciones || '', item.precioAdiciones || 0, item.subtotalItem];
      sheetDetalle.appendRow(rowDetalle);
    });
    
    Logger.log('Pedido creado exitosamente: ' + idPedido);
    return { success: true, message: 'Pedido creado exitosamente', idPedido: idPedido };
  } catch (error) {
    Logger.log('Error en crearPedido: ' + error.toString());
    return { success: false, message: 'Error al crear pedido: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

// DASHBOARD - FUNCIONES PRINCIPALES
// ============================================
function obtenerPedidosDashboard(filtros) {
  try {
    Logger.log('========================================');
    Logger.log('DASHBOARD - Obteniendo pedidos');
    Logger.log('Filtros recibidos: ' + JSON.stringify(filtros));
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetUsuarios = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);
    
    if (!sheetPedidos) return { success: false, message: 'Hoja de Pedidos no encontrada', pedidos: [] };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const dataUsuarios = sheetUsuarios ? sheetUsuarios.getDataRange().getValues() : [];
    
    const headers = dataPedidos[0];
    const tieneNuevaEstructura = headers.length >= 16 && (String(headers[2]).toLowerCase().includes('marca') || String(headers[3]).toLowerCase().includes('fecha'));
    Logger.log('Estructura detectada: ' + (tieneNuevaEstructura ? 'NUEVA' : 'ANTIGUA'));
    
    const COL = tieneNuevaEstructura ? {
      ID_PEDIDO: 0, ID_CLIENTE: 1, MARCA_TIEMPO: 2, FECHA: 3, ESTADO_LOGISTICO: 4,
      TIPO_ENTREGA: 5, DIRECCION_ENTREGA: 6, TELEFONO_CONTACTO: 7, SUBTOTAL: 8,
      DESCUENTO: 9, TOTAL: 10, METODO_PAGO: 11, NOTAS: 12, ESTADO_CIERRE: 13,
      MOTIVO_CANCELACION: 14, ATENDIDO_POR: 15
    } : {
      ID_PEDIDO: 0, ID_CLIENTE: 1, MARCA_TIEMPO: 2, FECHA: 2, ESTADO_LOGISTICO: 3,
      TIPO_ENTREGA: 4, DIRECCION_ENTREGA: 5, TELEFONO_CONTACTO: 6, SUBTOTAL: 7,
      DESCUENTO: 8, TOTAL: 9, METODO_PAGO: 10, NOTAS: 11, ESTADO_CIERRE: 12,
      MOTIVO_CANCELACION: 13, ATENDIDO_POR: 14
    };
    
    // Crear mapa de clientes
    const clientesMap = {};
    for (let i = 1; i < dataUsuarios.length; i++) {
      const row = dataUsuarios[i];
      const idCliente = String(row[0]).trim().toUpperCase();
      clientesMap[idCliente] = { nombre: row[3] || 'Sin nombre', email: row[1] || '', telefono: row[4] || '', direccion: row[5] || '' };
    }
    
    const pedidos = [];
    
    // Normalizar fecha del filtro a string YYYY-MM-DD
    let fechaFiltroStr = null;
    if (filtros && filtros.fecha) {
      fechaFiltroStr = String(filtros.fecha).trim();
      if (!/^\d{4}-\d{2}-\d{2}$/.test(fechaFiltroStr)) {
        fechaFiltroStr = normalizarFechaAString(filtros.fecha);
      }
      Logger.log('Fecha filtro normalizada: ' + fechaFiltroStr);
    }
    
    for (let i = 1; i < dataPedidos.length; i++) {
      const row = dataPedidos[i];
      const marcaTiempo = row[COL.MARCA_TIEMPO];
      const fechaPedido = marcaTiempo instanceof Date ? marcaTiempo : new Date(marcaTiempo);
      const estadoLogistico = String(row[COL.ESTADO_LOGISTICO] || 'Solicitado').trim();
      const estadoCierre = String(row[COL.ESTADO_CIERRE] || 'Abierto').trim();
      const tipoEntrega = String(row[COL.TIPO_ENTREGA] || '').trim();
      
      // Filtro de fecha usando comparación de strings
      if (fechaFiltroStr) {
        let fechaPedidoStr = null;
        if (tieneNuevaEstructura && row[COL.FECHA]) {
          fechaPedidoStr = normalizarFechaAString(row[COL.FECHA]);
        } else {
          fechaPedidoStr = normalizarFechaAString(marcaTiempo);
        }
        // Comparación directa de strings - SIN problemas de timezone
        if (fechaPedidoStr !== fechaFiltroStr) {
          continue;
        }
      }
      
      // FILTRO: Estado Logístico (Etapa)
      if (filtros && filtros.etapa && filtros.etapa !== 'Todos') {
        if (estadoLogistico.toLowerCase() !== filtros.etapa.toLowerCase()) continue;
      }
      
      // FILTRO: Estado de Cierre Contable
      if (filtros && filtros.estadoCierre && filtros.estadoCierre !== 'Todos') {
        if (estadoCierre.toLowerCase() !== filtros.estadoCierre.toLowerCase()) continue;
      }
      
      // FILTRO: Tipo de Entrega
      if (filtros && filtros.tipoEntrega && filtros.tipoEntrega !== 'Todos') {
        if (tipoEntrega.toLowerCase() !== filtros.tipoEntrega.toLowerCase()) continue;
      }
      
      // FILTRO: Vista según rol
      if (filtros && filtros.vistaRol) {
        if (filtros.vistaRol === 'cocina') {
          if (!['solicitado', 'recibido'].includes(estadoLogistico.toLowerCase())) continue;
        } else if (filtros.vistaRol === 'domicilio') {
          if (estadoLogistico.toLowerCase() !== 'enviado') continue;
        }
      }
      
      const idCliente = String(row[COL.ID_CLIENTE]).trim().toUpperCase();
      const cliente = clientesMap[idCliente] || { nombre: 'Cliente no encontrado', email: '', telefono: '', direccion: '' };
      
      pedidos.push({
        idPedido: String(row[COL.ID_PEDIDO]),
        idCliente: String(row[COL.ID_CLIENTE]),
        nombreCliente: cliente.nombre,
        emailCliente: cliente.email,
        telefonoCliente: cliente.telefono || row[COL.TELEFONO_CONTACTO],
        fechaPedido: fechaPedido.toISOString(),
        estadoLogistico: estadoLogistico,
        tipoEntrega: tipoEntrega,
        direccionEntrega: String(row[COL.DIRECCION_ENTREGA] || cliente.direccion || ''),
        telefonoContacto: String(row[COL.TELEFONO_CONTACTO] || ''),
        subtotal: Number(row[COL.SUBTOTAL]) || 0,
        descuento: Number(row[COL.DESCUENTO]) || 0,
        total: Number(row[COL.TOTAL]) || 0,
        metodoPago: String(row[COL.METODO_PAGO] || 'Efectivo'),
        notas: String(row[COL.NOTAS] || ''),
        estadoCierre: estadoCierre,
        motivoCancelacion: String(row[COL.MOTIVO_CANCELACION] || ''),
        atendidoPor: String(row[COL.ATENDIDO_POR] || '')
      });
    }
    
    pedidos.sort((a, b) => new Date(b.fechaPedido) - new Date(a.fechaPedido));
    Logger.log('Total pedidos encontrados: ' + pedidos.length);
    return { success: true, pedidos: pedidos, total: pedidos.length };
  } catch (error) {
    Logger.log('ERROR en obtenerPedidosDashboard: ' + error.toString());
    Logger.log('Stack: ' + error.stack);
    return { success: false, message: 'Error al cargar pedidos: ' + error.toString(), pedidos: [] };
  }
}

function obtenerDetallePedidoDashboard(idPedido) {
  try {
    Logger.log('DASHBOARD - Obteniendo detalle de pedido: ' + idPedido);
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const sheetProductos = ss.getSheetByName(CONFIG.SHEETS.PRODUCTOS);
    const sheetUsuarios = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const dataDetalle = sheetDetalle.getDataRange().getValues();
    const dataProductos = sheetProductos.getDataRange().getValues();
    const dataUsuarios = sheetUsuarios.getDataRange().getValues();
    
    const headers = dataPedidos[0];
    const tieneNuevaEstructura = headers.length >= 16;
    
    const COL = tieneNuevaEstructura ? {
      ID_PEDIDO: 0, ID_CLIENTE: 1, MARCA_TIEMPO: 2, FECHA: 3, ESTADO_LOGISTICO: 4,
      TIPO_ENTREGA: 5, DIRECCION_ENTREGA: 6, TELEFONO_CONTACTO: 7, SUBTOTAL: 8,
      DESCUENTO: 9, TOTAL: 10, METODO_PAGO: 11, NOTAS: 12, ESTADO_CIERRE: 13,
      MOTIVO_CANCELACION: 14, ATENDIDO_POR: 15
    } : {
      ID_PEDIDO: 0, ID_CLIENTE: 1, MARCA_TIEMPO: 2, FECHA: 2, ESTADO_LOGISTICO: 3,
      TIPO_ENTREGA: 4, DIRECCION_ENTREGA: 5, TELEFONO_CONTACTO: 6, SUBTOTAL: 7,
      DESCUENTO: 8, TOTAL: 9, METODO_PAGO: 10, NOTAS: 11, ESTADO_CIERRE: 12,
      MOTIVO_CANCELACION: 13, ATENDIDO_POR: 14
    };
    
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let pedido = null;
    
    for (let i = 1; i < dataPedidos.length; i++) {
      const row = dataPedidos[i];
      if (String(row[COL.ID_PEDIDO]).trim().toUpperCase() === idPedidoStr) {
        pedido = {
          idPedido: String(row[COL.ID_PEDIDO]),
          idCliente: String(row[COL.ID_CLIENTE]),
          fechaPedido: row[COL.MARCA_TIEMPO] instanceof Date ? row[COL.MARCA_TIEMPO].toISOString() : String(row[COL.MARCA_TIEMPO]),
          estadoLogistico: String(row[COL.ESTADO_LOGISTICO] || 'Solicitado'),
          tipoEntrega: String(row[COL.TIPO_ENTREGA] || ''),
          direccionEntrega: String(row[COL.DIRECCION_ENTREGA] || ''),
          telefonoContacto: String(row[COL.TELEFONO_CONTACTO] || ''),
          subtotal: Number(row[COL.SUBTOTAL]) || 0,
          descuento: Number(row[COL.DESCUENTO]) || 0,
          total: Number(row[COL.TOTAL]) || 0,
          metodoPago: String(row[COL.METODO_PAGO] || 'Efectivo'),
          notas: String(row[COL.NOTAS] || ''),
          estadoCierre: String(row[COL.ESTADO_CIERRE] || 'Abierto'),
          motivoCancelacion: String(row[COL.MOTIVO_CANCELACION] || ''),
          atendidoPor: String(row[COL.ATENDIDO_POR] || '')
        };
        break;
      }
    }
    
    if (!pedido) return { success: false, message: 'Pedido no encontrado' };
    
    const idClienteStr = pedido.idCliente.trim().toUpperCase();
    for (let i = 1; i < dataUsuarios.length; i++) {
      if (String(dataUsuarios[i][0]).trim().toUpperCase() === idClienteStr) {
        pedido.nombreCliente = dataUsuarios[i][3] || 'Sin nombre';
        pedido.emailCliente = dataUsuarios[i][1] || '';
        pedido.telefonoCliente = dataUsuarios[i][4] || '';
        pedido.direccionCliente = dataUsuarios[i][5] || '';
        break;
      }
    }
    
    const items = [];
    for (let i = 1; i < dataDetalle.length; i++) {
      const row = dataDetalle[i];
      if (String(row[1]).trim().toUpperCase() === idPedidoStr) {
        let nombreProducto = 'Producto no encontrado';
        const idProductoBuscado = String(row[3]).trim().toUpperCase();
        for (let j = 1; j < dataProductos.length; j++) {
          if (String(dataProductos[j][0]).trim().toUpperCase() === idProductoBuscado) {
            nombreProducto = dataProductos[j][2];
            break;
          }
        }
        items.push({ idDetalle: String(row[0]), idProducto: String(row[3]), nombreProducto: nombreProducto, cantidad: Number(row[4]) || 1, precioUnitario: Number(row[5]) || 0, adiciones: String(row[6] || ''), precioAdiciones: Number(row[7]) || 0, subtotalItem: Number(row[8]) || 0 });
      }
    }
    pedido.items = items;
    
    const historialResult = obtenerHistorialPedido(idPedido);
    pedido.historial = historialResult.success ? historialResult.historial : [];
    
    return { success: true, pedido: pedido };
  } catch (error) {
    Logger.log('Error en obtenerDetallePedidoDashboard: ' + error.toString());
    return { success: false, message: 'Error al cargar detalle: ' + error.toString() };
  }
}

function cambiarEstadoPedido(idPedido, nuevoEstado, idUsuario, motivoCancelacion, esSuperAdmin, nombreUsuario, rolUsuario) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    Logger.log('CAMBIO DE ESTADO - Pedido: ' + idPedido + ', Nuevo Estado: ' + nuevoEstado);
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    
    const headers = dataPedidos[0];
    const tieneNuevaEstructura = headers.length >= 16;
    
    const COL = tieneNuevaEstructura ? { ESTADO_LOGISTICO: 4, ESTADO_CIERRE: 13, MOTIVO_CANCELACION: 14, ATENDIDO_POR: 15 } : { ESTADO_LOGISTICO: 3, ESTADO_CIERRE: 12, MOTIVO_CANCELACION: 13, ATENDIDO_POR: 14 };
    
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let filaEncontrada = -1;
    let estadoActual = '';
    let idCliente = '';
    
    for (let i = 1; i < dataPedidos.length; i++) {
      if (String(dataPedidos[i][0]).trim().toUpperCase() === idPedidoStr) {
        filaEncontrada = i + 1;
        estadoActual = String(dataPedidos[i][COL.ESTADO_LOGISTICO] || 'Solicitado');
        idCliente = String(dataPedidos[i][1]);
        break;
      }
    }
    
    if (filaEncontrada === -1) return { success: false, message: 'Pedido no encontrado' };
    
    const estadoNormalizado = nuevoEstado.trim();
    
    if (estadoNormalizado === 'Cancelado') {
      if (estadoActual === 'Entregado') return { success: false, message: 'No se puede cancelar un pedido ya entregado' };
      if (estadoActual === 'Cancelado') return { success: false, message: 'El pedido ya está cancelado' };
      if (!motivoCancelacion || motivoCancelacion.trim() === '') return { success: false, message: 'Debe proporcionar un motivo de cancelación' };
      
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_LOGISTICO + 1).setValue('Cancelado');
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_CIERRE + 1).setValue('Cancelado');
      sheetPedidos.getRange(filaEncontrada, COL.MOTIVO_CANCELACION + 1).setValue(motivoCancelacion);
      sheetPedidos.getRange(filaEncontrada, COL.ATENDIDO_POR + 1).setValue(idUsuario);
      
      registrarCambioEstado(idPedido, idCliente, idUsuario, nombreUsuario || 'Usuario', rolUsuario || 'N/A', estadoActual, 'Cancelado', 'Cancelado: ' + motivoCancelacion);
      return { success: true, message: 'Pedido cancelado exitosamente', nuevoEstado: 'Cancelado' };
    }
    
    const indiceActual = CONFIG.FLUJO_ESTADOS.indexOf(estadoActual);
    const indiceNuevo = CONFIG.FLUJO_ESTADOS.indexOf(estadoNormalizado);
    
    if (indiceNuevo === -1) return { success: false, message: 'Estado no válido: ' + estadoNormalizado };
    
    if (!esSuperAdmin) {
      if (indiceNuevo !== indiceActual + 1) {
        return { success: false, message: 'Transición no permitida. De "' + estadoActual + '" solo puede pasar a "' + CONFIG.FLUJO_ESTADOS[indiceActual + 1] + '"' };
      }
    }
    
    sheetPedidos.getRange(filaEncontrada, COL.ESTADO_LOGISTICO + 1).setValue(estadoNormalizado);
    
    if (estadoNormalizado === 'Entregado') {
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_CIERRE + 1).setValue('Cerrado');
    } else if (estadoActual === 'Entregado' && estadoNormalizado !== 'Entregado') {
      sheetPedidos.getRange(filaEncontrada, COL.ESTADO_CIERRE + 1).setValue('Abierto');
    }
    
    sheetPedidos.getRange(filaEncontrada, COL.ATENDIDO_POR + 1).setValue(idUsuario);
    
    const notaAuditoria = esSuperAdmin && indiceNuevo !== indiceActual + 1 ? 'Cambio directo por Super Admin' : 'Cambio secuencial';
    registrarCambioEstado(idPedido, idCliente, idUsuario, nombreUsuario || 'Usuario', rolUsuario || 'N/A', estadoActual, estadoNormalizado, notaAuditoria);
    
    return { success: true, message: 'Estado actualizado a "' + estadoNormalizado + '"', nuevoEstado: estadoNormalizado };
  } catch (error) {
    Logger.log('Error en cambiarEstadoPedido: ' + error.toString());
    return { success: false, message: 'Error al cambiar estado: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function obtenerEstadisticasDashboard(fecha) {
  try {
    const filtros = { fecha: fecha || new Date().toISOString().split('T')[0] };
    const resultado = obtenerPedidosDashboard(filtros);
    if (!resultado.success) return resultado;
    
    const pedidos = resultado.pedidos;
    const stats = {
      totalPedidos: pedidos.length,
      pendientes: pedidos.filter(p => p.estadoLogistico === 'Solicitado').length,
      enPreparacion: pedidos.filter(p => p.estadoLogistico === 'Recibido').length,
      enCamino: pedidos.filter(p => p.estadoLogistico === 'Enviado').length,
      entregados: pedidos.filter(p => p.estadoLogistico === 'Entregado').length,
      cancelados: pedidos.filter(p => p.estadoCierre === 'Cancelado').length,
      ventasTotal: pedidos.filter(p => p.estadoCierre !== 'Cancelado').reduce((sum, p) => sum + p.total, 0),
      ventasCerradas: pedidos.filter(p => p.estadoCierre === 'Cerrado').reduce((sum, p) => sum + p.total, 0)
    };
    return { success: true, estadisticas: stats };
  } catch (error) {
    Logger.log('Error en obtenerEstadisticasDashboard: ' + error.toString());
    return { success: false, message: 'Error al obtener estadísticas: ' + error.toString() };
  }
}

function exportarPedidosCSV(filtros) {
  try {
    const resultado = obtenerPedidosDashboard(filtros);
    if (!resultado.success) return { success: false, message: resultado.message };
    
    const pedidos = resultado.pedidos;
    const headers = ['ID Pedido', 'Fecha', 'Cliente', 'Email', 'Teléfono', 'Estado Logístico', 'Tipo Entrega', 'Dirección', 'Subtotal', 'Descuento', 'Total', 'Método Pago', 'Estado Cierre', 'Notas'];
    let csv = headers.join(',') + '\n';
    
    pedidos.forEach(p => {
      const fecha = new Date(p.fechaPedido);
      const fechaFormateada = fecha.toLocaleDateString('es-CO') + ' ' + fecha.toLocaleTimeString('es-CO');
      const row = [p.idPedido, fechaFormateada, '"' + (p.nombreCliente || '').replace(/"/g, '""') + '"', p.emailCliente || '', p.telefonoCliente || p.telefonoContacto || '', p.estadoLogistico, p.tipoEntrega, '"' + (p.direccionEntrega || '').replace(/"/g, '""') + '"', p.subtotal, p.descuento, p.total, p.metodoPago, p.estadoCierre, '"' + (p.notas || '').replace(/"/g, '""') + '"'];
      csv += row.join(',') + '\n';
    });
    
    return { success: true, csv: csv, filename: 'pedidos_' + new Date().toISOString().split('T')[0] + '.csv' };
  } catch (error) {
    Logger.log('Error en exportarPedidosCSV: ' + error.toString());
    return { success: false, message: 'Error al exportar: ' + error.toString() };
  }
}

// MIGRACIÓN DE ESTRUCTURA (EJECUTAR UNA VEZ)
// ============================================
function migrarEstructuraPedidos() {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(60000);
    Logger.log('INICIANDO MIGRACIÓN DE ESTRUCTURA');
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheet = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    if (!sheet) return { success: false, message: 'Hoja de Pedidos no encontrada' };
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    
    if (headers.length >= 16 && String(headers[3]).toLowerCase().includes('fecha')) {
      Logger.log('La estructura ya está actualizada');
      return { success: true, message: 'La estructura ya está actualizada' };
    }
    
    sheet.insertColumnAfter(3);
    sheet.getRange(1, 3).setValue('Marca_Tiempo');
    sheet.getRange(1, 4).setValue('Fecha');
    
    for (let i = 2; i <= data.length; i++) {
      const timestamp = sheet.getRange(i, 3).getValue();
      if (timestamp instanceof Date) {
        const fechaSolo = Utilities.formatDate(timestamp, 'America/Bogota', 'dd/MM/yyyy');
        sheet.getRange(i, 4).setValue(fechaSolo);
      } else if (timestamp) {
        try {
          const fecha = new Date(timestamp);
          const fechaSolo = Utilities.formatDate(fecha, 'America/Bogota', 'dd/MM/yyyy');
          sheet.getRange(i, 4).setValue(fechaSolo);
        } catch (e) {
          sheet.getRange(i, 4).setValue('');
        }
      }
    }
    
    sheet.getRange(2, 4, data.length - 1, 1).setNumberFormat('dd/MM/yyyy');
    Logger.log('✅ Migración completada exitosamente');
    return { success: true, message: 'Migración completada. Se agregó la columna Fecha y se actualizaron ' + (data.length - 1) + ' registros.' };
  } catch (error) {
    Logger.log('❌ Error en migración: ' + error.toString());
    return { success: false, message: 'Error en migración: ' + error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function validarPinMesero(pin) {
  try {
    Logger.log('🔐 Validando PIN: ' + pin);
    
    if (!pin || String(pin).length !== 4) {
      return { success: false, message: 'PIN debe tener 4 dígitos' };
    }
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheet = ss.getSheetByName(CONFIG.SHEETS.USUARIOS);
    
    if (!sheet) {
      Logger.log('❌ Hoja USUARIOS no encontrada');
      return { success: false, message: 'Error de configuración' };
    }
    
    const data = sheet.getDataRange().getValues();
    const pinStr = String(pin).trim();
    const pinNum = parseInt(pin, 10);
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const pinUsuario = row[10];
      const activo = row[8];
      const rol = String(row[7] || '').trim();
      
      let pinCoincide = false;
      if (pinUsuario !== null && pinUsuario !== undefined && pinUsuario !== '') {
        const pinUsuarioStr = String(pinUsuario).trim();
        const pinUsuarioNum = parseInt(pinUsuario, 10);
        pinCoincide = (pinUsuarioStr === pinStr) || (pinUsuarioNum === pinNum);
      }
      
      if (pinCoincide && activo === true) {
        const rolesPermitidos = ['Administrador', 'Admin', 'Mesero', 'Cocina', 'Cajero', 'Domiciliario'];
        const esRolPermitido = rolesPermitidos.some(r => r.toLowerCase() === rol.toLowerCase());
        
        if (!esRolPermitido) {
          continue;
        }
        
        Logger.log('✅ PIN válido - Usuario: ' + row[3] + ' | Rol: ' + rol);
        
        return {
          success: true,
          usuario: {
            id: String(row[0]).trim(),
            nombre: row[3] || 'Usuario',
            email: row[1] || '',
            telefono: row[4] || '',
            rol: rol,
            esAdmin: ['Administrador', 'Admin'].some(r => r.toLowerCase() === rol.toLowerCase())
          }
        };
      }
    }
    
    Logger.log('❌ PIN no encontrado');
    return { success: false, message: 'PIN inválido' };
    
  } catch (error) {
    Logger.log('❌ Error: ' + error.toString());
    return { success: false, message: 'Error: ' + error.toString() };
  }
}

function obtenerMesasPanel() {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    let sheet = ss.getSheetByName('Mesas');
    
    if (!sheet) {
      Logger.log('⚠️ Hoja Mesas no existe, creando...');
      return crearTablaMesasConDatos();
    }
    
    const data = sheet.getDataRange().getValues();
    
    if (data.length <= 1) {
      Logger.log('⚠️ Hoja Mesas vacía, agregando datos...');
      return agregarDatosMesasExistente(sheet);
    }
    
    const mesas = [];
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const activo = row[8];
      
      const esActivo = activo === true || activo === '' || activo === null || 
                       activo === undefined || String(activo).toUpperCase() === 'TRUE' ||
                       String(activo).toUpperCase() === 'SI' || String(activo).toUpperCase() === 'SÍ';
      
      if (esActivo && row[0]) {
        let totalConsumo = 0;
        const idPedidoActivo = String(row[5] || '').trim();
        
        if (idPedidoActivo && idPedidoActivo !== '') {
          totalConsumo = obtenerTotalPedidoMesa(idPedidoActivo);
        }
        
        mesas.push({
          id: String(row[0]).trim(),
          nombre: String(row[1] || 'Mesa').trim(),
          capacidad: Number(row[2]) || 4,
          ubicacion: String(row[3] || 'Principal').trim(),
          estado: String(row[4] || 'Libre').trim(),
          idPedidoActivo: idPedidoActivo,
          meseroAsignado: String(row[6] || '').trim(),
          horaOcupacion: row[7] ? new Date(row[7]).toISOString() : null,
          totalConsumo: totalConsumo
        });
      }
    }
    
    Logger.log('✅ Mesas cargadas: ' + mesas.length);
    return { success: true, mesas: mesas };
    
  } catch (error) {
    Logger.log('❌ Error en obtenerMesasPanel: ' + error.toString());
    return { success: false, mesas: [], message: error.toString() };
  }
}

function abrirMesa(idMesa, datos) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    Logger.log('🔑 Abriendo mesa: ' + idMesa);
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetMesas = ss.getSheetByName('Mesas');
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    
    if (!sheetMesas || !sheetPedidos) {
      return { success: false, message: 'Hojas no encontradas' };
    }
    
    const dataMesas = sheetMesas.getDataRange().getValues();
    const idMesaStr = String(idMesa).trim().toUpperCase();
    let filaEncontrada = -1;
    
    for (let i = 1; i < dataMesas.length; i++) {
      if (String(dataMesas[i][0]).trim().toUpperCase() === idMesaStr) {
        filaEncontrada = i + 1;
        const estadoActual = String(dataMesas[i][4] || '').trim();
        if (estadoActual !== 'Libre') {
          return { success: false, message: 'Mesa no disponible: ' + estadoActual };
        }
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      return { success: false, message: 'Mesa no encontrada' };
    }
    
    const idClienteNormalizado = obtenerOCrearClienteGenerico();
    
    const lastRow = sheetPedidos.getLastRow();
    const idPedido = 'PED' + String(lastRow).padStart(6, '0');
    const timestamp = obtenerTimestampColombia();
    const fechaSolo = obtenerFechaColombia(timestamp);
    
    const rowPedido = [
      idPedido, idClienteNormalizado, timestamp, fechaSolo, 'Solicitado',
      'Mesa', '', '', 0, 0, 0, '', datos.notas || '', 'Abierto', '',
      datos.meseroId || '', 'Panel Meseros', idMesa, datos.meseroId || '',
      datos.comensales || 1
    ];
    
    sheetPedidos.appendRow(rowPedido);
    
    sheetMesas.getRange(filaEncontrada, 5).setValue('Ocupada');
    sheetMesas.getRange(filaEncontrada, 6).setValue(idPedido);
    sheetMesas.getRange(filaEncontrada, 7).setValue(datos.meseroId || '');
    sheetMesas.getRange(filaEncontrada, 8).setValue(timestamp);
    
    registrarCambioEstado(idPedido, idClienteNormalizado, datos.meseroId, datos.meseroNombre || 'Mesero', 'Mesero', 'N/A', 'Solicitado', 'Mesa ' + idMesa + ' abierta');
    
    return { success: true, idPedido: idPedido };
    
  } catch (error) {
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function cerrarMesa(idMesa, datosCierre) {
  var idPedidoActivo = '';
  var filaEncontrada = -1;
  var estadoAnteriorPedido = '';
  var idClientePedido = '';
  
  try {
    Logger.log('========================================');
    Logger.log('CERRAR MESA V18 FINAL - Inicio');
    Logger.log('Mesa: ' + idMesa);
    Logger.log('Timestamp: ' + new Date().toISOString());
    Logger.log('========================================');
    
    if (!idMesa) {
      Logger.log('ERROR: ID de mesa no proporcionado');
      return { success: false, message: 'ID de mesa requerido' };
    }
    
    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheetMesas = ss.getSheetByName('Mesas');
    var sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    
    if (!sheetMesas) {
      Logger.log('ERROR: Hoja Mesas no encontrada');
      return { success: false, message: 'Hoja Mesas no encontrada' };
    }
    
    if (!sheetPedidos) {
      Logger.log('ERROR: Hoja Pedidos no encontrada');
      return { success: false, message: 'Hoja Pedidos no encontrada' };
    }
    
    var dataMesas = sheetMesas.getDataRange().getValues();
    var idMesaStr = String(idMesa).trim().toUpperCase();
    
    for (var i = 1; i < dataMesas.length; i++) {
      if (String(dataMesas[i][0]).trim().toUpperCase() === idMesaStr) {
        filaEncontrada = i + 1;
        idPedidoActivo = String(dataMesas[i][5] || '').trim();
        Logger.log('Mesa encontrada - Fila: ' + filaEncontrada + ', Pedido: ' + (idPedidoActivo || 'NINGUNO'));
        break;
      }
    }
    
    if (filaEncontrada === -1) {
      Logger.log('ERROR: Mesa no encontrada: ' + idMesaStr);
      return { success: false, message: 'Mesa no encontrada' };
    }
    
    if (!idPedidoActivo || idPedidoActivo === '') {
      Logger.log('Mesa sin pedido activo - Liberando directamente');
      
      sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
      sheetMesas.getRange(filaEncontrada, 6).setValue('');
      sheetMesas.getRange(filaEncontrada, 7).setValue('');
      sheetMesas.getRange(filaEncontrada, 8).setValue('');
      SpreadsheetApp.flush();
      
      Logger.log('Mesa liberada (sin pedido)');
      return { 
        success: true, 
        idPedido: null, 
        message: 'Mesa liberada correctamente'
      };
    }
    
    var dataPedidos = sheetPedidos.getDataRange().getValues();
    var idPedidoStr = idPedidoActivo.toUpperCase();
    var filaPedido = -1;
    var estadoLogisticoActual = '';
    var estadoCierreActual = '';
    
    for (var j = 1; j < dataPedidos.length; j++) {
      if (String(dataPedidos[j][0]).trim().toUpperCase() === idPedidoStr) {
        filaPedido = j + 1;
        idClientePedido = String(dataPedidos[j][1] || '');
        estadoLogisticoActual = String(dataPedidos[j][4] || '').trim();
        estadoCierreActual = String(dataPedidos[j][13] || '').trim();
        estadoAnteriorPedido = estadoLogisticoActual;
        Logger.log('Pedido encontrado - Fila: ' + filaPedido);
        Logger.log('Estado actual: ' + estadoLogisticoActual + ' / ' + estadoCierreActual);
        break;
      }
    }
    
    if (filaPedido === -1) {
      Logger.log('Pedido no encontrado - Liberando mesa');
      
      sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
      sheetMesas.getRange(filaEncontrada, 6).setValue('');
      sheetMesas.getRange(filaEncontrada, 7).setValue('');
      sheetMesas.getRange(filaEncontrada, 8).setValue('');
      SpreadsheetApp.flush();
      
      return { 
        success: true, 
        idPedido: idPedidoActivo, 
        message: 'Mesa liberada (pedido no encontrado)'
      };
    }
    
    var yaEstaCerrado = (estadoCierreActual.toLowerCase() === 'cerrado' || 
                         estadoCierreActual.toLowerCase() === 'cancelado');
    var yaEstaEntregado = (estadoLogisticoActual.toLowerCase() === 'entregado');
    
    if (yaEstaCerrado || yaEstaEntregado) {
      Logger.log('Pedido ya procesado - Solo liberando mesa');
      
      sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
      sheetMesas.getRange(filaEncontrada, 6).setValue('');
      sheetMesas.getRange(filaEncontrada, 7).setValue('');
      sheetMesas.getRange(filaEncontrada, 8).setValue('');
      SpreadsheetApp.flush();
      
      Logger.log('Mesa liberada (pedido ya estaba cerrado)');
      return { 
        success: true, 
        idPedido: idPedidoActivo, 
        message: 'Mesa cerrada correctamente'
      };
    }
    
    var metodoPago = (datosCierre && datosCierre.metodoPago) ? datosCierre.metodoPago : 'Efectivo';
    
    Logger.log('Actualizando pedido...');
    
    sheetPedidos.getRange(filaPedido, 5).setValue('Entregado');
    sheetPedidos.getRange(filaPedido, 12).setValue(metodoPago);
    sheetPedidos.getRange(filaPedido, 14).setValue('Cerrado');
    
    Logger.log('Pedido actualizado: Entregado, ' + metodoPago + ', Cerrado');
    
    Logger.log('Liberando mesa...');
    
    sheetMesas.getRange(filaEncontrada, 5).setValue('Libre');
    sheetMesas.getRange(filaEncontrada, 6).setValue('');
    sheetMesas.getRange(filaEncontrada, 7).setValue('');
    sheetMesas.getRange(filaEncontrada, 8).setValue('');
    
    SpreadsheetApp.flush();
    
    Logger.log('Mesa liberada correctamente');
    
    try {
      registrarCambioEstadoSinLock(
        idPedidoActivo, 
        idClientePedido || 'MESA-' + idMesa, 
        (datosCierre && datosCierre.meseroId) ? datosCierre.meseroId : '',
        (datosCierre && datosCierre.meseroNombre) ? datosCierre.meseroNombre : 'Mesero',
        'Mesero',
        estadoAnteriorPedido || 'Ocupada',
        'Entregado',
        'Mesa ' + idMesa + ' cerrada - Pago: ' + metodoPago
      );
    } catch (logError) {
      Logger.log('Advertencia historial: ' + logError.toString());
    }
    
    Logger.log('========================================');
    Logger.log('CERRAR MESA V18 - EXITO');
    Logger.log('Mesa: ' + idMesa + ' | Pedido: ' + idPedidoActivo);
    Logger.log('========================================');
    
    return { 
      success: true, 
      idPedido: idPedidoActivo, 
      message: 'Mesa cerrada correctamente'
    };
    
  } catch (error) {
    Logger.log('========================================');
    Logger.log('ERROR CRITICO en cerrarMesa');
    Logger.log('Error: ' + error.toString());
    Logger.log('========================================');
    
    try {
      if (filaEncontrada > 0) {
        var ssRecovery = SpreadsheetApp.openById(CONFIG.SHEET_ID);
        var sheetRecovery = ssRecovery.getSheetByName('Mesas');
        if (sheetRecovery) {
          sheetRecovery.getRange(filaEncontrada, 5).setValue('Libre');
          sheetRecovery.getRange(filaEncontrada, 6).setValue('');
          sheetRecovery.getRange(filaEncontrada, 7).setValue('');
          sheetRecovery.getRange(filaEncontrada, 8).setValue('');
          SpreadsheetApp.flush();
          Logger.log('Mesa liberada en recuperacion');
        }
      }
    } catch (e) {
      Logger.log('Error en recuperacion: ' + e.toString());
    }
    
    return { 
      success: false, 
      message: 'Error: ' + error.toString()
    };
  }
}

function cancelarPedidoBackend(idPedido, motivo, idMesa, datosUsuario) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    Logger.log('🚫 Cancelando pedido: ' + idPedido);
    
    if (!idPedido) return { success: false, message: 'ID de pedido no proporcionado' };
    if (!motivo || motivo.trim() === '') return { success: false, message: 'Debe proporcionar un motivo' };
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetMesas = ss.getSheetByName('Mesas');
    
    if (!sheetPedidos) return { success: false, message: 'Hoja de Pedidos no encontrada' };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let filaPedido = -1;
    let idCliente = '';
    let estadoActual = '';
    
    for (let i = 1; i < dataPedidos.length; i++) {
      if (String(dataPedidos[i][0]).trim().toUpperCase() === idPedidoStr) {
        filaPedido = i + 1;
        idCliente = String(dataPedidos[i][1]);
        estadoActual = String(dataPedidos[i][4] || 'Solicitado');
        break;
      }
    }
    
    if (filaPedido === -1) return { success: false, message: 'Pedido no encontrado' };
    if (estadoActual === 'Entregado') return { success: false, message: 'No se puede cancelar pedido entregado' };
    
    sheetPedidos.getRange(filaPedido, 5).setValue('Entregado');
    sheetPedidos.getRange(filaPedido, 14).setValue('Cancelado');
    sheetPedidos.getRange(filaPedido, 15).setValue(motivo);
    
    if (idMesa && sheetMesas) {
      const dataMesas = sheetMesas.getDataRange().getValues();
      const idMesaStr = String(idMesa).trim().toUpperCase();
      
      for (let i = 1; i < dataMesas.length; i++) {
        if (String(dataMesas[i][0]).trim().toUpperCase() === idMesaStr) {
          const filaMesa = i + 1;
          sheetMesas.getRange(filaMesa, 5).setValue('Libre');
          sheetMesas.getRange(filaMesa, 6).setValue('');
          sheetMesas.getRange(filaMesa, 7).setValue('');
          sheetMesas.getRange(filaMesa, 8).setValue('');
          break;
        }
      }
    }
    
    const usuarioId = datosUsuario && datosUsuario.id ? datosUsuario.id : 'SISTEMA';
    const usuarioNombre = datosUsuario && datosUsuario.nombre ? datosUsuario.nombre : 'Sistema';
    
    registrarCambioEstado(idPedido, idCliente, usuarioId, usuarioNombre, 'Mesero', estadoActual, 'Cancelado', 'CANCELADO: ' + motivo);

function agregarItemsPedido(idPedido, items, datosUsuario) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    Logger.log('🛒 Agregando items: ' + idPedido);
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    
    if (!sheetPedidos || !sheetDetalle) return { success: false, message: 'Hojas no encontradas' };
    
    const dataPedidos = sheetPedidos.getDataRange().getValues();
    const idPedidoStr = String(idPedido).trim().toUpperCase();
    let filaPedido = -1;
    let idCliente = '';
    let subtotalActual = 0;
    
    for (let i = 1; i < dataPedidos.length; i++) {
      if (String(dataPedidos[i][0]).trim().toUpperCase() === idPedidoStr) {
        filaPedido = i + 1;
        idCliente = String(dataPedidos[i][1]);
        subtotalActual = Number(dataPedidos[i][8]) || 0;
        break;
      }
    }
    
    if (filaPedido === -1) return { success: false, message: 'Pedido no encontrado' };
    
    const dataDetalle = sheetDetalle.getDataRange().getValues();
    let ultimoNumero = 0;
    for (let i = 1; i < dataDetalle.length; i++) {
      if (String(dataDetalle[i][1]).trim().toUpperCase() === idPedidoStr) {
        const partes = String(dataDetalle[i][0]).split('-');
        if (partes.length > 1) {
          const num = parseInt(partes[1], 10);
          if (num > ultimoNumero) ultimoNumero = num;
        }
      }
    }
    
    let nuevoSubtotal = 0;
    items.forEach(function(item, index) {
      const idDetalle = idPedido + '-' + String(ultimoNumero + index + 1).padStart(3, '0');
      const subtotalItem = ((item.precioUnitario || 0) + (item.precioAdiciones || 0)) * (item.cantidad || 1);
      
      sheetDetalle.appendRow([
        idDetalle, idPedido, idCliente, item.idProducto, item.cantidad || 1,
        item.precioUnitario || 0, item.adiciones || '', item.precioAdiciones || 0, subtotalItem
      ]);
      nuevoSubtotal += subtotalItem;
    });
    
    const subtotalFinal = subtotalActual + nuevoSubtotal;
    sheetPedidos.getRange(filaPedido, 9).setValue(subtotalFinal);
    sheetPedidos.getRange(filaPedido, 11).setValue(subtotalFinal);
    
    Logger.log('✅ Items agregados. Total: ' + subtotalFinal);
    return { success: true, subtotal: subtotalFinal, total: subtotalFinal };
    
  } catch (error) {
    return { success: false, message: error.toString() };
  } finally {
    lock.releaseLock();
  }
}

function obtenerItemsPedidoMesero(idPedido) {
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    const sheetDetalle = ss.getSheetByName(CONFIG.SHEETS.DETALLE_PEDIDOS);
    const sheetProductos = ss.getSheetByName(CONFIG.SHEETS.PRODUCTOS);
    const sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);

    if (!sheetDetalle || !sheetProductos) return { success: false, items: [] };

    const dataDetalle = sheetDetalle.getDataRange().getValues();
    const dataProductos = sheetProductos.getDataRange().getValues();

    // V18.1 SYNC: Obtener Estado_Logistico de la tabla Pedidos
    var estadoLogistico = 'Solicitado';
    if (sheetPedidos) {
      const dataPedidos = sheetPedidos.getDataRange().getValues();
      const idPedidoUpper = String(idPedido).trim().toUpperCase();

      // Detectar estructura de columnas (nueva vs antigua)
      const headers = dataPedidos[0] || [];
      var colEstado = 4; // Default nueva estructura
      for (var h = 0; h < headers.length; h++) {
        if (String(headers[h]).toLowerCase().indexOf('estado_logistico') !== -1 ||
            String(headers[h]).toLowerCase() === 'estadologistico') {
          colEstado = h;
          break;
        }
      }

      // Buscar el pedido y obtener su estado
      for (var p = 1; p < dataPedidos.length; p++) {
        if (String(dataPedidos[p][0]).trim().toUpperCase() === idPedidoUpper) {
          estadoLogistico = String(dataPedidos[p][colEstado] || 'Solicitado').trim();
          break;
        }
      }
    }

    var productosMap = {};
    for (var i = 1; i < dataProductos.length; i++) {
      productosMap[String(dataProductos[i][0]).trim().toUpperCase()] = {
        nombre: dataProductos[i][2],
        precio: dataProductos[i][4]
      };
    }

    const idPedidoStr = String(idPedido).trim().toUpperCase();
    const items = [];

    for (var j = 1; j < dataDetalle.length; j++) {
      if (String(dataDetalle[j][1]).trim().toUpperCase() === idPedidoStr) {
        const idProducto = String(dataDetalle[j][3]).trim().toUpperCase();
        const producto = productosMap[idProducto] || { nombre: 'Producto', precio: 0 };

        items.push({
          idDetalle: String(dataDetalle[j][0]),
          idProducto: String(dataDetalle[j][3]),
          nombreProducto: producto.nombre,
          cantidad: Number(dataDetalle[j][4]) || 1,
          precioUnitario: Number(dataDetalle[j][5]) || 0,
          adiciones: String(dataDetalle[j][6] || ''),
          precioAdiciones: Number(dataDetalle[j][7]) || 0,
          subtotalItem: Number(dataDetalle[j][8]) || 0
        });
      }
    }

    // V18.1 SYNC: Incluir estadoLogistico en la respuesta
    return { success: true, items: items, estadoLogistico: estadoLogistico };
  } catch (error) {
    return { success: false, items: [], estadoLogistico: 'Solicitado' };
  }
}

function finalizarPedido(idPedido, datosCierre) {
  try {
    Logger.log('========================================');
    Logger.log('FINALIZAR PEDIDO V18 - Inicio');
    Logger.log('Pedido: ' + idPedido);
    Logger.log('========================================');
    
    if (!idPedido) {
      return { success: false, message: 'ID de pedido requerido' };
    }
    
    var ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    var sheetPedidos = ss.getSheetByName(CONFIG.SHEETS.PEDIDOS);
    
    if (!sheetPedidos) {
      return { success: false, message: 'Hoja Pedidos no encontrada' };
    }
    
    var data = sheetPedidos.getDataRange().getValues();
    var idPedidoStr = String(idPedido).trim().toUpperCase();
    var encontrado = false;
    
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim().toUpperCase() === idPedidoStr) {
        var fila = i + 1;
        encontrado = true;
        
        var estadoCierre = String(data[i][13] || '').trim().toLowerCase();
        if (estadoCierre === 'cerrado' || estadoCierre === 'cancelado') {
          Logger.log('Pedido ya estaba cerrado');
          return { success: true, message: 'Pedido procesado', yaEstabaCerrado: true };
        }
        
        if (datosCierre && datosCierre.metodoPago) {
          sheetPedidos.getRange(fila, 12).setValue(datosCierre.metodoPago);
        }
        
        if (datosCierre && datosCierre.cerrar) {
          sheetPedidos.getRange(fila, 5).setValue('Entregado');
          sheetPedidos.getRange(fila, 14).setValue('Cerrado');
        }
        
        SpreadsheetApp.flush();
        Logger.log('Pedido finalizado correctamente');
        break;
      }
    }
    
    if (!encontrado) {
      Logger.log('Pedido no encontrado: ' + idPedidoStr);
      return { success: false, message: 'Pedido no encontrado' };
    }
    
    return { success: true, message: 'Pedido finalizado' };
    
  } catch (error) {
    Logger.log('Error en finalizarPedido: ' + error.toString());
    return { success: false, message: error.toString() };
  }
}

function buscarOcrearClienteUnificado(datosCliente) {
  try {
    if (!datosCliente) {
      return '2222222';
    }
    
    const celularLimpio = String(datosCliente.telefono || datosCliente.celular || '').replace(/\D/g, '').trim();
    
    if (!celularLimpio || celularLimpio.length < 7) {
      return obtenerOCrearClienteGenerico();
    }
    
    const ss = SpreadsheetApp.openById(CONFIG.SHEET_ID);
    let sheet = ss.getSheetByName(CONFIG.SHEETS.CLIENTES);
    
    if (!sheet) {
      sheet = ss.insertSheet(CONFIG.SHEETS.CLIENTES);
      const headers = ['ID_Key_Clientes', 'Nombre_Completo', 'Celular', 'Domicilio', 'Notas_Adicionales', 'Fecha_Registro', 'Email', 'Password', 'Rol', 'Activo'];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
      sheet.setFrozenRows(1);
    }
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const tieneColumnaEmail = headers.length >= 7;
    
    for (let i = 1; i < data.length; i++) {
      const celularEnFila = String(data[i][2] || '').replace(/\D/g, '').trim();
      
      if (celularEnFila === celularLimpio) {
        const fila = i + 1;
        let actualizado = false;
        
        if (datosCliente.nombre && datosCliente.nombre.trim()) {
          const nombreActual = String(data[i][1] || '').trim();
          if (!nombreActual || nombreActual === 'CONSUMIDOR FINAL') {
            sheet.getRange(fila, 2).setValue(datosCliente.nombre.trim());
            actualizado = true;
          }
        }
        
        if (datosCliente.direccion && datosCliente.direccion.trim()) {
          sheet.getRange(fila, 4).setValue(datosCliente.direccion.trim());
          actualizado = true;
        }
        
        if (tieneColumnaEmail && datosCliente.email && datosCliente.email.trim()) {
          const emailActual = String(data[i][6] || '').trim();
          if (!emailActual) {
            sheet.getRange(fila, 7).setValue(datosCliente.email.trim().toLowerCase());
            actualizado = true;
          }
        }
        
        if (actualizado) {
          Logger.log('V16 Cliente actualizado: ' + celularLimpio);
        }
        
        return celularLimpio;
      }
    }
    
    const timestamp = obtenerTimestampColombia();
    const idCliente = celularLimpio;
    
    if (tieneColumnaEmail) {
      const newRow = [
        idCliente,
        datosCliente.nombre || '',
        celularLimpio,
        datosCliente.direccion || '',
        datosCliente.notas || '',
        timestamp,
        datosCliente.email ? datosCliente.email.trim().toLowerCase() : '',
        datosCliente.password || '',
        'Cliente',
        true
      ];
      sheet.appendRow(newRow);
    } else {
      const newRow = [
        idCliente,
        datosCliente.nombre || '',
        celularLimpio,
        datosCliente.direccion || '',
        datosCliente.notas || '',
        timestamp
      ];
      sheet.appendRow(newRow);
    }
    
    Logger.log('V16 Cliente nuevo creado: ' + idCliente);
    return idCliente;
    
  } catch (error) {
    Logger.log('Error en buscarOcrearClienteUnificado: ' + error.toString());
    return obtenerOCrearClienteGenerico();
  }
}

            id: String(data[i][0]),
            nombre: String(data[i][1] || 'Sin nombre'),
            celular: String(data[i][2] || '-'),
            direccion: String(data[i][3] || '-'),
            notas: String(data[i][4] || ''),
            email: data[i][6] ? String(data[i][6]) : '-',
            rol: data[i][8] ? String(data[i][8]) : 'Cliente'
          }
        };
      }
    }
    
    return { success: false, cliente: null };
    
  } catch (error) {
    Logger.log('Error en obtenerDatosCliente: ' + error.toString());
    return { success: false, cliente: null };
  }
}
